SERVICE DECISION RECORDS (SDR)



SDR-TRT-001: Multi-Model Ensemble for Claim Verification



Decision: Use an ensemble of three verification strategies (Knowledge Graph lookup, external
fact-check API, source-based verification) with weighted voting for claim verdicts.

Rationale: No single verification method is sufficient. Knowledge Graph provides historical
context, external APIs provide third-party validation, and source analysis provides provenance.
Ensemble reduces false positives/negatives.

Alternatives: Single-method verification (rejected — insufficient accuracy); LLM-only verification
(rejected — hallucination risk).

Trade-offs: Higher latency (parallel calls), higher cost (multiple API calls), but significantly higher
accuracy.



SDR-TRT-002: Cryptographic Evidence Chain

Decision: All evidence and verification decisions stored in an append-only, cryptographically
chained ledger using SHA-256 hashing with Merkle tree verification.

Rationale: Editorial integrity requires tamper-proof evidence. Cryptographic chaining provides
non-repudiable audit trail for every verification decision.

Alternatives: Traditional audit tables (rejected — can be modified); blockchain (rejected —
unnecessary decentralization, high cost).

Trade-offs: Additional storage and write latency, but essential for trust.



SDR-TRT-003: Bayesian Confidence with Prior-Based Cold Start



Decision: Use Bayesian confidence scoring with informative priors based on source type and
category for cold-start scenarios, transitioning to evidence-based posteriors as data
accumulates.

Rationale: New sources and claims have no history. Informative priors prevent both over-
trusting and under-trusting unknown sources while allowing evidence to dominate over time.

Alternatives: Frequentist scoring (rejected — poor cold-start behavior); simple averaging
(rejected — doesn't handle uncertainty well).

Trade-offs: Prior specification requires domain expertise, but provides principled uncertainty
quantification.



SDR-TRT-004: Human-in-the-Loop for High-Stakes Decisions



Decision: Automatically escalate stories with CRITICAL risk scores, health/medical claims,
financial information, or political content for human editorial review regardless of confidence
scores.

Rationale: AI confidence alone is insufficient for life-impacting content. Human judgment is the
final arbiter for high-stakes categories.

Alternatives: Fully automated (rejected — too risky); human review for everything (rejected —
not scalable).

Trade-offs: Editorial staffing requirements, but essential for trust and safety.

```go
// services/truth-engine/source-intelligence/internal/domain/reputation.go

package domain



// ReputationEngine calculates and updates source trust scores

type ReputationEngine struct {

      factCheckHistory FactCheckRepository

      correctionTracker CorrectionTracker

    weights           ReputationWeights

}



type ReputationWeights struct {

    FactualAccuracy float64 `json:"factual_accuracy"` // 0.40

    CorrectionRate float64 `json:"correction_rate"` // 0.20

    SourceLongevity float64 `json:"source_longevity"` // 0.10

    OwnershipTransparency float64 `json:"ownership_transparency"` // 0.10

    PeerCitations float64 `json:"peer_citations"`       // 0.10

    ConsistencyScore float64 `json:"consistency_score"` // 0.10

}



// CalculateReputation computes the current reputation score for a source

func (re *ReputationEngine) CalculateReputation(

    ctx context.Context,

    sourceID uuid.UUID,

) (*ReputationScore, error) {



    // Get source

    source, err := re.sourceRepo.GetByID(ctx, sourceID)

    if err != nil {

        return nil, fmt.Errorf("get source: %w", err)

    }



    // Get fact-check history

    history, err := re.factCheckHistory.GetRecent(ctx, sourceID, 200)

if err != nil {

    return nil, fmt.Errorf("get fact check history: %w", err)

}



score := &ReputationScore{

    SourceID: sourceID,

    CalculatedAt: time.Now(),

    DataPoints: len(history),

}



// 1. Factual Accuracy Score

score.FactualAccuracy = re.calculateAccuracy(history)



// 2. Correction Rate

score.CorrectionRate = re.calculateCorrectionRate(ctx, sourceID)



// 3. Source Longevity

score.Longevity = re.calculateLongevity(source)



// 4. Ownership Transparency

score.OwnershipTransparency = re.calculateTransparency(source)



// 5. Peer Citations

score.PeerCitations = re.calculatePeerCitations(ctx, sourceID)



// 6. Consistency Score

score.Consistency = re.calculateConsistency(ctx, sourceID)



// Weighted combination

w := re.weights

score.OverallScore =

    score.FactualAccuracy * w.FactualAccuracy +

    score.CorrectionRate * w.CorrectionRate +

    score.Longevity * w.SourceLongevity +

    score.OwnershipTransparency * w.OwnershipTransparency +

    score.PeerCitations * w.PeerCitations +

    score.Consistency * w.ConsistencyScore



// Apply Bayesian prior when data is sparse

if len(history) < 10 {

    prior := re.getInformativePrior(source)

    weight := float64(len(history)) / 10.0

    score.OverallScore = weight*score.OverallScore + (1-weight)*prior

}



// Determine trust tier

score.TrustTier = re.determineTrustTier(score.OverallScore)



// Calculate confidence interval

score.ConfidenceInterval = re.calculateConfidenceInterval(len(history), score.OverallScore)



return score, nil

}



// calculateAccuracy computes historical factual accuracy with time decay

func (re *ReputationEngine) calculateAccuracy(history []FactCheckResult) float64 {

    if len(history) == 0 {

        return 0

    }



    totalWeight := 0.0

    weightedCorrect := 0.0



    for _, result := range history {

        // Recent results weighted more heavily

        ageInDays := time.Since(result.EvaluatedAt).Hours() / 24

        weight := math.Exp(-ageInDays / 90) // 90-day half-life



        if result.Verdict == VerdictTrue || result.Verdict == VerdictMostlyTrue {

            weightedCorrect += weight

        }

        totalWeight += weight

    }



    if totalWeight == 0 {

        return 0

    }

    return (weightedCorrect / totalWeight) * 100

}



// getInformativePrior returns a prior estimate based on source category

func (re *ReputationEngine) getInformativePrior(source *Source) float64 {

    priors := map[SourceCategory]float64{

        CategoryOfficial:      85.0, // Government, official sources

        CategoryAuthoritative: 80.0, // Major news wires, academic journals

        CategoryEstablished: 65.0, // Established newspapers, broadcasters

        CategoryEmerging:        45.0, // Digital natives, blogs

        CategoryUnverified: 25.0, // Unknown sources

        CategoryDeprecated: 10.0, // Previously unreliable

    }



    if prior, exists := priors[source.Category]; exists {

        return prior

    }

    return 50.0 // Neutral prior for unknown categories

}



type ReputationScore struct {

    SourceID           uuid.UUID `json:"source_id"`

    OverallScore        float64 `json:"overall_score"`

    FactualAccuracy         float64 `json:"factual_accuracy"`

    CorrectionRate          float64 `json:"correction_rate"`

    Longevity          float64 `json:"longevity"`

      OwnershipTransparency float64 `json:"ownership_transparency"`

      PeerCitations         float64 `json:"peer_citations"`

      Consistency           float64 `json:"consistency"`

      TrustTier        TrustTier `json:"trust_tier"`

      ConfidenceInterval float64 `json:"confidence_interval"`

      DataPoints           int      `json:"data_points"`

      CalculatedAt          time.Time `json:"calculated_at"`

}
```

```go
// CheckBlacklist verifies if a source is blacklisted

func (slm *SourceListManager) CheckBlacklist(ctx context.Context, sourceID uuid.UUID)
(*BlacklistCheck, error) {

    entry, err := slm.blacklistRepo.FindActive(ctx, sourceID)

    if err == ErrNotFound {

         return &BlacklistCheck{Blacklisted: false}, nil

    }

    if err != nil {

         return nil, fmt.Errorf("check blacklist: %w", err)

    }



    return &BlacklistCheck{

         Blacklisted: true,

         Reason:     entry.Reason,

         Category: entry.Category,

         IsPermanent: entry.IsPermanent,

         AddedAt:    entry.AddedAt,

      }, nil

}
```

```go
package domain



// SourceAuthenticator verifies the authenticity of news sources

type SourceAuthenticator struct {

type SourceAuthentication struct {

    IsAuthentic     bool          `json:"is_authentic"`

    Confidence       float64        `json:"confidence"`

}



type AuthCheckResult struct {

    CheckName string `json:"check_name"`

    Passed       bool `json:"passed"`

    Confidence float64 `json:"confidence"`

    Detail     string `json:"detail"`

    Evidence string `json:"evidence,omitempty"`

}



// AuthenticateSource performs comprehensive source authentication

func (sa *SourceAuthenticator) AuthenticateSource(

    ctx context.Context,

    source *Source,

) (*SourceAuthentication, error) {



  auth := &SourceAuthentication{

      SourceID: source.ID,

      URL:      source.URL,

      VerifiedAt: time.Now(),

  }



  // Run all authentication checks in parallel

  type checkResult struct {

      name string

      result AuthCheckResult

  }



  results := make(chan checkResult, 6)



  // Check 1: DNS verification

  go func() {

      result := sa.dnsVerifier.Verify(ctx, source.URL)

      results <- checkResult{"dns", result}

  }()



  // Check 2: TLS certificate verification

  go func() {

      result := sa.tlsVerifier.Verify(ctx, source.URL)

      results <- checkResult{"tls", result}

}()



// Check 3: Domain age and history

go func() {

  result := sa.domainAnalyzer.Analyze(ctx, source.URL)

  results <- checkResult{"domain", result}

}()



// Check 4: Impersonation detection

go func() {

  result := sa.impersonationDetector.Check(ctx, source)

  results <- checkResult{"impersonation", result}

}()



// Check 5: WHOIS information

go func() {

  result := sa.whoisChecker.Check(ctx, source.URL)

  results <- checkResult{"whois", result}

}()



// Check 6: Government source verification (if applicable)

go func() {

  if source.SourceType == SourceTypeGovernment {

      result := sa.verifyGovernmentSource(ctx, source)

      results <- checkResult{"government", result}

  }

}()



// Collect results

collectedResults := make(map[string]AuthCheckResult)

for i := 0; i < 6; i++ {

    cr := <-results

    collectedResults[cr.name] = cr.result

    auth.Checks = append(auth.Checks, cr.result)

}



// Determine authentication status

criticalChecks := []string{"dns", "tls", "domain"}

allCriticalPassed := true

totalConfidence := 0.0



for _, name := range criticalChecks {

    if result, exists := collectedResults[name]; exists && !result.Passed {

        allCriticalPassed = false

    }

}



for _, result := range collectedResults {

    totalConfidence += result.Confidence

}



auth.IsAuthentic = allCriticalPassed

      auth.Confidence = totalConfidence / float64(len(collectedResults))



      return auth, nil

}
```

3.2 Impersonation Detection



```go

// ImpersonationDetector identifies fake/clone websites

type ImpersonationDetector struct {

      knownDomains *KnownDomainRegistry

      similarityEngine *DomainSimilarityEngine

}



type ImpersonationCheck struct {

      IsSuspicious    bool     `json:"is_suspicious"`

      RiskLevel      string `json:"risk_level"`     // LOW, MEDIUM, HIGH, CRITICAL

      Impersonating string `json:"impersonating,omitempty"` // Target being impersonated

      Indicators     []string `json:"indicators"`

      Confidence         float64 `json:"confidence"`

}



func (id *ImpersonationDetector) Check(ctx context.Context, source *Source) AuthCheckResult
{

check := ImpersonationCheck{}

indicators := []string{}



// 1. Typosquatting detection

typosquatMatches := id.detectTyposquatting(source.URL)

if len(typosquatMatches) > 0 {

    indicators = append(indicators,

      fmt.Sprintf("URL similar to known domain: %v", typosquatMatches))

    check.Impersonating = typosquatMatches[0]

}



// 2. Domain age check (new domains are suspicious)

domainAge := id.getDomainAge(source.URL)

if domainAge < 30*24*time.Hour { // Less than 30 days

    indicators = append(indicators,

      fmt.Sprintf("Domain registered %d days ago", int(domainAge.Hours()/24)))

}



// 3. WHOIS privacy check

if id.hasPrivateWHOIS(source.URL) && domainAge < 90*24*time.Hour {

    indicators = append(indicators, "Private WHOIS on recently registered domain")

}



// 4. Content similarity to known brands

if source.Name != "" {

    brandSimilarity := id.checkBrandSimilarity(source.Name)

    if brandSimilarity > 0.85 {

        indicators = append(indicators,

          fmt.Sprintf("Name similar to known brand (%.0f%%)", brandSimilarity*100))

    }

}



// 5. SSL certificate anomalies

certAnomalies := id.checkCertificateAnomalies(source.URL)

indicators = append(indicators, certAnomalies...)



check.Indicators = indicators

check.IsSuspicious = len(indicators) >= 3

check.Confidence = math.Min(float64(len(indicators))*0.3, 1.0)



// Determine risk level

switch {

case len(indicators) >= 5:

    check.RiskLevel = "CRITICAL"

case len(indicators) >= 3:

    check.RiskLevel = "HIGH"

case len(indicators) >= 1:

    check.RiskLevel = "MEDIUM"

default:

    check.RiskLevel = "LOW"

}

      return AuthCheckResult{

          CheckName: "impersonation",

          Passed:    !check.IsSuspicious || check.RiskLevel == "LOW",

          Confidence: 1.0 - check.Confidence,

          Detail:   fmt.Sprintf("Impersonation risk: %s (%d indicators)",

            check.RiskLevel, len(indicators)),

          Evidence: strings.Join(indicators, "; "),

      }

}
```

```python
"""

Claim Extraction Pipeline — Decomposes content into atomic, verifiable claims.

"""

from dataclasses import dataclass, field

from typing import List, Optional, Dict, Any

from enum import Enum

import structlog



logger = structlog.get_logger(__name__)




class ClaimType(Enum):

  FACTUAL = "FACTUAL"            # Verifiable statement of fact

  NUMERICAL = "NUMERICAL"             # Contains numbers/statistics

  ATTRIBUTION = "ATTRIBUTION"         # "X said Y"

  TEMPORAL = "TEMPORAL"              # Date/time claim

  LOCATION = "LOCATION"            # Geographic claim

  CAUSAL = "CAUSAL"             # Cause-effect claim

  COMPARATIVE = "COMPARATIVE"              # Comparison claim

  PREDICTIVE = "PREDICTIVE"        # Future prediction

  QUOTATION = "QUOTATION"            # Direct or indirect quote

async def decompose_sentence(

  self,

  sentence: str,

  full_text: str,

) -> List[AtomicClaim]:

  """

  Decompose a sentence into atomic, independently verifiable claims.

  """

  claims = []



  # Extract entities

  entities = await self.ner_model.extract(sentence)



  # Extract relationships (Subject-Predicate-Object triples)

  triples = await self.relation_extractor.extract_triples(sentence)



  for triple in triples:

    # Classify claim type

    claim_types = await self.classify_claim(sentence, triple)



    # Extract numerical values if present

    numerical_values = self.numerical_extractor.extract(sentence)

    if numerical_values and ClaimType.NUMERICAL not in claim_types:

          claim_types.append(ClaimType.NUMERICAL)

# Extract temporal information

temporal = self.extract_temporal_info(sentence)

if temporal and ClaimType.TEMPORAL not in claim_types:

  claim_types.append(ClaimType.TEMPORAL)



# Extract location information

location = self.extract_location_info(sentence)

if location and ClaimType.LOCATION not in claim_types:

  claim_types.append(ClaimType.LOCATION)



# Check for quotations

quote_info = self.quote_extractor.extract(sentence)

if quote_info:

  claim_types.append(ClaimType.QUOTATION if quote_info["is_direct"]

            else ClaimType.ATTRIBUTION)



claim = AtomicClaim(

  claim_id=str(uuid.uuid4()),

  claim_text=sentence.strip(),

  claim_type=claim_types,

  subject=triple.get("subject"),

  predicate=triple.get("predicate"),

  object=triple.get("object"),

  entities=entities,

  temporal_context=temporal,

  location_context=location,

          numerical_values=numerical_values,

          attributed_to=quote_info.get("attributed_to") if quote_info else None,

          attribution_type=quote_info.get("type") if quote_info else None,

          confidence=self.calculate_extraction_confidence(

               sentence, triple, entities

          ),

          source_text_offset=self.find_offset(sentence, full_text),

          language=await self.detect_language(sentence),

async def classify_claim(

  self,

  sentence: str,

  triple: Dict[str, Any],

) -> List[ClaimType]:

  """Classify the type(s) of a claim."""



  types = [ClaimType.FACTUAL] # Base type



  # Check for causal language

  causal_indicators = ["because", "therefore", "as a result", "due to",

                   "caused by", "leads to", "results in"]

  if any(indicator in sentence.lower() for indicator in causal_indicators):

     types.append(ClaimType.CAUSAL)



  # Check for comparative language

  comparative_indicators = ["more than", "less than", "compared to",

                    "higher than", "lower than", "versus"]

  if any(indicator in sentence.lower() for indicator in comparative_indicators):

     types.append(ClaimType.COMPARATIVE)



  # Check for predictive language

  predictive_indicators = ["will", "going to", "expected to", "forecast",

                   "projected", "predicted", "estimate"]

  if any(indicator in sentence.lower() for indicator in predictive_indicators):

     types.append(ClaimType.PREDICTIVE)



  return types



def calculate_extraction_confidence(

  self,

  sentence: str,

  triple: Dict[str, Any],

  entities: List[Dict[str, Any]],

) -> float:

  """Calculate confidence in the extraction quality."""



  confidence = 0.8 # Base confidence

  # Higher confidence if all triple elements present

  if triple.get("subject") and triple.get("predicate") and triple.get("object"):

    confidence += 0.1



  # Higher confidence if entities are well-recognized

  high_conf_entities = sum(1 for e in entities if e.get("confidence", 0) > 0.9)

  if high_conf_entities > 0:

    confidence += 0.05 * min(high_conf_entities, 4)



  # Lower confidence for very short or very long claims

  word_count = len(sentence.split())

  if word_count < 3:

    confidence -= 0.2

  elif word_count > 50:

    confidence -= 0.1



  return max(0.0, min(1.0, confidence))



def deduplicate_claims(self, claims: List[AtomicClaim]) -> List[AtomicClaim]:

  """Remove duplicate claims based on semantic similarity."""

  unique_claims = []

  seen_hashes = set()



  for claim in claims:

    # Create a hash of the core claim elements

            claim_key = f"{claim.subject}|{claim.predicate}|{claim.object}"

            claim_hash = hashlib.sha256(claim_key.encode()).hexdigest()



            if claim_hash not in seen_hashes:

               seen_hashes.add(claim_hash)

               unique_claims.append(claim)



          return unique_claims
```

```python
class QuoteExtractor:

      """Extracts and validates direct and indirect quotations."""



      # Patterns for direct quotes

      DIRECT_QUOTE_PATTERNS = [

          r'"([^"]+)"',              # Double quotes

          r'\'([^\']+)\'',           # Single quotes

          r'"([^"]+)"\s*,\s*(\w+\s+\w+)',    # Quote with attribution

          r'(\w+\s+\w+)\s+said\s*[,:]?\s*"([^"]+)"', # Attribution with quote

      ]



      # Attribution verbs

      ATTRIBUTION_VERBS = [

    "said", "stated", "announced", "declared", "reported",

    "claimed", "asserted", "noted", "added", "explained",

    "according to", "per", "as stated by", "as reported by",

]



def extract(self, text: str) -> Optional[Dict[str, Any]]:

    """

    Extract quotation and attribution from text.

    Returns None if no quotation found.

    """

    # Try direct quote patterns

    for pattern in self.DIRECT_QUOTE_PATTERNS:

      match = re.search(pattern, text, re.IGNORECASE)

      if match:

          quote_text = match.group(1) if len(match.groups()) >= 1 else match.group(2)

          attribution = self.extract_attribution(text)



          return {

              "type": "DIRECT_QUOTE",

              "quote_text": quote_text.strip(),

              "attributed_to": attribution,

              "is_direct": True,

              "is_anonymous": attribution is None or attribution.lower() in [

                   "sources", "anonymous", "unnamed", "officials"

              ],

          }

  # Try indirect quote patterns

  for verb in self.ATTRIBUTION_VERBS:

    pattern = rf'(\w[\w\s]{{2,50}})\s+{verb}\s+that\s+(.{{10,}})'

    match = re.search(pattern, text, re.IGNORECASE)

    if match:

       return {

           "type": "INDIRECT_QUOTE",

           "quote_text": match.group(2).strip(),

           "attributed_to": match.group(1).strip(),

           "is_direct": False,

           "is_anonymous": False,

       }



  return None



def extract_attribution(self, text: str) -> Optional[str]:

  """Extract who a quote is attributed to."""

  for verb in self.ATTRIBUTION_VERBS:

    # Pattern: "PERSON said"

    pattern = rf'([\w\s]{{2,50}})\s+{verb}'

    match = re.search(pattern, text, re.IGNORECASE)

    if match:

       return match.group(1).strip()



  return None
```

```go
    verification := &ClaimVerification{

        ClaimID: claim.ClaimID,

    }



    // Run all verification methods in parallel

    type methodResult struct {

        name string

        result *VerificationMethodResult

    }



    results := make(chan methodResult, 4)

// Method 1: Knowledge Graph lookup

go func() {

  result := cv.verifyAgainstKnowledgeGraph(ctx, claim)

  results <- methodResult{"knowledge_graph", result}

}()



// Method 2: External fact-check APIs

go func() {

  result := cv.verifyAgainstExternalAPIs(ctx, claim)

  results <- methodResult{"external_apis", result}

}()



// Method 3: Source-based analysis

go func() {

  result := cv.verifyAgainstSources(ctx, claim, story)

  results <- methodResult{"source_analysis", result}

}()



// Method 4: Context matching

go func() {

  result := cv.verifyContextMatch(ctx, claim, story)

  results <- methodResult{"context_match", result}

}()



// Collect results

collectedResults := make(map[string]*VerificationMethodResult)

    for i := 0; i < 4; i++ {

        mr := <-results

        collectedResults[mr.name] = mr.result

        verification.MethodResults = append(verification.MethodResults, mr.result)

    }



    // Ensemble verdict

    verification.Verdict, verification.Confidence = cv.ensembleVerdict(collectedResults)



    // Aggregate evidence

    verification.SupportingEvidence = cv.aggregateSupportingEvidence(collectedResults)

    verification.ContradictingEvidence = cv.aggregateContradictingEvidence(collectedResults)



    // Store evidence

    cv.evidenceStore.Store(ctx, verification)



    return verification, nil

}



// ensembleVerdict combines multiple verification methods into a final verdict

func (cv *ClaimVerifier) ensembleVerdict(

    results map[string]*VerificationMethodResult,

) (Verdict, float64) {



    // Convert each verdict to a numerical score (-1 to 1)

    scores := make(map[string]float64)

    totalWeight := 0.0



    for name, result := range results {

        weight := cv.getMethodWeight(name)

        numericScore := cv.verdictToScore(result.Verdict)

        scores[name] = numericScore * weight * result.Confidence

        totalWeight += weight

    }



    if totalWeight == 0 {

        return VerdictUnverifiable, 0.0

    }



    // Weighted average

    aggregateScore := 0.0

    for _, score := range scores {

        aggregateScore += score

    }

    aggregateScore /= totalWeight



    // Convert back to verdict

    verdict := cv.scoreToVerdict(aggregateScore)

    confidence := math.Abs(aggregateScore)



    return verdict, confidence

}

// verdictToScore converts a verdict to a numerical score

func (cv *ClaimVerifier) verdictToScore(verdict Verdict) float64 {

      scores := map[Verdict]float64{

          VerdictTrue:       1.0,

          VerdictMostlyTrue: 0.6,

          VerdictHalfTrue:     0.1,

          VerdictMostlyFalse: -0.4,

          VerdictFalse:      -1.0,

          VerdictUnverifiable: 0.0,

          VerdictMisleading: -0.7,

          VerdictOutOfContext: -0.5,

      }



      if score, exists := scores[verdict]; exists {

          return score

      }

      return 0.0

}
```

```go
// ValidateCrossSource checks claim against multiple independent sources

func (csv *CrossSourceValidator) ValidateCrossSource(

  ctx context.Context,

  claim *AtomicClaim,

  sources []*SourceReference,

) (*CrossValidationResult, error) {



  result := &CrossValidationResult{

      ClaimID: claim.ClaimID,

  }



  // 1. Find all sources reporting on this claim

  reportingSources, err := csv.findReportingSources(ctx, claim)

  if err != nil {

      return nil, fmt.Errorf("find reporting sources: %w", err)

  }



  // 2. Determine source independence

  // Sources that share parent companies or syndication are not independent

  independentGroups := csv.groupIndependentSources(ctx, reportingSources)



  // 3. For each independent group, evaluate support

  supporting := 0

  conflicting := 0

  var sourceDetails []SourceValidation

for _, group := range independentGroups {

    detail := csv.evaluateSourceGroup(ctx, group, claim)

    sourceDetails = append(sourceDetails, detail)



    if detail.SupportsClaim {

        supporting++

    } else if detail.ContradictsClaim {

        conflicting++

    }

}



result.IndependentSources = supporting + conflicting

result.ConflictingSources = conflicting

result.SourceDetails = sourceDetails



// 4. Calculate corroboration score

if result.IndependentSources > 0 {

    result.CorroborationScore =

        float64(supporting) / float64(result.IndependentSources) * 100

}



// 5. Determine consensus level

result.ConsensusLevel = csv.determineConsensus(supporting, conflicting)

result.ConsensusConfidence = csv.calculateConsensusConfidence(

    supporting, conflicting, len(sourceDetails),

    return result, nil

}



// groupIndependentSources identifies truly independent sources using graph analysis

func (csv *CrossSourceValidator) groupIndependentSources(

    ctx context.Context,

    sources []*Source,

) [][]*Source {



    // Use Neo4j to find ownership and syndication relationships

    // Sources connected within 2 hops are NOT independent

    query := `

        MATCH (s1:Source {source_id: $id1}), (s2:Source {source_id: $id2})

        OPTIONAL MATCH path = shortestPath((s1)-[:OWNS|SYNDICATES|CITES*1..2]-(s2))

        RETURN s1.source_id as source1, s2.source_id as source2,

            CASE WHEN path IS NULL THEN true ELSE false END as independent

    `



    // Build independence matrix and group using union-find

    uf := NewUnionFind(len(sources))



    for i, sourceA := range sources {

        for j := i + 1; j < len(sources); j++ {

          sourceB := sources[j]

        result, _ := csv.sourceGraph.Run(ctx, query, map[string]interface{}{

             "id1": sourceA.ID,

             "id2": sourceB.ID,

        })



        if result.Record().Values[2].(bool) {

             // Sources are independent — don't union

             continue

        }



        // Sources are related — union them

        uf.Union(i, j)

    }

}



// Group by union-find root

groups := make(map[int][]*Source)

for i, source := range sources {

    root := uf.Find(i)

    groups[root] = append(groups[root], source)

}



var result [][]*Source

for _, group := range groups {

    result = append(result, group)

      }



      return result

}
```

```go
// CalculateStoryConfidence computes the confidence score for a story

  sourceScores []*ReputationScore,

confidence.DimensionalScores.CompletenessScore = ce.calculateCompleteness(verifications)

// calculateSourceTrust aggregates source reputation scores

func (ce *ConfidenceEngine) calculateSourceTrust(scores []*ReputationScore) float64 {

    if len(scores) == 0 {

        return 50.0 // Neutral

    }



    // Weight by source count (more sources = more reliable)

    totalWeight := 0.0

    weightedScore := 0.0



    for _, score := range scores {

        weight := score.OverallScore / 100 // Higher score sources weight more

        weightedScore += score.OverallScore * weight

        totalWeight += weight

    }



    if totalWeight == 0 {

        return 50.0

    }



    return weightedScore / totalWeight

}

// calculateClaimAccuracy aggregates claim verification results

func (ce *ConfidenceEngine) calculateClaimAccuracy(verifications []*ClaimVerification) float64 {

    if len(verifications) == 0 {

        return 50.0

    }



    totalConfidence := 0.0

    count := 0



    for _, v := range verifications {

        if v.Verdict == VerdictUnverifiable {

            continue // Skip unverifiable claims in accuracy calculation

        }

        totalConfidence += v.Confidence * 100

        count++

    }



    if count == 0 {

        return 50.0

    }



    return totalConfidence / float64(count)

}



// bootstrapConfidenceInterval estimates confidence interval via bootstrap resampling

func (ce *ConfidenceEngine) bootstrapConfidenceInterval(

  verifications []*ClaimVerification,

  iterations int,

) ConfidenceInterval {



  if len(verifications) == 0 {

      return ConfidenceInterval{Lower: 0, Upper: 100, Width: 100}

  }



  scores := make([]float64, len(verifications))

  for i, v := range verifications {

      scores[i] = v.Confidence

  }



  // Bootstrap resampling

  means := make([]float64, iterations)

  rng := rand.New(rand.NewSource(time.Now().UnixNano()))



  for i := 0; i < iterations; i++ {

      // Sample with replacement

      sample := make([]float64, len(scores))

      for j := range sample {

          sample[j] = scores[rng.Intn(len(scores))]

      }



      // Calculate mean

          sum := 0.0

          for _, s := range sample {

              sum += s

          }

          means[i] = sum / float64(len(sample))

      }



      // Sort and get percentiles

      sort.Float64s(means)

      lower := means[int(float64(iterations)*0.05)]

      upper := means[int(float64(iterations)*0.95)]



      return ConfidenceInterval{

          Lower: lower * 100,

          Upper: upper * 100,

          Width: (upper - lower) * 100,

      }

}
```

```go
// generateRecommendation determines what should happen with the story

    thresholds := DefaultThresholds()



    // Check if breaking news (lower threshold)

    if isBreaking {
```

```go
    ctx context.Context,

) (*EditorialDecision, error) {



    // Determine review priority

    priority := ede.calculateReviewPriority(story, decision.Confidence)



    // Create review task

        TaskID: uuid.NewV7().String(),

        Reason: decision.Reason,

        Deadline: ede.calculateDeadline(priority),

        StorySummary: story.Summary,

        ConfidenceSummary: decision.Confidence,

        CreatedAt: time.Now(),

    }

        return nil, fmt.Errorf("submit review task: %w", err)

    }

func (ede *EditorialDecisionEngine) calculateReviewPriority(

    story *CanonicalStory,

    confidence *StoryConfidence,

) ReviewPriority {



    // Breaking news always critical

    if story.Priority == PriorityBreaking {

        return ReviewPriorityCritical

    }



    // High risk = critical

    if confidence.RiskLevel == RiskLevelCritical || confidence.RiskLevel == RiskLevelHigh {

        return ReviewPriorityCritical

    }



    // Sensitive categories = high

    sensitiveCategories := []string{"HEALTH", "FINANCIAL", "POLITICAL", "EMERGENCY"}

    for _, cat := range sensitiveCategories {

        if story.Category == cat {

            return ReviewPriorityHigh

        }

    }



    return ReviewPriorityMedium

}



func (ede *EditorialDecisionEngine) calculateDeadline(priority ReviewPriority) time.Time {

    now := time.Now()



    deadlines := map[ReviewPriority]time.Duration{

        ReviewPriorityCritical: 5 * time.Minute,

        ReviewPriorityHigh:     30 * time.Minute,

        ReviewPriorityMedium: 2 * time.Hour,

        ReviewPriorityLow:      8 * time.Hour,

      }



      if deadline, exists := deadlines[priority]; exists {

          return now.Add(deadline)

      }

      return now.Add(4 * time.Hour)

}
```

```go
// ChainHasher calculates SHA-256 hashes for ledger chain integrity

type ChainHasher struct{}



func (ch *ChainHasher) CalculateHash(entry *LedgerEntry) string {

    // Build canonical representation for hashing

    data := fmt.Sprintf("%s|%s|%s|%s|%s|%s|%s|%s",

        entry.EntryID,

        entry.PreviousHash,

        entry.Timestamp.Format(time.RFC3339Nano),

        entry.Action,

        entry.Actor.ActorID,

        entry.Target.TargetType,

        entry.Target.TargetID,

        entry.AfterState,
```

```
 CONSENSUS: Retry 2x with expanded source search

 CONFIDENCE: Recalculate if new evidence arrives within 24h

```



10.2 State Machine Implementation

```go

// services/truth-engine/editorial-validator/internal/domain/state_machine.go

package domain



type TruthState string



const (

    TruthStateNew          TruthState = "NEW"

    TruthStateDiscovered       TruthState = "DISCOVERED"

    TruthStateSourceVerified TruthState = "SOURCE_VERIFIED"

    TruthStateClaimsExtracted TruthState = "CLAIMS_EXTRACTED"

    TruthStateFactChecking      TruthState = "FACT_CHECKING"

    TruthStateConsensus        TruthState = "CONSENSUS"

    TruthStateConfidenceScoring TruthState = "CONFIDENCE_SCORING"

    TruthStateEditorialDecision TruthState = "EDITORIAL_DECISION"

    TruthStateApproved        TruthState = "APPROVED"

    TruthStateRejected        TruthState = "REJECTED"

    TruthStateHumanReview        TruthState = "HUMAN_REVIEW"

    TruthStatePublished       TruthState = "PUBLISHED"

    TruthStateFailed       TruthState = "FAILED"

)



type TruthStateMachine struct {

    story   *CanonicalStory

    currentState TruthState

    transitions map[TruthState]map[TruthEvent]TruthState

    ledger   *TruthLedger

}



type TruthEvent string



const (

    EventStoryReceived       TruthEvent = "STORY_RECEIVED"

    EventSourcesVerified     TruthEvent = "SOURCES_VERIFIED"

    EventClaimsExtracted     TruthEvent = "CLAIMS_EXTRACTED"

    EventFactCheckComplete TruthEvent = "FACT_CHECK_COMPLETE"

    EventFactCheckFailed     TruthEvent = "FACT_CHECK_FAILED"

    EventConsensusReached TruthEvent = "CONSENSUS_REACHED"

    EventConsensusFailed      TruthEvent = "CONSENSUS_FAILED"

    EventConfidenceCalculated TruthEvent = "CONFIDENCE_CALCULATED"

    EventAutoApproved         TruthEvent = "AUTO_APPROVED"

    EventAutoRejected        TruthEvent = "AUTO_REJECTED"

    EventHumanReviewRequired TruthEvent = "HUMAN_REVIEW_REQUIRED"

    EventHumanApproved         TruthEvent = "HUMAN_APPROVED"

    EventHumanRejected        TruthEvent = "HUMAN_REJECTED"

    EventPublished         TruthEvent = "PUBLISHED"

    EventTimeout           TruthEvent = "TIMEOUT"

    EventMisinfoCritical    TruthEvent = "MISINFO_CRITICAL"

    EventRetryExhausted      TruthEvent = "RETRY_EXHAUSTED"

)

func NewTruthStateMachine(story *CanonicalStory, ledger *TruthLedger) *TruthStateMachine {

  sm := &TruthStateMachine{

      story:      story,

      currentState: TruthStateNew,

      ledger:      ledger,

  }



  sm.transitions = map[TruthState]map[TruthEvent]TruthState{

      TruthStateNew: {

           EventStoryReceived: TruthStateDiscovered,

      },

      TruthStateDiscovered: {

           EventSourcesVerified: TruthStateSourceVerified,

           EventTimeout:      TruthStateHumanReview,

      },

      TruthStateSourceVerified: {

           EventClaimsExtracted: TruthStateClaimsExtracted,

           EventTimeout:      TruthStateHumanReview,

      },

      TruthStateClaimsExtracted: {

           EventFactCheckComplete: TruthStateFactChecking,

           EventFactCheckFailed: TruthStateHumanReview,

      },

      TruthStateFactChecking: {

           EventFactCheckComplete: TruthStateConsensus,

           EventFactCheckFailed: TruthStateFactChecking, // Retry

     EventRetryExhausted:    TruthStateHumanReview,

     EventMisinfoCritical: TruthStateRejected,

},

TruthStateConsensus: {

     EventConsensusReached: TruthStateConfidenceScoring,

     EventConsensusFailed:   TruthStateConsensus, // Retry

     EventRetryExhausted:    TruthStateHumanReview,

},

TruthStateConfidenceScoring: {

     EventConfidenceCalculated: TruthStateEditorialDecision,

},

TruthStateEditorialDecision: {

     EventAutoApproved:       TruthStateApproved,

     EventAutoRejected:      TruthStateRejected,

     EventHumanReviewRequired: TruthStateHumanReview,

},

TruthStateHumanReview: {

     EventHumanApproved: TruthStateApproved,

     EventHumanRejected: TruthStateRejected,

     EventTimeout:     TruthStateHumanReview, // Escalate

},

TruthStateApproved: {

     EventPublished: TruthStatePublished,

},

TruthStateRejected: {

     // Terminal state — can be resubmitted

        },

        TruthStateFailed: {

             EventStoryReceived: TruthStateDiscovered, // Retry from beginning

        },

    }



    return sm

}



func (sm *TruthStateMachine) Transition(event TruthEvent, metadata interface{}) error {

    validEvents, exists := sm.transitions[sm.currentState]

    if !exists {

        return fmt.Errorf("no transitions from state: %s", sm.currentState)

    }



    nextState, valid := validEvents[event]

    if !valid {

        return fmt.Errorf("invalid transition: %s -[%s]-> ?", sm.currentState, event)

    }



    // Record in immutable ledger

    sm.ledger.RecordEntry(context.Background(),

        LedgerAction(fmt.Sprintf("STATE_%s", event)),

        LedgerActor{ActorID: "system", ActorType: "SYSTEM", Name: "Truth State Machine"},

        LedgerTarget{TargetType: "STORY", TargetID: sm.story.ID.String()},

        map[string]interface{}{"from_state": sm.currentState},

          map[string]interface{}{"to_state": nextState, "event": event},

          metadata,

      )



      sm.currentState = nextState

      sm.story.TruthState = string(nextState)

      sm.story.UpdatedAt = time.Now()



      return nil

}
```
