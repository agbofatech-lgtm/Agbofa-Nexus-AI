SERVICE DECISION RECORDS (SDR)



SDR-ANA-001: Unified Analytics Data Lake with Iceberg



Decision: All analytics events flow into S3 via Kafka Connect (Iceberg sink), with Apache Iceberg
as the table format. Real-time analytics use ClickHouse; batch processing uses Apache Spark.
PostgreSQL stores operational metadata and pre-computed aggregations.

Rationale: Iceberg provides schema evolution, time travel, and partition evolution — essential
for evolving analytics schemas. ClickHouse provides sub-second query performance for
dashboards. Spark handles complex aggregations and ML training.

Alternatives: Snowflake (rejected — vendor lock-in, higher cost at scale); BigQuery (rejected —
GCP dependency); ClickHouse-only (rejected — limited for complex batch ML pipelines).

Trade-offs: Multiple query engines to manage, but each optimized for its workload. Iceberg
provides unified storage layer.



SDR-ANA-002: Real-Time + Batch Lambda Architecture

Decision: Implement a Lambda architecture with real-time stream processing (Kafka Streams →
ClickHouse) for sub-second dashboards and alerts, and batch processing (Spark on Iceberg) for
deep analytics, ML training, and historical reporting.

Rationale: Breaking news analytics require sub-second latency. ML training requires hours of
historical data. Lambda architecture serves both needs with appropriate tools.

Alternatives: Kappa architecture (streaming only — rejected, batch still needed for ML); pure
batch (rejected — too slow for real-time dashboards).

Trade-offs: Two code paths for similar logic, but necessary for latency requirements.



SDR-ANA-003: Feature Store for ML Features



Decision: Implement a centralized Feature Store (Feast) for all ML features used across
recommendation, personalization, and optimization models. Features are computed in batch
and served in real-time via Redis.

Rationale: Consistent feature definitions across training and serving prevent training-serving
skew. Feature reuse across models reduces duplication.

Alternatives: Inline feature computation per model (rejected — inconsistency, duplication);
custom feature store (rejected — reinventing the wheel).

Trade-offs: Additional infrastructure (Feast + Redis), but significant improvement in ML quality
and velocity.



SDR-ANA-004: Closed-Loop AI Optimization



Decision: All content performance data flows back to AI agents via structured feedback events
on Kafka. Each AI agent subscribes to performance metrics for content it generated.
Optimization signals trigger prompt refinement, model selection changes, and parameter
tuning.

Rationale: AI-generated content quality improves only with feedback. Closed-loop enables
continuous improvement without manual intervention.

Alternatives: Manual feedback collection (rejected — doesn't scale); periodic batch updates
(rejected — too slow for breaking news optimization).

Trade-offs: Risk of feedback loops amplifying biases, mitigated by diversity constraints and
human oversight.

```go
// EventCollector ingests, validates, and normalizes all analytics events

type EventCollector struct {

// AnalyticsEvent is the base type for all analytics events

type AnalyticsEvent struct {

    EventVersion string           `json:"event_version"`

// Event types

const (

    EventTypeStoryImpression = "story.impression"

    EventTypeStoryClick      = "story.click"

    EventTypeStoryEngagement = "story.engagement"

    EventTypeStoryShare       = "story.share"

    EventTypeStoryComment        = "story.comment"

    EventTypeStoryReadComplete = "story.read_complete"

    EventTypeAIGeneration      = "ai.generation"

    EventTypeAIQuality       = "ai.quality"

    EventTypeAIOptimization     = "ai.optimization"

    EventTypeDistribution     = "distribution.publish"

    EventTypeRevenue          = "revenue.impression"

    EventTypeRevenueClick      = "revenue.click"

    EventTypeSubscription     = "revenue.subscription"

    EventTypeUserSession        = "user.session"

    EventTypeUserSignup         = "user.signup"

    EventTypeSearchQuery         = "search.query"

    EventTypeSearchClick       = "search.click"

)



// ProcessEvent validates and enriches an incoming event

func (ec *EventCollector) ProcessEvent(

    ctx context.Context,

    rawEvent []byte,

) error {



    // 1. Parse event

    var event AnalyticsEvent

    if err := json.Unmarshal(rawEvent, &event); err != nil {

        return fmt.Errorf("parse event: %w", err)

    }



    // 2. Validate against schema

    if err := ec.schemaRegistry.Validate(event.EventType, event.EventVersion, rawEvent); err != nil
{

        return fmt.Errorf("schema validation: %w", err)

    }



    // 3. Deduplicate (idempotency)

    if ec.deduplicator.IsDuplicate(event.EventID) {

        return nil // Already processed

    }



    // 4. Normalize and enrich

    enriched := ec.normalizer.Enrich(ctx, &event)



    // 5. Write to ClickHouse for real-time querying

    if err := ec.writeToClickHouse(ctx, enriched); err != nil {

        return fmt.Errorf("clickhouse write: %w", err)

    }



    // 6. Forward to Kafka for downstream consumers

    if err := ec.forwardToKafka(ctx, enriched); err != nil {

        return fmt.Errorf("kafka forward: %w", err)

    }



    // 7. Mark as processed

    ec.deduplicator.MarkProcessed(event.EventID)



    return nil

}



// writeToClickHouse inserts the event into ClickHouse for real-time analytics

func (ec *EventCollector) writeToClickHouse(ctx context.Context, event *AnalyticsEvent) error {

    return ec.clickhouse.Exec(ctx, `

        INSERT INTO analytics_events (

             event_id, event_type, event_version, tenant_id, story_id,

             content_id, platform, user_id, session_id, timestamp,

             ingested_at, metrics, dimensions, metadata

           ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)

      `,

           event.EventID, event.EventType, event.EventVersion, event.TenantID,

           event.StoryID, event.ContentID, event.Platform, event.UserID,

           event.SessionID, event.Timestamp, event.IngestedAt,

           event.Metrics, event.Dimensions, event.Metadata,

      )

}
```

2.2 Event Normalization



```go

// EventNormalizer enriches events with derived data

type EventNormalizer struct {

      geoIP        *GeoIPResolver

      userAgent *UserAgentParser

      tenantCache *redis.Client

}



func (en *EventNormalizer) Enrich(ctx context.Context, event *AnalyticsEvent) *AnalyticsEvent {

      // Add geo information from IP

      if ip, exists := event.Dimensions["ip_address"]; exists {

      geo := en.geoIP.Resolve(ip)

      event.Dimensions["country"] = geo.Country

      event.Dimensions["region"] = geo.Region

      event.Dimensions["city"] = geo.City

      event.Dimensions["timezone"] = geo.Timezone

  }



  // Parse user agent

  if ua, exists := event.Dimensions["user_agent"]; exists {

      parsed := en.userAgent.Parse(ua)

      event.Dimensions["browser"] = parsed.Browser

      event.Dimensions["browser_version"] = parsed.BrowserVersion

      event.Dimensions["os"] = parsed.OS

      event.Dimensions["device_type"] = parsed.DeviceType // mobile, desktop, tablet

  }



  // Add tenant metadata from cache

  if tenantMeta, err := en.tenantCache.Get(ctx, fmt.Sprintf("tenant:%s",
event.TenantID)).Result(); err == nil {

      var meta map[string]string

      json.Unmarshal([]byte(tenantMeta), &meta)

      event.Dimensions["tenant_tier"] = meta["tier"]

      event.Dimensions["tenant_region"] = meta["region"]

  }



  return event

}
```

2.3 ClickHouse Schema



```sql

-- ClickHouse: Real-time Analytics Events Table

CREATE TABLE analytics_events (

      -- Identifiers

      event_id String,

      event_type LowCardinality(String),

      event_version String,

      tenant_id String,

      story_id String,

      content_id String,

      platform LowCardinality(String),

      user_id String,

      session_id String,



      -- Timestamps

      timestamp DateTime64(3),

      ingested_at DateTime64(3) DEFAULT now64(3),



      -- Metrics (flexible key-value)

      metrics Map(String, Float64),

  -- Dimensions (flexible key-value)

  dimensions Map(String, String),



  -- Additional metadata

  metadata String DEFAULT '{}', -- JSON



  -- Indexes

  INDEX idx_story story_id TYPE bloom_filter GRANULARITY 1,

  INDEX idx_event_type event_type TYPE set(0) GRANULARITY 1,

  INDEX idx_platform platform TYPE set(0) GRANULARITY 1

) ENGINE = MergeTree()

PARTITION BY toYYYYMM(timestamp)

ORDER BY (tenant_id, event_type, platform, timestamp)

TTL timestamp + INTERVAL 90 DAY DELETE

SETTINGS index_granularity = 8192;



-- Materialized View: Hourly Story Metrics

CREATE MATERIALIZED VIEW story_metrics_hourly

ENGINE = SummingMergeTree()

PARTITION BY toYYYYMM(hour)

ORDER BY (tenant_id, story_id, platform, hour)

AS SELECT

  tenant_id,

  story_id,

  platform,

  toStartOfHour(timestamp) as hour,

  countIf(event_type = 'story.impression') as impressions,

  countIf(event_type = 'story.click') as clicks,

  countIf(event_type = 'story.engagement') as engagements,

  countIf(event_type = 'story.share') as shares,

  countIf(event_type = 'story.comment') as comments,

  countIf(event_type = 'story.read_complete') as read_completions,

  sum(metrics['read_depth']) as total_read_depth,

  sum(metrics['time_on_page']) as total_time_on_page,

  sum(metrics['revenue']) as total_revenue

FROM analytics_events

WHERE event_type LIKE 'story.%'

GROUP BY tenant_id, story_id, platform, hour;



-- Materialized View: AI Performance Metrics

CREATE MATERIALIZED VIEW ai_performance_hourly

ENGINE = SummingMergeTree()

PARTITION BY toYYYYMM(hour)

ORDER BY (tenant_id, ai_agent_id, model, hour)

AS SELECT

  tenant_id,

  dimensions['ai_agent_id'] as ai_agent_id,

  dimensions['model'] as model,

  dimensions['content_type'] as content_type,

  toStartOfHour(timestamp) as hour,

  count() as total_generations,

  countIf(event_type = 'ai.quality' AND metrics['passed'] = 1) as passed_generations,

      avg(metrics['quality_score']) as avg_quality_score,

      avg(metrics['generation_time_ms']) as avg_generation_time_ms,

      sum(metrics['tokens_used']) as total_tokens,

      sum(metrics['cost_usd']) as total_cost_usd

FROM analytics_events

WHERE event_type LIKE 'ai.%'

GROUP BY tenant_id, ai_agent_id, model, content_type, hour;
```

```go
// services/analytics/realtime-analytics/internal/domain/story_performance.go

package domain



// StoryPerformanceAnalyzer computes real-time story metrics

type StoryPerformanceAnalyzer struct {

    clickhouse *clickhouse.Client

    redis     *redis.Client

}



type StoryPerformance struct {

    StoryID         string             `json:"story_id"`

    Headline         string             `json:"headline"`

    PublishedAt       time.Time             `json:"published_at"`



    // Aggregate Metrics

    TotalImpressions int64                  `json:"total_impressions"`

    TotalClicks     int64               `json:"total_clicks"`

    TotalShares       int64              `json:"total_shares"`

    ShareRate        float64             `json:"share_rate"`

    AvgReadDepth         float64            `json:"avg_read_depth"`     // 0-100%

    AvgTimeOnPage           float64          `json:"avg_time_on_page"` // seconds

    CompletionRate       float64            `json:"completion_rate"`



    // Platform Breakdown

    PlatformMetrics map[string]*PlatformStoryMetrics `json:"platform_metrics"`



    // Velocity

    EngagementsPerMinute float64                 `json:"engagements_per_minute"`

    Trend          string             `json:"trend"`       // rising, stable, falling

    ViralScore      float64             `json:"viral_score"`    // 0-100



    // Revenue

    TotalRevenue        float64           `json:"total_revenue"`

    RevenuePerImpression float64                `json:"revenue_per_impression"`



    // AI Attribution

    AIAgentID        string             `json:"ai_agent_id,omitempty"`

    AIModel         string             `json:"ai_model,omitempty"`

    AIGenerationScore float64                `json:"ai_generation_score,omitempty"`

}



// GetStoryPerformance retrieves comprehensive story performance metrics

func (spa *StoryPerformanceAnalyzer) GetStoryPerformance(

    ctx context.Context,

  storyID string,

  timeWindow time.Duration,

) (*StoryPerformance, error) {



  // Try cache first

  cacheKey := fmt.Sprintf("story_perf:%s:%s", storyID, timeWindow)

  if cached, err := spa.redis.Get(ctx, cacheKey).Result(); err == nil {

      var perf StoryPerformance

      json.Unmarshal([]byte(cached), &perf)

      return &perf, nil

  }



  // Query ClickHouse

  query := `

      SELECT

        story_id,

        platform,

        countIf(event_type = 'story.impression') as impressions,

        countIf(event_type = 'story.click') as clicks,

        countIf(event_type = 'story.engagement') as engagements,

        countIf(event_type = 'story.share') as shares,

        countIf(event_type = 'story.comment') as comments,

        countIf(event_type = 'story.read_complete') as completions,

        avg(metrics['read_depth']) as avg_read_depth,

        avg(metrics['time_on_page']) as avg_time,

        sum(metrics['revenue']) as revenue

    FROM analytics_events

    WHERE story_id = ?

    AND timestamp >= now() - INTERVAL ?

    GROUP BY story_id, platform

`



rows, err := spa.clickhouse.Query(ctx, query, storyID, timeWindow)

if err != nil {

    return nil, fmt.Errorf("clickhouse query: %w", err)

}

defer rows.Close()



perf := &StoryPerformance{

    StoryID:      storyID,

    PlatformMetrics: make(map[string]*PlatformStoryMetrics),

}



for rows.Next() {

    var platform string

    var impressions, clicks, engagements, shares, comments, completions int64

    var avgReadDepth, avgTime, revenue float64



    rows.Scan(&perf.StoryID, &platform, &impressions, &clicks, &engagements,

      &shares, &comments, &completions, &avgReadDepth, &avgTime, &revenue)



    pm := &PlatformStoryMetrics{

    Platform:        platform,

    Impressions:       impressions,

    Clicks:        clicks,

    Engagements:         engagements,

    Shares:         shares,

    Comments:          comments,

    ReadCompletions: completions,

    AvgReadDepth:        avgReadDepth,

    AvgTimeOnPage:           avgTime,

    Revenue:          revenue,

}



if impressions > 0 {

    pm.CTR = float64(clicks) / float64(impressions) * 100

    pm.EngagementRate = float64(engagements) / float64(impressions) * 100

}



perf.PlatformMetrics[platform] = pm



// Aggregate totals

perf.TotalImpressions += impressions

perf.TotalClicks += clicks

perf.TotalEngagements += engagements

perf.TotalShares += shares

perf.TotalComments += comments

perf.TotalReadCompletions += completions

        perf.TotalRevenue += revenue

    }



    // Calculate derived metrics

    if perf.TotalImpressions > 0 {

        perf.CTR = float64(perf.TotalClicks) / float64(perf.TotalImpressions) * 100

        perf.EngagementRate = float64(perf.TotalEngagements) / float64(perf.TotalImpressions) *
100

        perf.ShareRate = float64(perf.TotalShares) / float64(perf.TotalImpressions) * 100

        perf.RevenuePerImpression = perf.TotalRevenue / float64(perf.TotalImpressions)

    }



    // Calculate velocity and viral score

    perf.EngagementsPerMinute = spa.calculateVelocity(ctx, storyID)

    perf.ViralScore = spa.calculateViralScore(perf)

    perf.Trend = spa.determineTrend(ctx, storyID, timeWindow)



    // Cache for 5 minutes

    data, _ := json.Marshal(perf)

    spa.redis.Set(ctx, cacheKey, data, 5*time.Minute)



    return perf, nil

}



// calculateViralScore computes a viral score based on engagement velocity and sharing
patterns

func (spa *StoryPerformanceAnalyzer) calculateViralScore(perf *StoryPerformance) float64 {

weights := map[string]float64{

    "share_velocity": 0.35,

    "engagement_rate": 0.25,

    "growth_rate":     0.20,

    "external_referrers": 0.20,

}



score := 0.0



// Share velocity (shares per impression)

if perf.TotalImpressions > 0 {

    shareRate := float64(perf.TotalShares) / float64(perf.TotalImpressions)

    score += math.Min(shareRate*1000, 100) * weights["share_velocity"]

}



// Engagement rate

score += perf.EngagementRate * weights["engagement_rate"]



// Growth rate (velocity trend)

if perf.EngagementsPerMinute > 10 {

    score += 80 * weights["growth_rate"]

} else if perf.EngagementsPerMinute > 1 {

    score += 40 * weights["growth_rate"]

}



return math.Min(100, score)

}
```

```python
    self,

    user_id: str,

  preferences: ContentPreferenceProfile,

  """Classify a user into audience segments."""



  assignments = []



  for segment_name, definition in self.SEGMENT_DEFINITIONS.items():

     score = self._match_criteria(behavior, preferences, definition["criteria"])

     if score > 0.5:

          assignments.append(SegmentAssignment(

               segment_name=segment_name,

               score=score,

               priority=definition["priority"],

               description=definition["description"],

          ))



  return sorted(assignments, key=lambda x: x.score, reverse=True)



def _match_criteria(

  self,

  behavior: BehaviorProfile,

  preferences: ContentPreferenceProfile,

  criteria: dict,

) -> float:

  """Calculate how well a user matches segment criteria."""

      total_weight = 0.0

      matched_weight = 0.0



      for criterion, expected_range in criteria.items():

        actual_value = self._get_criterion_value(criterion, behavior, preferences)

        if actual_value is not None:

          weight = 1.0

          if isinstance(expected_range, tuple) and len(expected_range) == 2:

             low, high = expected_range

             if low <= actual_value <= high:

               matched_weight += weight

             else:

               # Partial match based on proximity

               distance = min(abs(actual_value - low), abs(actual_value - high))

               max_distance = max(abs(low), abs(high))

               if max_distance > 0:

                 partial = 1.0 - (distance / max_distance)

                 matched_weight += weight * max(0, partial)

          total_weight += weight



      if total_weight == 0:

        return 0.0

      return matched_weight / total_weight
```

```go
    &ContentBasedFiltering{weight: 0.25},

    &FreshnessBoost{weight: 0.10},

    &PersonalizedRelevance{weight: 0.10},

    &ExplorationDiversification{weight: 0.10},

}

var allCandidates []*Recommendation

var mu sync.Mutex

var wg sync.WaitGroup



for _, strategy := range strategies {

    wg.Add(1)

    go func(s RecommendationStrategy) {

      defer wg.Done()

          return

      }



      // Apply strategy weight

      for _, c := range candidates {

      }



      mu.Lock()

      allCandidates = append(allCandidates, candidates...)

      mu.Unlock()

    }(strategy)

}



wg.Wait()



// 4. Merge and deduplicate

merged := re.mergeAndDeduplicate(allCandidates)



// 5. Re-rank with ML model

ranked, err := re.rankWithML(ctx, userProfile, merged, request.Context)

    if err != nil {

             return merged[i].Score > merged[j].Score

        ranked = merged

    // 6. Apply diversity constraints

    // 7. Apply business rules

    filtered := re.applyBusinessRules(ctx, diversified, request)



    if len(filtered) > request.Limit {

        filtered = filtered[:request.Limit]

    }
```

```go
// UpdateTrendingScores refreshes trending story scores (called by background job)

func (tb *TrendingBoost) UpdateTrendingScores(ctx context.Context) error {

    // Calculate trending score: weighted combination of recent velocity and engagement

    query := `

        SELECT

          story_id,

          count() as recent_views,

          sum(metrics['engagement_score']) as total_engagement,

          count(DISTINCT user_id) as unique_readers

        FROM analytics_events

        WHERE event_type IN ('story.impression', 'story.engagement')

        AND timestamp >= now() - INTERVAL 1 HOUR

        GROUP BY story_id

        HAVING recent_views > 10

        ORDER BY total_engagement DESC

        LIMIT 100

    `



    rows, _ := tb.clickhouse.Query(ctx, query)

    defer rows.Close()



    pipe := tb.redis.Pipeline()



    for rows.Next() {

          var storyID string

          var views, uniqueReaders int64

          var engagement float64

          rows.Scan(&storyID, &views, &engagement, &uniqueReaders)



          // Trending score formula

          score := math.Log1p(float64(views)) * engagement * math.Log1p(float64(uniqueReaders))



          pipe.ZAdd(ctx, "trending:stories", &redis.Z{

               Score: score,

               Member: storyID,

          })

      }



      // Set expiry on trending set

      pipe.Expire(ctx, "trending:stories", 2*time.Hour)



      _, err := pipe.Exec(ctx)

      return err

}
```

```go
// AttributeRevenue attributes revenue to specific stories

func (rie *RevenueIntelligenceEngine) AttributeRevenue(

    ctx context.Context,

    storyID string,

    timeWindow time.Duration,

) (*StoryRevenue, error) {



    query := `

      SELECT

           story_id,

           sum(metrics['ad_revenue']) as ad_revenue,

           sum(metrics['subscription_value']) as subscription_attr,

           countIf(event_type = 'revenue.impression') as ad_impressions,

           countIf(event_type = 'revenue.click') as ad_clicks

      FROM analytics_events

      WHERE story_id = ?

      AND event_type LIKE 'revenue.%'

        AND timestamp >= now() - INTERVAL ?

        GROUP BY story_id

    `



    row := rie.clickhouse.QueryRow(ctx, query, storyID, timeWindow)



    var revenue StoryRevenue

    var adImpressions, adClicks int64



    row.Scan(&revenue.StoryID, &revenue.AdRevenue, &revenue.SubscriptionAttr,

        &adImpressions, &adClicks)



    revenue.TotalRevenue = revenue.AdRevenue + revenue.SubscriptionAttr



    // Estimate cost based on AI generation cost

    revenue.Cost = rie.getStoryCost(ctx, storyID)

    revenue.NetRevenue = revenue.TotalRevenue - revenue.Cost



    return &revenue, nil

}



// CalculateROI computes return on investment for content

func (rie *RevenueIntelligenceEngine) CalculateROI(

    ctx context.Context,

    tenantID string,

    timeWindow time.Duration,

) (float64, error) {



      query := `

          SELECT

            sum(metrics['revenue']) as total_revenue,

            sum(metrics['cost']) as total_cost

          FROM analytics_events

          WHERE tenant_id = ?

          AND event_type IN ('revenue.impression', 'revenue.click', 'ai.generation')

          AND timestamp >= now() - INTERVAL ?

      `



      row := rie.clickhouse.QueryRow(ctx, query, tenantID, timeWindow)



      var totalRevenue, totalCost float64

      row.Scan(&totalRevenue, &totalCost)



      if totalCost == 0 {

          return 0, nil

      }



      roi := ((totalRevenue - totalCost) / totalCost) * 100

      return roi, nil

}
```

```go
type FeatureSet struct {

    EntityID string          `json:"entity_id"`

    EntityType string         `json:"entity_type"` // user, story, content

    Features map[string]interface{} `json:"features"`

    GeneratedAt time.Time           `json:"generated_at"`

}



// GetOnlineFeatures retrieves features for real-time inference

func (fs *FeatureStore) GetOnlineFeatures(

    ctx context.Context,

    entityType string,

    entityID string,

    featureNames []string,

) (*FeatureSet, error) {



    // Check Redis cache first

    cacheKey := fmt.Sprintf("features:%s:%s", entityType, entityID)

    cached, err := fs.redis.Get(ctx, cacheKey).Result()

    if err == nil {

      var features FeatureSet

      json.Unmarshal([]byte(cached), &features)



      // Filter requested features

      filtered := make(map[string]interface{})

      for _, name := range featureNames {

          if val, ok := features.Features[name]; ok {

              filtered[name] = val

          }

      }

      features.Features = filtered

      return &features, nil

    }



    // Fetch from Feast

    feastFeatures, err := fs.feastServer.GetOnlineFeatures(ctx, &feast.GetOnlineFeaturesRequest{

         EntityType: entityType,

         EntityID: entityID,

         Features: featureNames,

    })

    if err != nil {

         return nil, fmt.Errorf("feast fetch: %w", err)

    }



    features := &FeatureSet{

         EntityID: entityID,

         EntityType: entityType,

         Features: feastFeatures.Features,

         GeneratedAt: time.Now(),

    }



    // Cache for 5 minutes

    data, _ := json.Marshal(features)

    fs.redis.Set(ctx, cacheKey, data, 5*time.Minute)



    return features, nil

}
```
