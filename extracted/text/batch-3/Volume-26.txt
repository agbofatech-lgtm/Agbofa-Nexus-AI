SERVICE DECISION RECORDS (SDR)

SDR-CF-001: Fact-Locking Before Generation



Decision: All verified facts, quotes, and statistics are extracted into an immutable LockedFactSet
before any content generation begins. All generation prompts include the locked facts as
constraints with explicit "DO NOT MODIFY" instructions.

Rationale: Prevents AI hallucination and ensures factual consistency across all content variants.
The LockedFactSet serves as the single source of truth for all generation.

Alternatives: Real-time fact verification during generation (rejected — latency, inconsistency);
post-generation fact checking (rejected — higher revision cost).

Trade-offs: Additional preprocessing step, but eliminates hallucination risk.



SDR-CF-002: Multi-Pass Generation Pipeline



Decision: Content generation uses a three-pass architecture: structural outline (Pass 1), full draft
(Pass 2), editorial refinement (Pass 3). Each pass is scored by an editorial quality model before
proceeding.

Rationale: Single-pass generation produces inconsistent quality. Multi-pass allows early
detection of structural issues and progressive refinement.

Alternatives: Single-pass generation (rejected — quality issues); iterative human-AI
collaboration (rejected — not scalable).

Trade-offs: Higher token usage (3x), but significantly higher quality output.



SDR-CF-003: Platform-Native Generation (Not Translation)



Decision: Platform-specific content (social posts, video scripts) is generated natively using
platform-aware prompts rather than generating once and adapting.

Rationale: True platform-native content requires different narrative structures, tones, and
conventions. Translation/adaptation produces inferior results.

Alternatives: Generate canonical version, then adapt (partially adopted for efficiency in breaking
news MVP).

Trade-offs: More LLM calls, but significantly better platform engagement.



SDR-CF-004: Model Tiering for Cost Optimization



Decision: Route generation tasks to models based on content criticality: Claude-3-Opus for full
articles and sensitive content, Claude-3-Sonnet for standard generation, Claude-3-Haiku for
headlines/SEO/social posts.

Rationale: Full articles require highest quality; headlines and social posts can use faster, cheaper
models. Tiering reduces costs by 40-60% without quality degradation.

Alternatives: Single model for everything (rejected — cost prohibitive at scale); open-source
only (rejected — quality gap).

Trade-offs: Multiple model management, but significant cost savings.

// services/content-factory/content-assembler/internal/domain/fact_locker.go

package domain



// FactLocker extracts and cryptographically seals all verified facts

type FactLocker struct {

      truthEngine TruthEngineClient

      storyGraph StoryGraphClient

}



// LockFacts creates an immutable fact set for generation

func (fl *FactLocker) LockFacts(ctx context.Context, storyID uuid.UUID) (*LockedFactSet, error) {



    // 1. Retrieve verified story from Truth Engine

    verifiedStory, err := fl.truthEngine.GetVerifiedStory(ctx, storyID)

    if err != nil {

        return nil, fmt.Errorf("get verified story: %w", err)

    }



    // 2. Verify story has passed Truth Engine verification

    if verifiedStory.VerificationStatus != "VERIFIED" {

        return nil, ErrStoryNotVerified

    }



    locked := &LockedFactSet{

        LockID: uuid.NewV7().String(),

        StoryID: storyID,

        LockedAt: time.Now(),

    }



    // 3. Extract core facts (5W1H)

    locked.CoreFacts = fl.extractCoreFacts(verifiedStory)



    // 4. Lock verified claims — MUST NOT be altered

locked.VerifiedClaims = fl.lockClaims(verifiedStory.VerifiedClaims)



// 5. Lock quotes VERBATIM

locked.AttributedQuotes = fl.lockQuotes(verifiedStory.Quotes)



// 6. Lock statistics with exact values

locked.Statistics = fl.lockStatistics(verifiedStory.Statistics)



// 7. Lock entity references with canonical names

locked.Entities = fl.lockEntities(verifiedStory.Entities)



// 8. Lock timeline events in chronological order

locked.Timeline = fl.lockTimeline(verifiedStory.Timeline)



// 9. Lock source attributions

locked.SourceAttributions = fl.lockAttributions(verifiedStory.Sources)



// 10. Generate required attributions based on sources

locked.RequiredAttributions = fl.generateRequiredAttributions(verifiedStory)



// 11. Generate forbidden phrases (clickbait, sensationalism)

locked.ForbiddenPhrases = fl.generateForbiddenPhrases(verifiedStory)



// 12. Calculate cryptographic hash for integrity verification

locked.Hash = fl.calculateHash(locked)

      return locked, nil

}



func (fl *FactLocker) lockQuotes(quotes []VerifiedQuote) []LockedQuote {

      locked := make([]LockedQuote, len(quotes))

      for i, q := range quotes {

          locked[i] = LockedQuote{

              QuoteID:        q.ID,

              QuoteText:      q.Text,    // VERBATIM — exact text preserved

              AttributedTo:    q.AttributedTo,

              Attribution:    q.Attribution, // Exact attribution preserved

              CannotBeParaphrased: true,         // CRITICAL: quotes cannot be paraphrased

          }

      }

      return locked

}



func (fl *FactLocker) calculateHash(facts *LockedFactSet) string {

      // Create canonical representation for hashing

      data, _ := json.Marshal(facts.CoreFacts)

      data = append(data, []byte(facts.LockedAt.String())...)



      hash := sha256.Sum256(data)

      return hex.EncodeToString(hash[:])

}

func (gp *GenerationPlanner) CreatePlan(

    ctx context.Context,

    story *VerifiedStory,

    lockedFacts *LockedFactSet,

    overrides *PlanOverrides,

) (*GenerationPlan, error) {

// Get archetype rules

rules := ArchetypeRulesConfig[archetype]



plan := &GenerationPlan{

    PlanID:        uuid.NewV7(),

    StoryID:       story.ID,

    StoryArchetype: archetype,

    ContentTypes: rules.DefaultContentTypes,

    TargetPlatforms: rules.DefaultPlatforms,

    TargetLanguages: []string{story.Language}, // Source language default

    ToneProfile:     rules.RecommendedTone,

    ModelTier:       rules.ModelTier,

    Priority:      rules.Priority,

    CreatedAt:       time.Now(),

}



// Apply overrides

if overrides != nil {

    if len(overrides.AdditionalContentTypes) > 0 {

        plan.ContentTypes = append(plan.ContentTypes, overrides.AdditionalContentTypes...)

    }

    if len(overrides.TargetLanguages) > 0 {

            plan.TargetLanguages = append(plan.TargetLanguages, overrides.TargetLanguages...)

        }

        if overrides.ModelTier != "" {

            plan.ModelTier = overrides.ModelTier

        }

    }



    // For breaking news, add translation if global story

    if archetype == ArchetypeBreakingNews && story.GeographicScope == "GLOBAL" {

        plan.TargetLanguages = append(plan.TargetLanguages, "es", "fr", "ar", "zh")

    }



    return plan, nil

}



func (gp *GenerationPlanner) classifyArchetype(story *VerifiedStory) StoryArchetype {

    switch {

    case story.State == "BREAKING" && time.Since(story.CreatedAt) < 2*time.Hour:

        return ArchetypeBreakingNews

    case story.State == "DEVELOPING":

        return ArchetypeDevelopingStory

    case story.Category == "EXPLAINER" || story.Category == "SCIENCE":

        return ArchetypeExplainer

    case story.Category == "INVESTIGATIVE":

        return ArchetypeInvestigative

    case story.Category == "HUMAN_INTEREST":

          return ArchetypeHumanInterest

      case story.IsLiveBlog:

          return ArchetypeLiveBlog

      default:

          return ArchetypeFeatureAnalysis

      }

}

// services/content-factory/content-assembler/internal/application/orchestrator.go

package application



// ContentOrchestrator coordinates the entire content generation pipeline

type ContentOrchestrator struct {

      factLocker     *domain.FactLocker

      planner       *domain.GenerationPlanner

      articleGen     ArticleGeneratorClient

      headlineGen      HeadlineGeneratorClient

      summaryGen        SummaryGeneratorClient

      seoEngine       SEOEngineClient

      multimediaFactory MultimediaFactoryClient

      socialGen      SocialContentGeneratorClient

      translationSvc TranslationServiceClient

    brandVoice       BrandVoiceClient

    qualityAssurance QualityAssuranceClient

    packager        ContentPackagerClient

    jobRepo         JobRepository

    eventBus        EventBus

}



// ProcessStory orchestrates the full content generation pipeline

func (co *ContentOrchestrator) ProcessStory(

    ctx context.Context,

    storyID uuid.UUID,

    overrides *domain.PlanOverrides,

) (*domain.ContentJob, error) {



    startTime := time.Now()



    // Create job

    job := &domain.ContentJob{

        JobID: uuid.NewV7(),

        StoryID: storyID,

        Status: domain.JobStatusReceived,

        CreatedAt: time.Now(),

    }



    co.jobRepo.Save(ctx, job)

// Phase 1: Lock Facts

lockedFacts, err := co.factLocker.LockFacts(ctx, storyID)

if err != nil {

    job.Status = domain.JobStatusRejected

    co.jobRepo.Save(ctx, job)

    return nil, fmt.Errorf("fact locking: %w", err)

}

job.LockedFacts = lockedFacts



// Phase 2: Create Generation Plan

plan, err := co.planner.CreatePlan(ctx, storyID, lockedFacts, overrides)

if err != nil {

    job.Status = domain.JobStatusRejected

    co.jobRepo.Save(ctx, job)

    return nil, fmt.Errorf("planning: %w", err)

}

job.GenerationPlan = plan

job.Status = domain.JobStatusPlanning

co.jobRepo.Save(ctx, job)



// Phase 3: Parallel Generation

job.Status = domain.JobStatusGenerating

co.jobRepo.Save(ctx, job)



content := &domain.ContentPackage{

    PackageID: uuid.NewV7(),

    StoryID: storyID,

}



// Launch all generation tasks in parallel

var wg sync.WaitGroup

var mu sync.Mutex

errors := make(chan error, len(plan.ContentTypes))



for _, ct := range plan.ContentTypes {

    wg.Add(1)

    go func(contentType domain.ContentType) {

      defer wg.Done()



      result, err := co.generateContent(ctx, contentType, storyID, lockedFacts, plan)

      if err != nil {

            errors <- fmt.Errorf("%s: %w", contentType, err)

            return

      }



      mu.Lock()

      co.mergeContent(content, contentType, result)

      mu.Unlock()

    }(ct)

}



wg.Wait()

close(errors)



// Check for errors

var genErrors []error

for err := range errors {

    genErrors = append(genErrors, err)

}



if len(genErrors) > 0 && len(genErrors) == len(plan.ContentTypes) {

    // All generation failed

    job.Status = domain.JobStatusRejected

    co.jobRepo.Save(ctx, job)

    return nil, fmt.Errorf("all generation failed: %v", genErrors)

}



job.ContentPackage = content



// Phase 4: Quality Assurance

job.Status = domain.JobStatusQualityCheck

co.jobRepo.Save(ctx, job)



qaResult, err := co.qualityAssurance.RunFullQA(ctx, content, lockedFacts)

if err != nil {

    job.Status = domain.JobStatusRejected

    co.jobRepo.Save(ctx, job)

    return nil, fmt.Errorf("QA failed: %w", err)

}



if !qaResult.Passed {

    // Retry failed components

    if qaResult.RetryableIssues > 0 {

        job.Status = domain.JobStatusRetrying

        co.jobRepo.Save(ctx, job)



        // Auto-correct and retry

        content, err = co.autoCorrectAndRetry(ctx, content, qaResult, lockedFacts, plan)

        if err != nil {

            job.Status = domain.JobStatusRejected

            co.jobRepo.Save(ctx, job)

            return nil, fmt.Errorf("retry failed: %w", err)

        }

    } else {

        job.Status = domain.JobStatusRejected

        co.jobRepo.Save(ctx, job)

        return nil, fmt.Errorf("QA critical issues: %v", qaResult.CriticalIssues)

    }

# services/content-factory/article-generator/src/generator/fact_checker.py

"""

Fact adherence verification — ensures generated content doesn't deviate from locked facts.

class FactAdherenceChecker:

      """Verifies generated content matches locked facts exactly."""



      async def verify_fact_adherence(

        self,

        sections: List[ArticleSection],

        locked_facts: LockedFactSet,

      ) -> FactAdherenceReport:

        """Check all generated content against locked facts."""



        report = FactAdherenceReport()

        full_text = " ".join([s.content for s in sections])



        # Check 1: All must-include facts present

        for fact in locked_facts.core_facts:

          if fact.must_include:

    found = self.find_fact_in_text(fact, full_text)

    if not found:

      report.add_violation(FactViolation(

           severity="CRITICAL",

           fact_id=fact.fact_id,

           issue="MISSING_REQUIRED_FACT",

           detail=f"Required fact not found: {fact.fact_text[:100]}",

      ))



# Check 2: All quotes are verbatim

for quote in locked_facts.attributed_quotes:

  if quote.cannot_be_paraphrased:

    exact_match = self.verify_quote_verbatim(quote.quote_text, full_text)

    if not exact_match:

      report.add_violation(FactViolation(

           severity="CRITICAL",

           quote_id=quote.quote_id,

           issue="QUOTE_ALTERED",

           detail=f"Quote text modified from verified original",

      ))



# Check 3: No hallucinated claims (claims not in locked facts)

generated_claims = await self.extract_claims_from_text(full_text)

for claim in generated_claims:

  if not self.claim_in_locked_set(claim, locked_facts):

    report.add_violation(FactViolation(

         severity="CRITICAL",

         issue="HALLUCINATED_CLAIM",

         detail=f"Claim not in verified fact set: {claim[:100]}",

    ))



# Check 4: Required attributions present

for attr in locked_facts.required_attributions:

  if attr.lower() not in full_text.lower():

    report.add_violation(FactViolation(

         severity="HIGH",

         issue="MISSING_ATTRIBUTION",

         detail=f"Required attribution missing: {attr}",

    ))



# Check 5: Forbidden phrases absent

for phrase in locked_facts.forbidden_phrases:

  if phrase.lower() in full_text.lower():

    report.add_violation(FactViolation(

         severity="MEDIUM",

         issue="FORBIDDEN_PHRASE",

         detail=f"Forbidden phrase found: {phrase}",

    ))



report.passes = len([v for v in report.violations if v.severity == "CRITICAL"]) == 0



return report

async def score_headline(

  self,

  headline: str,

  locked_facts: LockedFactSet,

  rules: dict,

  headline_type: str,

) -> HeadlineScores:

  """Score a headline across multiple dimensions."""



  scores = HeadlineScores()



  # 1. Factual accuracy (CRITICAL — must not distort facts)

  scores.factual_accuracy = await self.check_factual_accuracy(headline, locked_facts)



  # 2. Clickbait detection (lower is better)

  scores.clickbait_risk = self.detect_clickbait(headline)



  # 3. Entity inclusion

  scores.entity_coverage = self.check_entity_inclusion(headline, locked_facts)



  # 4. Readability

  scores.readability = self.calculate_readability(headline)



  # 5. Length compliance

  scores.length_compliance = 1.0 if len(headline) <= rules["max_chars"] else 0.5

  # 6. Click-through prediction (ML model)

  scores.predicted_ctr = await self.predict_ctr(headline, headline_type)



  # 7. Forbidden pattern check

  scores.forbidden_check = self.check_forbidden_patterns(

      headline, rules.get("forbidden", [])

  )



  # Weighted overall score

  scores.overall_score = (

      scores.factual_accuracy * 0.35 +       # Highest weight

      (1.0 - scores.clickbait_risk) * 0.20 +

      scores.entity_coverage * 0.15 +

      scores.readability * 0.10 +

      scores.length_compliance * 0.05 +

      scores.predicted_ctr * 0.10 +

      scores.forbidden_check * 0.05

  )



  return scores



def detect_clickbait(self, headline: str) -> float:

  """Detect clickbait patterns. Returns 0-1 risk score."""



  clickbait_patterns = [

      (r"you won'?t believe", 0.9),

          (r"what happens next", 0.8),

          (r"this will shock you", 0.9),

          (r"mind.?\s*blowing", 0.7),

          (r"doctors hate", 0.9),

          (r"one weird trick", 0.9),

          (r"secret .+ don'?t want you to know", 0.8),

          (r"number \d+ will", 0.7),

          (r"changed my life", 0.6),

          (r"you need to see", 0.5),

      ]



      risk = 0.0

      headline_lower = headline.lower()



      for pattern, weight in clickbait_patterns:

          if re.search(pattern, headline_lower):

            risk = max(risk, weight)



      return risk

"""

AI Quality Assurance Pipeline — Validates all generated content

before packaging and distribution.

"""



class QualityAssurancePipeline:

      """Multi-stage quality validation for AI-generated content."""



      def __init__(self):

           self.checks = [

      FactConsistencyCheck(),

      HallucinationDetectionCheck(),

      GrammarCheck(),

      StyleComplianceCheck(),

      PlagiarismCheck(),

      ToxicityCheck(),

      BiasCheck(),

      ReadabilityCheck(),

      QuoteAccuracyCheck(),

      AttributionCheck(),

      SEOValidationCheck(),

  ]



async def run_full_qa(

  self,

  content: ContentPackage,

  locked_facts: LockedFactSet,

) -> QualityReport:

  """Run all QA checks on generated content."""



  report = QualityReport(content_id=content.package_id)



  # Run all checks in parallel

  tasks = [check.run(content, locked_facts) for check in self.checks]

  results = await asyncio.gather(*tasks, return_exceptions=True)

critical_issues = []

high_issues = []



for i, result in enumerate(results):

  if isinstance(result, Exception):

     report.add_error(f"Check {self.checks[i].__class__.__name__} failed: {result}")

     continue



  report.add_check_result(result)



  if not result.passes:

     if result.severity == "CRITICAL":

       critical_issues.append(result)

     elif result.severity == "HIGH":

       high_issues.append(result)



# Determine pass/fail

report.passes = len(critical_issues) == 0

report.retryable = len(high_issues) > 0 and len(critical_issues) == 0



# Calculate overall score

report.overall_score = self.calculate_overall_score(report)



# Generate publication readiness

report.publication_ready = (

  report.passes and

      report.overall_score >= 85.0

  )



  return report



def calculate_overall_score(self, report: QualityReport) -> float:

  """Calculate weighted overall quality score."""



  weights = {

      "fact_consistency": 0.30,

      "hallucination_detection": 0.25,

      "grammar": 0.10,

      "style_compliance": 0.10,

      "plagiarism": 0.10,

      "toxicity": 0.05,

      "bias": 0.05,

      "readability": 0.03,

      "quote_accuracy": 0.02,

  }



  total = 0.0

  total_weight = 0.0



  for check_result in report.check_results:

      weight = weights.get(check_result.check_name, 0.01)

      total += check_result.score * weight

          total_weight += weight



        if total_weight == 0:

          return 0.0



        return total / total_weight
