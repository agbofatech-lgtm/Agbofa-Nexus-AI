// services/truth-engine/shared/pkg/resilience/circuit_breaker.go

package resilience

import (

    "context"

    "fmt"

    "sync"

    "time"

)



// CircuitState represents the current state of a circuit breaker

type CircuitState string



const (

    CircuitClosed CircuitState = "CLOSED" // Normal operation

    CircuitOpen     CircuitState = "OPEN"   // Failing, reject requests

    CircuitHalfOpen CircuitState = "HALF_OPEN" // Testing recovery

)



// CircuitBreaker implements the circuit breaker pattern

type CircuitBreaker struct {

    name           string

    state         CircuitState

    failureThreshold int

    successThreshold int

    timeout         time.Duration

    failureCount     int

    successCount      int

    lastFailureTime time.Time

    lastSuccessTime time.Time

    mu          sync.RWMutex

    metrics       *CircuitBreakerMetrics

}



// CircuitBreakerConfig configures a circuit breaker

type CircuitBreakerConfig struct {

    Name         string       `json:"name"`

    FailureThreshold int        `json:"failure_threshold"` // Open after N failures

    SuccessThreshold int         `json:"success_threshold"` // Close after N successes

    Timeout       time.Duration `json:"timeout"`         // Time before half-open

    HalfOpenMaxReqs int            `json:"half_open_max_reqs"` // Max requests in half-open

}



var DefaultCircuitBreakerConfig = CircuitBreakerConfig{

    Timeout:          30 * time.Second,

// NewCircuitBreaker creates a new circuit breaker

func NewCircuitBreaker(config CircuitBreakerConfig) *CircuitBreaker {

    return &CircuitBreaker{

         name:        config.Name,

         state:      CircuitClosed,

         failureThreshold: config.FailureThreshold,

         successThreshold: config.SuccessThreshold,

         timeout:      config.Timeout,

    }

}

// Execute runs a function with circuit breaker protection

func (cb *CircuitBreaker) Execute(ctx context.Context, fn func(context.Context) error) error {

    // Check if circuit is open

    if err := cb.beforeRequest(); err != nil {

        return err

    }



    // Execute the function

    err := fn(ctx)



    // Record result

    cb.afterRequest(err)



    return err

}



func (cb *CircuitBreaker) beforeRequest() error {

    cb.mu.Lock()

    defer cb.mu.Unlock()



    switch cb.state {

    case CircuitClosed:

        return nil



    case CircuitOpen:

        // Check if timeout has elapsed

        if time.Since(cb.lastFailureTime) >= cb.timeout {

            cb.state = CircuitHalfOpen

            cb.successCount = 0

            return nil

        }

        return fmt.Errorf("circuit breaker %s is OPEN: %v since last failure",

            cb.name, time.Since(cb.lastFailureTime))



    case CircuitHalfOpen:

        return nil



    default:

        return nil

    }

}



func (cb *CircuitBreaker) afterRequest(err error) {

    cb.mu.Lock()

    defer cb.mu.Unlock()



    if err != nil {

        cb.onFailure()

    } else {

        cb.onSuccess()

    }

}



func (cb *CircuitBreaker) onFailure() {

    cb.failureCount++

    cb.successCount = 0

    cb.lastFailureTime = time.Now()



    switch cb.state {

    case CircuitClosed:

        if cb.failureCount >= cb.failureThreshold {

            cb.state = CircuitOpen

            cb.metrics.RecordCircuitOpen(cb.name)

        }

    case CircuitHalfOpen:

        cb.state = CircuitOpen

        cb.metrics.RecordCircuitOpen(cb.name)

    }

}



func (cb *CircuitBreaker) onSuccess() {

    cb.successCount++

    cb.failureCount = 0

    cb.lastSuccessTime = time.Now()



    if cb.state == CircuitHalfOpen && cb.successCount >= cb.successThreshold {

        cb.state = CircuitClosed

        cb.metrics.RecordCircuitClosed(cb.name)

    }

}



// State returns the current circuit state

func (cb *CircuitBreaker) State() CircuitState {

    cb.mu.RLock()

    defer cb.mu.RUnlock()

    return cb.state

}



// CircuitBreakerRegistry manages circuit breakers for all dependencies

type CircuitBreakerRegistry struct {

    breakers map[string]*CircuitBreaker

    mu      sync.RWMutex

}



func NewCircuitBreakerRegistry() *CircuitBreakerRegistry {

    registry := &CircuitBreakerRegistry{

        breakers: make(map[string]*CircuitBreaker),

    }



    // Initialize all configured circuit breakers

    for name, config := range CircuitBreakerConfigs {

        registry.breakers[name] = NewCircuitBreaker(config)

    }

      return registry

}



func (cbr *CircuitBreakerRegistry) Get(name string) (*CircuitBreaker, error) {

      cbr.mu.RLock()

      defer cbr.mu.RUnlock()



      cb, exists := cbr.breakers[name]

      if !exists {

          return nil, fmt.Errorf("circuit breaker not found: %s", name)

      }

      return cb, nil

}

              "CONNECTION_ERROR", "DEADLOCK_DETECTED",

              "SERIALIZATION_FAILURE", "TOO_MANY_CONNECTIONS",

         },

    },

    "evidence_storage": {

         MaxAttempts: 5,

         InitialDelay: 1 * time.Second,

         MaxDelay:     60 * time.Second,

         BackoffFactor: 2.0,

         RetryableErrors: []string{

              "STORAGE_FULL", "WRITE_TIMEOUT", "S3_SLOW_DOWN",

         },

    },

}



// RetryWithBackoff executes a function with exponential backoff retry

func RetryWithBackoff(

    ctx context.Context,

    config RetryConfig,

    operationName string,

    fn func(context.Context) error,

) error {



  var lastErr error

  delay := config.InitialDelay



  for attempt := 0; attempt <= config.MaxAttempts; attempt++ {

     // Check context cancellation

     select {

     case <-ctx.Done():

         return fmt.Errorf("%s cancelled: %w", operationName, ctx.Err())

     default:

     }



     // Execute operation

     err := fn(ctx)

     if err == nil {

         return nil

     }



     lastErr = err



     // Check if error is retryable

     if !isRetryableError(err, config.RetryableErrors) {

         return fmt.Errorf("%s: non-retryable error: %w", operationName, err)

     }

    // Don't sleep on last attempt

    if attempt == config.MaxAttempts {

        break

    }



    // Calculate backoff delay

    backoffDelay := float64(delay) * math.Pow(config.BackoffFactor, float64(attempt))

    if config.Jitter {

        // Add ±25% jitter

        jitterRange := backoffDelay * 0.5

        backoffDelay += (rand.Float64() - 0.5) * jitterRange

    }



    if backoffDelay > float64(config.MaxDelay) {

        backoffDelay = float64(config.MaxDelay)

    }



    sleepDuration := time.Duration(backoffDelay)



    // Wait with context awareness

    select {

    case <-time.After(sleepDuration):

    case <-ctx.Done():

        return fmt.Errorf("%s cancelled during retry: %w", operationName, ctx.Err())

    }

}

      return fmt.Errorf("%s: max retries (%d) exceeded: %w",

          operationName, config.MaxAttempts, lastErr)

}



func isRetryableError(err error, retryablePatterns []string) bool {

      errStr := err.Error()

      for _, pattern := range retryablePatterns {

          if containsIgnoreCase(errStr, pattern) {

              return true

          }

      }

      return false

}

// GetVerificationCapabilities returns what verification is possible given current state

func (dm *DegradationManager) GetVerificationCapabilities() VerificationCapabilities {

    dm.mu.RLock()

    defer dm.mu.RUnlock()



    caps := VerificationCapabilities{

        CanUseKnowledgeGraph: true,

        CanUseExternalAPIs:     true,

        CanUseGraphValidation: true,

        CanUseMediaAnalysis: true,

        CanCalculateConfidence: true,

        IsFullyOperational:   true,

    }



    if status, exists := dm.status["knowledge_graph"]; exists && status.IsDegraded {

        caps.CanUseKnowledgeGraph = false

        caps.IsFullyOperational = false

    }



    if status, exists := dm.status["external_factcheck_api"]; exists && status.IsDegraded {

        caps.CanUseExternalAPIs = false

        caps.IsFullyOperational = false

    }



    if status, exists := dm.status["neo4j"]; exists && status.IsDegraded {

        caps.CanUseGraphValidation = false

        caps.IsFullyOperational = false

    }



    if status, exists := dm.status["media_forensics"]; exists && status.IsDegraded {

        caps.CanUseMediaAnalysis = false

        caps.IsFullyOperational = false

    }



    // If too many dependencies degraded, cannot calculate confidence

    degradedCount := 0

    for _, status := range dm.status {

        if status.IsDegraded {

            degradedCount++

        }

    }

    if degradedCount >= 3 {

        caps.CanCalculateConfidence = false

    }



    return caps

}

      CanUseKnowledgeGraph bool `json:"can_use_knowledge_graph"`

      CanUseMediaAnalysis bool `json:"can_use_media_analysis"`

      CanCalculateConfidence bool `json:"can_calculate_confidence"`

      IsFullyOperational     bool `json:"is_fully_operational"`

}

// services/truth-engine/claim-verifier/internal/domain/contradiction.go

package domain



// ContradictionResolver handles conflicting evidence scenarios

type ContradictionResolver struct {

      evidenceWeights EvidenceWeights

      escalationThreshold float64

}



type ContradictionResult struct {

      ClaimID       string          `json:"claim_id"`

      HasContradiction bool              `json:"has_contradiction"`

      ContradictionLevel ContradictionLevel `json:"contradiction_level"`

    SupportingSources int              `json:"supporting_sources"`

    ContradictingSources int            `json:"contradicting_sources"`

    Resolution       string         `json:"resolution"`

    RequiresHumanReview bool                `json:"requires_human_review"`

    Detail         string         `json:"detail"`

}



type ContradictionLevel string

const (

    ContradictionNone         ContradictionLevel = "NONE"

    ContradictionMinor        ContradictionLevel = "MINOR"    // Minor disagreement

    ContradictionModerate ContradictionLevel = "MODERATE" // Significant disagreement

    ContradictionSevere ContradictionLevel = "SEVERE"         // Direct contradiction

    ContradictionIrreconcilable ContradictionLevel = "IRRECONCILABLE" // Cannot resolve

)



// ResolveContradiction handles conflicting evidence for a claim

func (cr *ContradictionResolver) ResolveContradiction(

    ctx context.Context,

    claim *AtomicClaim,

    evidence []*Evidence,

) (*ContradictionResult, error) {



    result := &ContradictionResult{

        ClaimID: claim.ClaimID,

    }

// Separate supporting and contradicting evidence

var supporting []*Evidence

var contradicting []*Evidence



for _, ev := range evidence {

    if ev.SupportsClaim {

        supporting = append(supporting, ev)

    } else {

        contradicting = append(contradicting, ev)

    }

}



result.SupportingSources = len(supporting)

result.ContradictingSources = len(contradicting)



// No contradiction if no contradicting evidence

if len(contradicting) == 0 {

    result.HasContradiction = false

    result.ContradictionLevel = ContradictionNone

    result.Resolution = "NO_CONTRADICTION"

    return result, nil

}



result.HasContradiction = true

// Calculate weighted scores for each side

supportingScore := cr.calculateWeightedScore(supporting)

contradictingScore := cr.calculateWeightedScore(contradicting)



// Determine contradiction level

scoreRatio := 0.0

if supportingScore+contradictingScore > 0 {

    scoreRatio = contradictingScore / (supportingScore + contradictingScore)

}



switch {

case scoreRatio < 0.2:

    result.ContradictionLevel = ContradictionMinor

    result.Resolution = "MAJORITY_SUPPORTS_CLAIM"

    result.Detail = fmt.Sprintf(

        "Minor contradiction: %.0f%% of weighted evidence contradicts (minority sources)",

        scoreRatio*100,

    )



case scoreRatio < 0.4:

    result.ContradictionLevel = ContradictionModerate

    result.Resolution = "WEIGHTED_MAJORITY_SUPPORTS"

    result.RequiresHumanReview = true

    result.Detail = fmt.Sprintf(

        "Moderate contradiction: %.0f%% of weighted evidence contradicts",

        scoreRatio*100,

        )



    case scoreRatio < 0.6:

        result.ContradictionLevel = ContradictionSevere

        result.Resolution = "EVIDENCE_SPLIT_REQUIRES_RESOLUTION"

        result.RequiresHumanReview = true

        result.Detail = fmt.Sprintf(

            "Severe contradiction: evidence is split (%.0f%% contradicts)",

            scoreRatio*100,

        )



    default:

        result.ContradictionLevel = ContradictionIrreconcilable

        result.Resolution = "CANNOT_RESOLVE_AUTOMATICALLY"

        result.RequiresHumanReview = true

        result.Detail = fmt.Sprintf(

            "Irreconcilable contradiction: %.0f%% of weighted evidence contradicts the claim",

            scoreRatio*100,

        )

    }



    return result, nil

}



func (cr *ContradictionResolver) calculateWeightedScore(evidence []*Evidence) float64 {

    score := 0.0

      for _, ev := range evidence {

          // Weight by source reliability and evidence quality

          weight := ev.SourceReliability * ev.EvidenceQuality

          score += weight

      }



      return score

}

      workflow.CurrentStep = i



      stepCtx, cancel := context.WithTimeout(ctx, step.Timeout)

      defer cancel()



      err := rm.executeRecoveryStep(stepCtx, step)

      if err != nil {

        step.Status = StepFailed

        workflow.Status = RecoveryFailed



        if step.RetryCount < 3 {

            step.RetryCount++

            i-- // Retry same step

            continue

        }

              return fmt.Errorf("recovery step %s failed: %w", step.Name, err)

          }



          step.Status = StepCompleted

      }



      workflow.Status = RecoveryCompleted



      // Reset circuit breaker

      if cb, err := rm.circuitBreakers.Get(component); err == nil {

          cb.Reset()

      }



      // Clear degradation

      rm.degradation.ClearDegradation(component)



      return nil

}
