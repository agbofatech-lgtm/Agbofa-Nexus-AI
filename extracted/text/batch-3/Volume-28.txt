SERVICE DECISION RECORDS (SDR)



SDR-DST-001: Connector Plugin Architecture with Hot-Swap Support



Decision: Each platform connector is an independent, hot-swappable plugin implementing a
unified PublisherConnector interface. Connectors are deployed as separate Kubernetes
deployments and communicate with the orchestrator via gRPC.

Rationale: Platforms change APIs frequently and have different rate limits, authentication
methods, and content requirements. Plugin architecture enables independent deployment,
scaling, and updating of each connector without affecting others.

Alternatives: Monolithic adapter layer (rejected — cannot scale independently, risky
deployments); third-party social media management API (rejected — latency, cost, vendor lock-
in).

Trade-offs: More deployments to manage, but each is simpler and independently scalable.



SDR-DST-002: Priority-Based Multi-Queue Architecture



Decision: Four priority queues (CRITICAL, HIGH, MEDIUM, LOW) implemented as Redis Streams
with Kafka for event sourcing. CRITICAL queue has dedicated workers for breaking news.
Preemptive scheduling allows CRITICAL jobs to claim workers from lower queues.

Rationale: Breaking news and corrections must publish immediately. Scheduled content can
wait. Dedicated resources for critical paths prevent head-of-line blocking.

Alternatives: Single queue with priority sorting (rejected — cannot preempt in-flight work);
Kafka-only queues (rejected — Redis Streams provide lower latency for priority queuing).

Trade-offs: More complex queue management, but essential for news platform requirements.

SDR-DST-003: Idempotent Publishing with Content Hashing



Decision: Every publication request includes a content hash (SHA-256 of canonical content).
Connectors check platform for existing identical content before publishing. Duplicate requests
are acknowledged without republishing.

Rationale: Network failures and retries can cause duplicate publications across platforms.
Content hashing enables safe exactly-once publishing semantics.

Alternatives: Idempotency keys alone (rejected — cannot detect duplicates across sessions);
platform-native dedup (rejected — inconsistent across platforms).

Trade-offs: Hash computation overhead, but negligible compared to duplicate publication cost.



SDR-DST-004: Event-Sourced Publication State



Decision: All publication state transitions are recorded as immutable events in Kafka.
Materialized views in PostgreSQL provide current state for queries. Event sourcing enables
complete audit trail and replay for recovery.

Rationale: Publishing is inherently event-driven. Event sourcing provides complete history,
enables replay for disaster recovery, and supports real-time monitoring dashboards.

Alternatives: Traditional CRUD state management (rejected — no audit trail, harder recovery);
blockchain (rejected — unnecessary decentralization).

Trade-offs: Additional Kafka storage, but essential for audit compliance.

// services/distribution/orchestrator/internal/application/orchestrator.go

package application



// PublicationOrchestrator coordinates the complete publication lifecycle

type PublicationOrchestrator struct {

      jobRepo     JobRepository

      scheduler    SchedulingClient

      queueManager QueueManagerClient

      contentAdapter ContentAdapterClient

      connectorManager *ConnectorManager

      deliveryMonitor DeliveryMonitorClient

    eventBus        EventBus

    stateMachine *PublicationStateMachine

}



// SubmitPublicationJob receives a content package and initiates publishing

func (po *PublicationOrchestrator) SubmitPublicationJob(

    ctx context.Context,

    contentPackage *ContentPackage,

    config *PublicationConfig,

) (*PublicationJob, error) {



    // 1. Validate content package

    if err := po.validateContentPackage(contentPackage); err != nil {

        return nil, fmt.Errorf("validate content package: %w", err)

    }



    // 2. Create publication job

    job := &PublicationJob{

        JobID:       uuid.NewV7(),

        ContentPackageID: contentPackage.PackageID,

        StoryID:     contentPackage.StoryID,

        TenantID:     contentPackage.TenantID,

        JobType:      config.JobType,

        Priority:    config.Priority,

        Status:      StatusReceived,

        TargetPlatforms: po.determinePlatforms(contentPackage, config),

    MaxRetries:        3,

    CreatedAt:        time.Now(),

    UpdatedAt:         time.Now(),

}



// 3. Handle scheduling

if config.ScheduledFor != nil {

    job.ScheduledFor = config.ScheduledFor

    job.Status = StatusScheduled

    po.jobRepo.Save(ctx, job)

    po.scheduler.Schedule(ctx, job.JobID, *config.ScheduledFor)



    // Publish scheduled event

    po.eventBus.Publish(ctx, &JobScheduledEvent{

         JobID:     job.JobID,

         StoryID:   job.StoryID,

         ScheduledFor: *config.ScheduledFor,

    })

    return job, nil

}



// 4. Handle embargo

if config.EmbargoUntil != nil {

    job.EmbargoUntil = config.EmbargoUntil

    job.ScheduledFor = config.EmbargoUntil

    job.Status = StatusScheduled

        po.jobRepo.Save(ctx, job)

        po.scheduler.Schedule(ctx, job.JobID, *config.EmbargoUntil)

        return job, nil

    }



    // 5. Immediate publishing — execute now

    return po.executeJob(ctx, job)

}



// executeJob performs the actual publication

func (po *PublicationOrchestrator) executeJob(

    ctx context.Context,

    job *PublicationJob,

) (*PublicationJob, error) {



    // Transition to validating

    po.transitionState(ctx, job, StatusValidating)



    // Validate all platform targets are available

    for _, target := range job.TargetPlatforms {

        if !po.connectorManager.IsAvailable(target.Platform) {

            return nil, fmt.Errorf("platform unavailable: %s", target.Platform)

        }

    }



    // Transition to adapting

po.transitionState(ctx, job, StatusAdapting)



// Adapt content for each platform

adaptedContent, err := po.contentAdapter.AdaptForPlatforms(

    ctx, job.ContentPackage, job.TargetPlatforms,

)

if err != nil {

    job.Status = StatusFailed

    job.LastError = err.Error()

    po.jobRepo.Save(ctx, job)

    return job, fmt.Errorf("content adaptation: %w", err)

}

job.AdaptedContent = adaptedContent



// Transition to queued

po.transitionState(ctx, job, StatusQueued)



// Submit to queue manager

if err := po.queueManager.Enqueue(ctx, job); err != nil {

    job.Status = StatusFailed

    job.LastError = err.Error()

    po.jobRepo.Save(ctx, job)

    return job, fmt.Errorf("queue enqueue: %w", err)

}



// Queue manager will handle the actual publishing

    // Results will be reported back via events



    return job, nil

}



// ProcessPlatformJob publishes to a single platform

func (po *PublicationOrchestrator) ProcessPlatformJob(

    ctx context.Context,

    platformJob *PlatformJob,

    adaptedContent *PlatformContent,

) (*PlatformJob, error) {



    platformJob.Status = PlatformStatusPublishing

    platformJob.AttemptCount++

    now := time.Now()

    platformJob.LastAttemptAt = &now



    // Get connector for platform

    connector, err := po.connectorManager.GetConnector(platformJob.Platform)

    if err != nil {

        platformJob.Status = PlatformStatusFailed

        platformJob.ErrorMessage = err.Error()

        return platformJob, err

    }



    // Check rate limits

if err := connector.CheckRateLimit(ctx); err != nil {

    platformJob.Status = PlatformStatusRetrying

    platformJob.ErrorMessage = err.Error()

    platformJob.ErrorCode = "RATE_LIMITED"



    // Schedule retry with backoff

    retryAfter := po.calculateRetryDelay(platformJob.AttemptCount)

    nextRetry := time.Now().Add(retryAfter)

    platformJob.NextRetryAt = &nextRetry



    return platformJob, ErrRateLimited

}



// Publish

result, err := connector.Publish(ctx, adaptedContent)

if err != nil {

    platformJob.Status = PlatformStatusFailed

    platformJob.ErrorMessage = err.Error()



    if platformJob.AttemptCount < platformJob.MaxAttempts {

        platformJob.Status = PlatformStatusRetrying

        retryAfter := po.calculateRetryDelay(platformJob.AttemptCount)

        nextRetry := time.Now().Add(retryAfter)

        platformJob.NextRetryAt = &nextRetry

    }

         return platformJob, err

    }



    // Success

    platformJob.Status = PlatformStatusPublished

    platformJob.PlatformPostID = result.PlatformPostID

    platformJob.PlatformURL = result.PlatformURL

    now = time.Now()

    platformJob.PublishedAt = &now



    // Publish analytics event

    po.eventBus.Publish(ctx, &PlatformPublishedEvent{

         JobID:      platformJob.PublicationJobID,

         PlatformJobID: platformJob.PlatformJobID,

         Platform:    platformJob.Platform,

         PlatformPostID: result.PlatformPostID,

         PublishedAt: now,

    })



    return platformJob, nil

}



func (po *PublicationOrchestrator) calculateRetryDelay(attemptCount int) time.Duration {

    // Exponential backoff: 1s, 5s, 30s, 2min, 10min

    delays := []time.Duration{

         1 * time.Second,

          5 * time.Second,

          30 * time.Second,

          2 * time.Minute,

          10 * time.Minute,

      }



      if attemptCount <= len(delays) {

          return delays[attemptCount-1]

      }

      return 30 * time.Minute

}

// services/distribution/scheduler/internal/domain/scheduler.go

package domain



// SchedulingEngine manages publication timing

type SchedulingEngine struct {

      redis       *redis.Client

    timezoneService *TimezoneService

    audienceOptimizer *AudienceTimeOptimizer

}



type ScheduleConfig struct {

    Type        ScheduleType     `json:"type"`

    PublishAt    *time.Time      `json:"publish_at,omitempty"`

    Timezone      string       `json:"timezone"`

    OptimalWindow *TimeWindow            `json:"optimal_window,omitempty"`

    RecurrenceRule *RecurrenceRule `json:"recurrence_rule,omitempty"`

    CampaignID     string       `json:"campaign_id,omitempty"`

}



type ScheduleType string

const (

    ScheduleImmediate ScheduleType = "IMMEDIATE"

    ScheduleOnce       ScheduleType = "ONCE"

    ScheduleOptimal ScheduleType = "OPTIMAL"

    ScheduleRecurring ScheduleType = "RECURRING"

    ScheduleCampaign ScheduleType = "CAMPAIGN"

)



// Schedule determines the optimal publication time

func (se *SchedulingEngine) Schedule(

    ctx context.Context,

    jobID uuid.UUID,

  config *ScheduleConfig,

) (*ScheduleResult, error) {



  result := &ScheduleResult{

      JobID: jobID,

  }



  switch config.Type {

  case ScheduleImmediate:

      result.PublishAt = time.Now()

      result.Immediate = true



  case ScheduleOnce:

      if config.PublishAt == nil {

          return nil, ErrMissingPublishTime

      }

      result.PublishAt = *config.PublishAt



  case ScheduleOptimal:

      optimalTime, err := se.calculateOptimalTime(ctx, config)

      if err != nil {

          return nil, fmt.Errorf("calculate optimal time: %w", err)

      }

      result.PublishAt = optimalTime

      result.OptimizedFor = config.OptimalWindow

    case ScheduleCampaign:

        campaignTime, err := se.getCampaignSlot(ctx, config.CampaignID)

        if err != nil {

             return nil, fmt.Errorf("get campaign slot: %w", err)

        }

        result.PublishAt = campaignTime

    }



    // Store scheduled job

    if !result.Immediate {

        score := float64(result.PublishAt.Unix())

        se.redis.ZAdd(ctx, "distribution:scheduled_jobs", &redis.Z{

             Score: score,

             Member: jobID.String(),

        })

    }



    return result, nil

}



// calculateOptimalTime determines the best publication time

func (se *SchedulingEngine) calculateOptimalTime(

    ctx context.Context,

    config *ScheduleConfig,

) (time.Time, error) {

    // Get audience timezone distribution

    audienceTZs := se.timezoneService.GetAudienceDistribution(ctx)



    // Get platform peak times

    platformPeaks := se.getPlatformPeakTimes(ctx)



    // Generate candidate times

    candidates := se.generateCandidateTimes(audienceTZs, platformPeaks, 48) // Next 48 hours



    // Score each candidate

    type scoredTime struct {

        time time.Time

        score float64

    }



    var best scoredTime

    for _, candidate := range candidates {

        score := se.scoreTimeSlot(candidate, audienceTZs, platformPeaks)

        if score > best.score {

            best = scoredTime{candidate, score}

        }

    }



    return best.time, nil

}

// scoreTimeSlot evaluates a time slot for publication

func (se *SchedulingEngine) scoreTimeSlot(

  t time.Time,

  audienceTZs *AudienceDistribution,

  platformPeaks map[string][]PeakTime,

) float64 {



  weights := map[string]float64{

      "audience_overlap": 0.40,

      "platform_peak":      0.30,

      "competition_gap":     0.15,

      "recency":         0.15,

  }



  score := 0.0



  // Audience overlap: how many audience members are in their peak hours?

  overlap := audienceTZs.GetOverlap(t)

  score += overlap * weights["audience_overlap"]



  // Platform peak: is this a known high-engagement time?

  if peaks, exists := platformPeaks["global"]; exists {

      for _, peak := range peaks {

        if t.Hour() >= peak.StartHour && t.Hour() <= peak.EndHour {

            score += 1.0 * weights["platform_peak"]

        }

          }

      }



      // Competition gap: avoid publishing when many other publications are scheduled

      competitionScore := se.calculateCompetitionGap(t)

      score += competitionScore * weights["competition_gap"]



      // Recency: prefer sooner rather than later

      hoursFromNow := time.Until(t).Hours()

      recencyScore := math.Max(0, 1.0-hoursFromNow/48.0)

      score += recencyScore * weights["recency"]



      return score

}

// Enqueue adds a job to the appropriate priority queue

func (qm *QueueManager) Enqueue(ctx context.Context, job *PublicationJob) error {

    // Create platform jobs for each target

    for _, target := range job.TargetPlatforms {

        platformJob := &PlatformJob{

            PlatformJobID: uuid.NewV7(),

            PublicationJobID: job.JobID,

            Platform:      target.Platform,

            Status:     PlatformStatusPending,

            MaxAttempts:      3,

            ContentHash:     job.AdaptedContent[target.Platform].ContentHash,

        }



        job.PlatformJobs = append(job.PlatformJobs, platformJob)



        // Determine effective priority

        effectivePriority := qm.determineEffectivePriority(job.Priority, target.Platform)

        // Add to Redis Stream for the priority level

        streamKey := fmt.Sprintf("distribution:queue:%s", strings.ToLower(string(effectivePriority)))

        jobData, _ := json.Marshal(platformJob)



        if err := qm.redis.XAdd(ctx, &redis.XAddArgs{

            Stream: streamKey,

            Values: map[string]interface{}{

                 "job": string(jobData),

            },

        }).Err(); err != nil {

            return fmt.Errorf("enqueue to redis stream: %w", err)

        }

    }



    return nil

}



// startWorkerPool starts workers for a priority level

func (qm *QueueManager) startWorkerPool(pool *WorkerPool) {

    streamKey := fmt.Sprintf("distribution:queue:%s", strings.ToLower(string(pool.Priority)))

    consumerGroup := fmt.Sprintf("workers-%s", strings.ToLower(string(pool.Priority)))



    // Create consumer group if not exists

    qm.redis.XGroupCreateMkStream(context.Background(), streamKey, consumerGroup, "0")



    for i := 0; i < pool.MaxWorkers; i++ {

        go qm.worker(pool, streamKey, consumerGroup, fmt.Sprintf("worker-%d", i))

    }

}



// worker processes jobs from a queue

func (qm *QueueManager) worker(pool *WorkerPool, streamKey, consumerGroup,
consumerName string) {

    for {

        // Read from stream

        results, err := qm.redis.XReadGroup(context.Background(), &redis.XReadGroupArgs{

            Group: consumerGroup,

            Consumer: consumerName,

            Streams: []string{streamKey, ">"},

            Count: 1,

            Block: 5 * time.Second,

        }).Result()



        if err != nil || len(results) == 0 {

            continue

        }



        for _, msg := range results[0].Messages {

            jobData := msg.Values["job"].(string)



            var platformJob PlatformJob

            json.Unmarshal([]byte(jobData), &platformJob)

            // Process the job

            qm.processJob(&platformJob)



            // Acknowledge

            qm.redis.XAck(context.Background(), streamKey, consumerGroup, msg.ID)

        }

    }

}



// preemptiveScheduler allows CRITICAL workers to steal from lower queues

func (qm *QueueManager) preemptiveScheduler() {

    ticker := time.NewTicker(1 * time.Second)

    defer ticker.Stop()



    for range ticker.C {

        // Check if CRITICAL queue has pending jobs

        criticalLen, _ := qm.redis.XLen(context.Background(), "distribution:queue:critical").Result()



        if criticalLen > 0 {

            // Steal workers from LOW and MEDIUM pools

            for _, priority := range []PublicationPriority{PriorityLow, PriorityMedium} {

              pool := qm.workerPools[priority]

              pool.mu.Lock()

              if pool.ActiveWorkers > pool.MaxWorkers/2 {

                // Temporarily redirect a worker to CRITICAL

                      pool.ActiveWorkers--

                      pool.mu.Unlock()



                      go func() {

                        // Process one CRITICAL job

                        qm.worker(qm.workerPools[PriorityCritical],

                             "distribution:queue:critical",

                             "workers-critical",

                             "preemptive-worker")



                        // Return to original pool

                        pool.mu.Lock()

                        pool.ActiveWorkers++

                        pool.mu.Unlock()

                      }()

                  } else {

                      pool.mu.Unlock()

                  }

              }

          }

      }

}

// GetValidToken retrieves a valid access token, refreshing if necessary

func (cm *CredentialManager) GetValidToken(

  ctx context.Context,

  platform string,

  tenantID uuid.UUID,

) (*PlatformCredentials, error) {



  cacheKey := fmt.Sprintf("token:%s:%s", platform, tenantID)



  // Check cache

  cached, err := cm.cache.Get(ctx, cacheKey).Result()

  if err == nil {

      var creds PlatformCredentials

      json.Unmarshal([]byte(cached), &creds)



      // Return if still valid (with 5-minute buffer)

      if creds.ExpiresAt != nil && time.Now().Add(5*time.Minute).Before(*creds.ExpiresAt) {

          return &creds, nil

      }

  }



  // Retrieve from Vault

  creds, err := cm.vault.GetCredentials(ctx, platform, tenantID.String())

  if err != nil {

      return nil, fmt.Errorf("vault retrieval: %w", err)

  }

// Refresh if expired

if creds.ExpiresAt != nil && time.Now().After(*creds.ExpiresAt) {

    refreshedCreds, err := cm.refreshToken(ctx, platform, creds)

    if err != nil {

        return nil, fmt.Errorf("token refresh: %w", err)

    }



    // Store refreshed credentials

    cm.vault.StoreCredentials(ctx, platform, tenantID.String(), refreshedCreds)

    creds = refreshedCreds

}



// Cache with appropriate TTL

ttl := 5 * time.Minute

if creds.ExpiresAt != nil {

    timeUntilExpiry := time.Until(*creds.ExpiresAt)

    if timeUntilExpiry < ttl {

        ttl = timeUntilExpiry / 2

    }

}



data, _ := json.Marshal(creds)

cm.cache.Set(ctx, cacheKey, data, ttl)



return creds, nil

}

      case "youtube":

          return cm.refreshYouTubeToken(ctx, creds)

      default:

          return nil, fmt.Errorf("unsupported platform for token refresh: %s", platform)

      }

         StoryID:    request.StoryID,

         Type:       string(request.CorrectionType),

         Severity:    string(request.Severity),

         PlatformCount: len(result.AffectedPlatforms),

    })



    return result, nil

}



// processRetraction handles full story retraction

func (ce *CorrectionEngine) processRetraction(

    ctx context.Context,

    request *CorrectionRequest,

    publishedJobs []*PublicationJob,

) *CorrectionResult {



    result := &CorrectionResult{

         CorrectionID: uuid.NewV7(),

         StoryID:    request.StoryID,

         Type:       CorrectionRetraction,

         StartedAt: time.Now(),

         PlatformUpdates: make(map[string]*PlatformUpdateResult),

}



// Generate retraction notice

notice := ce.generateRetractionNotice(request)



for _, job := range publishedJobs {

    for _, platformJob := range job.PlatformJobs {

      if platformJob.Status != PlatformStatusPublished {

          continue

      }



      connector, err := ce.connectorManager.GetConnector(platformJob.Platform)

      if err != nil {

          continue

      }



      switch platformJob.Platform {

      case "website":

          // Update article with retraction banner

          updatedContent := job.ContentPackage

          updatedContent.RetractionNotice = notice



          _, err = connector.Update(ctx, platformJob.PlatformPostID,

            ce.buildRetractionContent(platformJob.Platform, updatedContent))



      case "twitter", "facebook", "linkedin":

                // Post retraction notice as new post referencing original

                retractionPost := ce.buildRetractionPost(platformJob.Platform, notice, platformJob)

                _, err = connector.Publish(ctx, retractionPost)



            case "youtube":

                // Update video description and title

                updatedContent := ce.buildVideoRetractionContent(notice)

                _, err = connector.Update(ctx, platformJob.PlatformPostID, updatedContent)

            }



            result.PlatformUpdates[platformJob.Platform] = &PlatformUpdateResult{

                Success: err == nil,

                Error: errStr(err),

            }

            result.AffectedPlatforms = append(result.AffectedPlatforms, platformJob.Platform)

        }

    }



    return result

}



func (ce *CorrectionEngine) generateRetractionNotice(request *CorrectionRequest)
*RetractionNotice {

    severityLabels := map[CorrectionSeverity]string{

        SeverityLow:        "Correction",

        SeverityMedium: "Correction",

          SeverityHigh:    "Significant Correction",

          SeverityCritical: "Retraction",

      }



      return &RetractionNotice{

          Title:    fmt.Sprintf("%s: %s", severityLabels[request.Severity], request.Description),

          Body:      request.Description,

          Date:     time.Now(),

          Reason:     string(request.CorrectionType),

          EditorNote: request.EditorNote,

      }

}

package domain



// DeliveryMonitor tracks publication status across all platforms

type DeliveryMonitor struct {

type DeliveryStatus struct {

    JobID        uuid.UUID              `json:"job_id"`

    StoryID       uuid.UUID              `json:"story_id"`

    OverallStatus string                `json:"overall_status"`

    PlatformStatuses []PlatformDeliveryStatus `json:"platform_statuses"`

    StartedAt      time.Time             `json:"started_at"`

    CompletedAt       *time.Time              `json:"completed_at"`

    Duration       time.Duration           `json:"duration_ms"`

}



type PlatformDeliveryStatus struct {

    Platform       string      `json:"platform"`

    Status        string    `json:"status"`

    PlatformPostID string         `json:"platform_post_id,omitempty"`

    PlatformURL      string      `json:"platform_url,omitempty"`

    AttemptCount int             `json:"attempt_count"`

    LastAttemptAt *time.Time `json:"last_attempt_at"`

     PublishedAt        *time.Time `json:"published_at,omitempty"`

     ErrorMessage string          `json:"error_message,omitempty"`

}



// TrackDelivery monitors a publication job in real-time

func (dm *DeliveryMonitor) TrackDelivery(

     ctx context.Context,

     job *PublicationJob,

){



     ticker := time.NewTicker(10 * time.Second)

     defer ticker.Stop()



     for {

       select {

       case <-ticker.C:

             status := dm.assessDeliveryStatus(ctx, job)

             dm.recordMetrics(status)



             // Check for stalled deliveries

             if dm.isStalled(status) {

               dm.alertMgr.Trigger(ctx, Alert{

                 Severity: "WARNING",

                 Title: "Publication job stalled",

                 Detail: fmt.Sprintf("Job %s has been publishing for %v",

                    job.JobID, time.Since(job.CreatedAt)),

                })

            }



            // Check completion

            if status.OverallStatus == "COMPLETED" || status.OverallStatus == "FAILED" {

                dm.finalizeDelivery(ctx, status)

                return

            }



        case <-ctx.Done():

            return

        }

    }

}



// assessDeliveryStatus evaluates the current state of all platform jobs

func (dm *DeliveryMonitor) assessDeliveryStatus(

    ctx context.Context,

    job *PublicationJob,

) *DeliveryStatus {



    status := &DeliveryStatus{

        JobID:       job.JobID,

        StoryID: job.StoryID,

        StartedAt: job.CreatedAt,

    }

completedCount := 0

failedCount := 0



for _, platformJob := range job.PlatformJobs {

  platformStatus := PlatformDeliveryStatus{

      Platform:    platformJob.Platform,

      Status:     string(platformJob.Status),

      AttemptCount: platformJob.AttemptCount,

      ErrorMessage: platformJob.ErrorMessage,

  }



  if platformJob.PlatformPostID != "" {

      platformStatus.PlatformPostID = platformJob.PlatformPostID

      platformStatus.PlatformURL = platformJob.PlatformURL

  }

  if platformJob.PublishedAt != nil {

      platformStatus.PublishedAt = platformJob.PublishedAt

  }

  if platformJob.LastAttemptAt != nil {

      platformStatus.LastAttemptAt = platformJob.LastAttemptAt

  }



  switch platformJob.Status {

  case PlatformStatusPublished:

      completedCount++

    case PlatformStatusFailed:

        failedCount++

    }



    status.PlatformStatuses = append(status.PlatformStatuses, platformStatus)

}



totalPlatforms := len(job.PlatformJobs)



if completedCount == totalPlatforms {

    status.OverallStatus = "COMPLETED"

    now := time.Now()

    status.CompletedAt = &now

    status.Duration = now.Sub(job.CreatedAt)

} else if failedCount == totalPlatforms {

    status.OverallStatus = "FAILED"

    now := time.Now()

    status.CompletedAt = &now

    status.Duration = now.Sub(job.CreatedAt)

} else if completedCount > 0 {

    status.OverallStatus = "PARTIAL"

} else {

    status.OverallStatus = "IN_PROGRESS"

}



return status

}
