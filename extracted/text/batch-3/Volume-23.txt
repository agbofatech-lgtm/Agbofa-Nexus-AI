SERVICE DECISION RECORDS (SDR)



SDR-001: Argon2id for Password Hashing



Decision: Use Argon2id with calibrated parameters (time=3, memory=65536, threads=4) for all
password hashing.

Rationale: Argon2id is the OWASP-recommended algorithm, resistant to GPU and side-channel
attacks. Calibrated parameters provide ~500ms hashing time, balancing security and user
experience.

Alternatives: bcrypt (rejected — less memory-hard, more vulnerable to GPU attacks); scrypt
(rejected — less studied than Argon2id).

Trade-offs: Higher memory usage per hash, but acceptable for authentication service scale.

SDR-002: JWT with Asymmetric Signing (Ed25519)



Decision: Sign all JWTs using Ed25519 asymmetric keys, with key rotation every 30 days via
Vault.

Rationale: Ed25519 provides 128-bit security with small signatures and fast verification.
Asymmetric signing allows any service to verify tokens without sharing secrets.

Alternatives: RS256 (rejected — larger keys and signatures); HS256 (rejected — symmetric,
requires secret sharing).

Trade-offs: Key management complexity, but significantly better security posture.



SDR-003: Row-Level Security for Multi-Tenancy



Decision: Enforce tenant isolation at the database level using PostgreSQL Row-Level Security
(RLS) policies, with application-level tenant context injection.

Rationale: Defense in depth — even if application logic fails, database prevents cross-tenant
data access. RLS is more reliable than application-level filtering alone.

Alternatives: Separate databases per tenant (rejected — operational complexity); application-
level only (rejected — insufficient isolation).

Trade-offs: Slight query overhead, but essential for multi-tenant security.



SDR-004: SPIFFE/SPIRE for Service Identity



Decision: All service-to-service authentication uses SPIFFE/SPIRE-issued X.509 certificates with
mTLS, enforced by Istio Ambient Mesh.

Rationale: SPIFFE provides cryptographically verifiable workload identity without manual
certificate management. Automatic rotation eliminates certificate expiry risks.

Alternatives: Manual certificate management (rejected — operational burden); API keys
(rejected — weaker security model).

Trade-offs: SPIRE infrastructure overhead, but essential for zero-trust architecture.

2.1 Service Overview



Service ID: SVC-FND-TIS

Runtime: Golang 1.22+

gRPC Port: 9091

Health Port: 9090

Metrics Port: 9092



The Tenant & Identity Service is the root trust domain for all human and non-human actors in
the Agbofa Nexus AI platform. It manages the complete lifecycle of tenants, users,
authentication, and session management.

2.5 JWT Token Structure



```go

// internal/domain/token.go

package domain

import (

    "crypto/ed25519"

    "time"



    "github.com/golang-jwt/jwt/v5"

)



// TokenClaims represents the JWT claims structure

type TokenClaims struct {

    jwt.RegisteredClaims



    // Tenant context

    TenantID string `json:"tenant_id"`

    TenantSlug string `json:"tenant_slug"`



    // User context

    UserID       string `json:"user_id"`

    Email    string `json:"email"`

    DisplayName string `json:"display_name"`



    // Authorization

    Roles    []string `json:"roles"`

    Permissions []string `json:"permissions,omitempty"`



    // Session

    SessionID string `json:"session_id"`



    // Security

    TokenID        string `json:"jti"`

    IssuedAt int64 `json:"iat"`

    ExpiresAt int64 `json:"exp"`

    NotBefore int64 `json:"nbf"`

}



// TokenService manages JWT creation and validation

type TokenService struct {

    privateKey ed25519.PrivateKey

    publicKey ed25519.PublicKey

    keyID       string

    issuer      string

    ttl       time.Duration

}



// NewTokenService creates a token service with key rotation support

func NewTokenService(vaultClient VaultClient) (*TokenService, error) {

    // Fetch current signing key from Vault

    key, err := vaultClient.GetSigningKey(context.Background())

    if err != nil {

          return nil, fmt.Errorf("get signing key: %w", err)

    }

    return &TokenService{

       privateKey: key.Private,

       publicKey: key.Public,

       keyID:        key.ID,

       issuer:       "agbofa-nexus",

       ttl:        15 * time.Minute, // Access token: 15 minutes

    }, nil

}



// CreateAccessToken generates a new JWT access token

func (ts *TokenService) CreateAccessToken(user *User, session *Session) (string, error) {

    now := time.Now()



    claims := &TokenClaims{

       RegisteredClaims: jwt.RegisteredClaims{

             Issuer: ts.issuer,

             Subject: user.ID.String(),

             Audience: jwt.ClaimStrings{"agbofa-nexus"},

             ExpiresAt: jwt.NewNumericDate(now.Add(ts.ttl)),

             NotBefore: jwt.NewNumericDate(now),

             IssuedAt: jwt.NewNumericDate(now),

             ID:     uuid.NewV7().String(),

       },

       TenantID: user.TenantID.String(),

       UserID:        user.ID.String(),

       Email:        user.Email,

        DisplayName: user.DisplayName,

        Roles:        user.Roles,

        SessionID: session.ID.String(),

    }



    token := jwt.NewWithClaims(jwt.SigningMethodEdDSA, claims)

    token.Header["kid"] = ts.keyID



    signed, err := token.SignedString(ts.privateKey)

    if err != nil {

        return "", fmt.Errorf("sign token: %w", err)

    }



    return signed, nil

}



// ValidateAccessToken validates and parses a JWT access token

func (ts *TokenService) ValidateAccessToken(tokenString string) (*TokenClaims, error) {

    token, err := jwt.ParseWithClaims(tokenString, &TokenClaims{},

        func(token *jwt.Token) (interface{}, error) {

          // Verify signing algorithm

          if _, ok := token.Method.(*jwt.SigningMethodEd25519); !ok {

              return nil, fmt.Errorf("unexpected signing method: %v", token.Header["alg"])

          }



          // Verify key ID matches current key

               if token.Header["kid"] != ts.keyID {

                   // Could be rotated — check Vault for previous key

                   return ts.getKeyByID(token.Header["kid"].(string))

               }



               return ts.publicKey, nil

          },

          jwt.WithLeeway(30*time.Second), // 30 second clock skew tolerance

      )



      if err != nil {

          return nil, fmt.Errorf("parse token: %w", err)

      }



      claims, ok := token.Claims.(*TokenClaims)

      if !ok || !token.Valid {

          return nil, ErrInvalidToken

      }



      return claims, nil

}
```

```go

// internal/domain/refresh_token.go

package domain



import (

    "crypto/rand"

    "crypto/sha256"

    "encoding/hex"

    "time"

)



// RefreshTokenService manages opaque refresh tokens with rotation

type RefreshTokenService struct {

    store RefreshTokenStore

}



// GenerateRefreshToken creates a new cryptographically random refresh token

func (rts *RefreshTokenService) GenerateRefreshToken() (string, error) {

    // 256-bit random token

    bytes := make([]byte, 32)

    if _, err := rand.Read(bytes); err != nil {

        return "", fmt.Errorf("generate random: %w", err)

    }



    // URL-safe base64 encoding

    token := base64.RawURLEncoding.EncodeToString(bytes)

    return token, nil

}



// StoreRefreshToken hashes and stores the refresh token

func (rts *RefreshTokenService) StoreRefreshToken(

    ctx context.Context,

    token string,

    sessionID uuid.UUID,

    expiresAt time.Time,

) error {

    // Hash the token with SHA-256 before storage

    hash := sha256.Sum256([]byte(token))

    tokenHash := hex.EncodeToString(hash[:])



    return rts.store.Store(ctx, &StoredRefreshToken{

         TokenHash: tokenHash,

         SessionID: sessionID,

         ExpiresAt: expiresAt,

         CreatedAt: time.Now(),

    })

}



// RotateRefreshToken implements token rotation:

// 1. Validate old token

// 2. Revoke old token

// 3. Issue new token

// This prevents refresh token replay attacks

func (rts *RefreshTokenService) RotateRefreshToken(

  ctx context.Context,

  oldToken string,

) (*TokenRotation, error) {



  // 1. Hash the old token

  oldHash := sha256.Sum256([]byte(oldToken))

  oldTokenHash := hex.EncodeToString(oldHash[:])



  // 2. Find and validate the stored token

  stored, err := rts.store.FindByHash(ctx, oldTokenHash)

  if err != nil {

      return nil, ErrInvalidRefreshToken

  }



  // 3. Check expiration

  if time.Now().After(stored.ExpiresAt) {

      return nil, ErrRefreshTokenExpired

  }



  // 4. Check if already revoked (replay attack detection)

  if stored.RevokedAt != nil {

      // Token reuse detected — revoke ALL tokens for this user

      rts.store.RevokeAllForUser(ctx, stored.UserID)

      return nil, ErrTokenReuseDetected

  }

      // 5. Revoke old token

      if err := rts.store.Revoke(ctx, oldTokenHash); err != nil {

          return nil, fmt.Errorf("revoke old token: %w", err)

      }



      // 6. Generate new token

      newToken, err := rts.GenerateRefreshToken()

      if err != nil {

          return nil, fmt.Errorf("generate new token: %w", err)

      }



      // 7. Store new token with same session

      newExpiresAt := time.Now().Add(7 * 24 * time.Hour) // 7 days

      if err := rts.StoreRefreshToken(ctx, newToken, stored.SessionID, newExpiresAt); err != nil {

          return nil, fmt.Errorf("store new token: %w", err)

      }



      return &TokenRotation{

          NewToken: newToken,

          SessionID: stored.SessionID,

          ExpiresAt: newExpiresAt,

      }, nil

}
```

```go
// internal/domain/password.go

package domain



import (

    "crypto/subtle"

    "fmt"



    "golang.org/x/crypto/argon2"

)



// Argon2id parameters (OWASP recommended)

const (

    argon2Time = 3      // Number of iterations

    argon2Memory = 65536 // 64 MB

    argon2Threads = 4    // Number of threads

    argon2KeyLen = 32 // 256-bit key

    argon2SaltLen = 16 // 128-bit salt

)



// PasswordService manages password hashing and verification

type PasswordService struct{}



// HashPassword hashes a password using Argon2id

func (ps *PasswordService) HashPassword(password string) (string, error) {

  // Generate random salt

  salt := make([]byte, argon2SaltLen)

  if _, err := rand.Read(salt); err != nil {

      return "", fmt.Errorf("generate salt: %w", err)

  }



  // Hash password with Argon2id

  hash := argon2.IDKey(

      []byte(password),

      salt,

      argon2Time,

      argon2Memory,

      argon2Threads,

      argon2KeyLen,

  )



  // Encode as: $argon2id$v=19$m=65536,t=3,p=4$<salt>$<hash>

  encoded := fmt.Sprintf(

      "$argon2id$v=19$m=%d,t=%d,p=%d$%s$%s",

      argon2Memory,

      argon2Time,

      argon2Threads,

      base64.RawStdEncoding.EncodeToString(salt),

      base64.RawStdEncoding.EncodeToString(hash),

  )

    return encoded, nil

}



// VerifyPassword verifies a password against its hash using constant-time comparison

func (ps *PasswordService) VerifyPassword(password, encodedHash string) (bool, error) {

    // Parse encoded hash

    params, salt, hash, err := ps.decodeHash(encodedHash)

    if err != nil {

        return false, fmt.Errorf("decode hash: %w", err)

    }



    // Recompute hash with same parameters

    computedHash := argon2.IDKey(

        []byte(password),

        salt,

        params.iterations,

        params.memory,

        params.threads,

        params.keyLen,

    )



    // Constant-time comparison to prevent timing attacks

    return subtle.ConstantTimeCompare(hash, computedHash) == 1, nil

}

// ValidatePasswordStrength enforces password policies

func (ps *PasswordService) ValidatePasswordStrength(password string) error {

  if len(password) < 8 {

      return ErrPasswordTooShort

  }



  if len(password) > 128 {

      return ErrPasswordTooLong

  }



  hasUpper := false

  hasLower := false

  hasDigit := false

  hasSpecial := false



  for _, ch := range password {

      switch {

      case 'A' <= ch && ch <= 'Z':

          hasUpper = true

      case 'a' <= ch && ch <= 'z':

          hasLower = true

      case '0' <= ch && ch <= '9':

          hasDigit = true

      case isSpecialChar(ch):

          hasSpecial = true

      }

      }



      if !hasUpper {

          return ErrPasswordNoUppercase

      }

      if !hasLower {

          return ErrPasswordNoLowercase

      }

      if !hasDigit {

          return ErrPasswordNoDigit

      }

      if !hasSpecial {

          return ErrPasswordNoSpecial

      }



      return nil

}
```

```go
// internal/domain/authorization.go

package domain



// AuthorizationService evaluates access policies

type AuthorizationService struct {

      opaClient *opa.Client

      policyStore PolicyStore

      cache     *redis.Client

}



// AuthorizationRequest represents an access check

type AuthorizationRequest struct {

    Subject Subject `json:"subject"`

    Action string `json:"action"`

    Resource Resource `json:"resource"`

    Context Context `json:"context"`

}



type Subject struct {

    UserID string `json:"user_id"`

    TenantID string `json:"tenant_id"`

    Roles    []string `json:"roles"`

}



type Resource struct {

    Type     string `json:"type"`      // "story", "article", "source", "tenant"

    ID      string `json:"id"`      // Resource identifier

    TenantID string `json:"tenant_id"` // Owning tenant

    Attributes map[string]interface{} `json:"attributes,omitempty"`

}



type Context struct {

    IPAddress string `json:"ip_address"`

    UserAgent string `json:"user_agent"`

    Timestamp int64 `json:"timestamp"`

    IsEmergency bool `json:"is_emergency,omitempty"`

}

// AuthorizationResponse represents the authorization decision

type AuthorizationResponse struct {

    Allowed bool      `json:"allowed"`

    Reason string `json:"reason,omitempty"`

  Obligations []string `json:"obligations,omitempty"` // Required actions (e.g., "LOG_AUDIT",
"NOTIFY_ADMIN")

}



// CheckAuthorization evaluates whether a subject can perform an action on a resource

func (as *AuthorizationService) CheckAuthorization(

    ctx context.Context,

    req *AuthorizationRequest,

) (*AuthorizationResponse, error) {



    // Check cache first

    cacheKey := as.cacheKey(req)

    if cached, err := as.cache.Get(ctx, cacheKey); err == nil {

        var resp AuthorizationResponse

        json.Unmarshal([]byte(cached), &resp)

        return &resp, nil

    }



    // Evaluate OPA policy

    input := map[string]interface{}{

        "subject": req.Subject,

        "action": req.Action,

          "resource": req.Resource,

          "context": req.Context,

      }



      result, err := as.opaClient.Evaluate(ctx, "data.agbofa.authz.allow", input)

      if err != nil {

          return nil, fmt.Errorf("opa evaluation: %w", err)

      }



      resp := &AuthorizationResponse{

          Allowed: result.Allowed,

          Reason: result.Reason,

      }



      if result.Allowed && len(result.Obligations) > 0 {

          resp.Obligations = result.Obligations

      }



      // Cache decision (short TTL for real-time policy updates)

      as.cache.Set(ctx, cacheKey, resp, 30*time.Second)



      return resp, nil

}
```

7.2 Secret Rotation Implementation



```go

// internal/domain/key_rotation.go

package domain



// KeyRotationService manages automated key rotation

type KeyRotationService struct {

      vault VaultClient

      store KeyStore

      current *SigningKey

      mu    sync.RWMutex

}



type SigningKey struct {

      ID    string

      Public ed25519.PublicKey

    Private ed25519.PrivateKey

    CreatedAt time.Time

    ExpiresAt time.Time

}



// RotateKeys performs automated key rotation

func (krs *KeyRotationService) RotateKeys(ctx context.Context) error {

    krs.mu.Lock()

    defer krs.mu.Unlock()



    // 1. Generate new key pair

    public, private, err := ed25519.GenerateKey(rand.Reader)

    if err != nil {

        return fmt.Errorf("generate key pair: %w", err)

    }



    newKey := &SigningKey{

        ID:     fmt.Sprintf("key-%d", time.Now().Unix()),

        Public: public,

        Private: private,

        CreatedAt: time.Now(),

        ExpiresAt: time.Now().Add(60 * 24 * time.Hour), // 60 days (overlap with 30-day rotation)

    }



    // 2. Store private key in Vault

    if err := krs.vault.StoreSigningKey(ctx, newKey); err != nil {

          return fmt.Errorf("store key in vault: %w", err)

      }



      // 3. Store public key metadata for verification

      if err := krs.store.StorePublicKey(ctx, newKey); err != nil {

          return fmt.Errorf("store public key: %w", err)

      }



      // 4. Update current key (new tokens use this)

      oldKey := krs.current

      krs.current = newKey



      // 5. Schedule old key archival (keep for token verification during overlap)

      go krs.archiveKeyAfterWindow(ctx, oldKey, 30*24*time.Hour)



      return nil

}

```
