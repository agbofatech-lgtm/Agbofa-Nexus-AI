SERVICE DECISION RECORDS (SDR)



SDR-COR-001: Dual-Mode Ingestion Architecture



Decision: Implement two ingestion modes: poll-based for RSS/API sources (configurable
intervals) and stream-based for WebSocket/real-time sources. Both feed into a unified Kafka
topic for downstream processing.

Rationale: Different source types have different capabilities. Polling handles 80% of sources;
streaming handles breaking news and real-time APIs. Unified Kafka topic normalizes all ingest
into a single pipeline.

Alternatives: Poll-only (rejected — too slow for breaking news); stream-only (rejected — most
sources don't support streaming).

Trade-offs: Two code paths to maintain, but necessary for comprehensive coverage.



SDR-COR-002: pgvector for Story Embedding Similarity



Decision: Use PostgreSQL with pgvector extension as the primary vector store for story
similarity, deduplication, and clustering. Neo4j stores graph relationships.

Rationale: Simplifies operational stack by using PostgreSQL for both relational and vector data.
pgvector performance is sufficient for expected scale (500K documents/day). Neo4j handles
graph-specific queries.

Alternatives: Dedicated vector DB (Pinecone, Weaviate — rejected for operational simplicity);
pgvector-only for graph (rejected — graph queries too complex for relational).

Trade-offs: pgvector not as fast as dedicated vector DBs, but sufficient for our scale and simpler
to operate.



SDR-COR-003: Pluggable Source Connector Framework

Decision: Source connectors implement a common SourceConnector interface with plugin
registration. New source types added without modifying core ingestion engine.

Rationale: 10,000+ sources across dozens of types. Plugin architecture enables independent
development, testing, and deployment of connectors.

Alternatives: Hardcoded source types (rejected — not scalable); external integration platform
(rejected — latency, cost).

Trade-offs: Plugin management complexity, but essential for scale.



SDR-COR-004: Content Fingerprinting for Deduplication



Decision: Generate content fingerprints using MinHash + LSH (Locality-Sensitive Hashing) for
near-duplicate detection, combined with SHA-256 for exact deduplication.

Rationale: News content often has minor variations (different headlines, formatting).
MinHash+LSH catches near-duplicates that SHA-256 misses. Two-tier approach balances
accuracy and performance.

Alternatives: SHA-256 only (rejected — misses near-duplicates); full semantic comparison
(rejected — too expensive for 500K docs/day).

Trade-offs: Additional fingerprint computation, but 90%+ duplicate detection accuracy.

type SourceConnector interface {

    // Identity

    ID() string

    Name() string

    SourceType() SourceType

    Version() string



    // Lifecycle

    Initialize(ctx context.Context, config SourceConfig) error

    HealthCheck(ctx context.Context) (*SourceHealth, error)

    // Data Fetching

    Fetch(ctx context.Context, opts FetchOptions) (*FetchResult, error)

    RawData         json.RawMessage       `json:"raw_data,omitempty"` // Original payload

// RSSConnector implements SourceConnector for RSS/Atom feeds

func (rc *RSSConnector) Name() string                     { return rc.name }

    return rc.config.PollInterval

}



// Fetch retrieves RSS feed items and converts them to RawDocuments

func (rc *RSSConnector) Fetch(ctx context.Context, opts FetchOptions) (*FetchResult, error) {

    ctx, span := rc.tracer.Start(ctx, "rss.Fetch")

  defer span.End()



  startTime := time.Now()



  // Build request with conditional GET

  req, err := http.NewRequestWithContext(ctx, "GET", rc.config.BaseURL, nil)

  if err != nil {

      return nil, fmt.Errorf("create request: %w", err)

  }



  req.Header.Set("User-Agent", "AgbofaNexusAI/1.0 (News Aggregator)")

  req.Header.Set("Accept", "application/rss+xml, application/atom+xml, application/xml,
text/xml")



  // Add ETag/Last-Modified for conditional fetch

  if opts.Since != nil {

      req.Header.Set("If-Modified-Since", opts.Since.Format(http.TimeFormat))

  }



  // Execute request

  resp, err := rc.httpClient.Do(req)

  if err != nil {

      rc.metrics.RecordError("fetch")

      return nil, fmt.Errorf("http request: %w", err)

  }

  defer resp.Body.Close()

// Handle rate limiting

if resp.StatusCode == http.StatusTooManyRequests {

    retryAfter := resp.Header.Get("Retry-After")

    rc.metrics.RecordRateLimited()

    return nil, fmt.Errorf("rate limited, retry after: %s", retryAfter)

}



// Not modified

if resp.StatusCode == http.StatusNotModified {

    return &FetchResult{

       Documents: []*RawDocument{},

       HasMore: false,

       FetchedAt: time.Now(),

       Metadata: FetchMetadata{

             ItemsFetched: 0,

             ItemsNew:    0,

             Duration:   time.Since(startTime),

       },

    }, nil

}



if resp.StatusCode != http.StatusOK {

    return nil, fmt.Errorf("unexpected status: %d", resp.StatusCode)

}

// Parse feed

body, _ := io.ReadAll(resp.Body)

feed, err := rc.parser.ParseString(string(body))

if err != nil {

    return nil, fmt.Errorf("parse feed: %w", err)

}



// Convert to RawDocuments

documents := make([]*RawDocument, 0, len(feed.Items))

newCount := 0



for _, item := range feed.Items {

    // Generate stable external ID from GUID or URL

    externalID := rc.generateExternalID(item)



    // Parse published date

    var publishedAt *time.Time

    if item.PublishedParsed != nil {

        publishedAt = item.PublishedParsed

    } else if item.UpdatedParsed != nil {

        publishedAt = item.UpdatedParsed

    }



    // Skip if older than since time

    if opts.Since != nil && publishedAt != nil && publishedAt.Before(*opts.Since) {

        continue

}



doc := &RawDocument{

    DocumentID: uuid.NewV7().String(),

    ExternalID: externalID,

    SourceID: rc.config.SourceID,

    TenantID: rc.config.TenantID,

    ContentType: "article",

    Title:      item.Title,

    Body:          rc.extractBody(item),

    RawHTML:           item.Content,

    URL:           item.Link,

    PublishedAt: publishedAt,

    FetchedAt: time.Now(),

    Authors:        rc.extractAuthors(item),

    Categories: item.Categories,

    MediaURLs: rc.extractMediaURLs(item),

    Metadata: map[string]interface{}{

         "feed_title": feed.Title,

         "guid":      item.GUID,

    },

}



documents = append(documents, doc)

newCount++

        // Respect limit

        if opts.Limit > 0 && len(documents) >= opts.Limit {

             break

        }

    }



    duration := time.Since(startTime)

    rc.metrics.RecordFetch(duration, len(documents), newCount)



    return &FetchResult{

        Documents: documents,

        HasMore: len(documents) >= opts.Limit,

        FetchedAt: time.Now(),

        Metadata: FetchMetadata{

             ItemsFetched: len(documents),

             ItemsNew:     newCount,

             Duration:   duration,

        },

    }, nil

}



func (rc *RSSConnector) generateExternalID(item *gofeed.Item) string {

    // Prefer GUID, fall back to URL hash

    if item.GUID != "" {

        hash := sha256.Sum256([]byte(item.GUID))

        return hex.EncodeToString(hash[:16])

    }

    if item.Link != "" {

        hash := sha256.Sum256([]byte(item.Link))

        return hex.EncodeToString(hash[:16])

    }

    // Last resort: hash title + published date

    data := item.Title

    if item.PublishedParsed != nil {

        data += item.PublishedParsed.String()

    }

    hash := sha256.Sum256([]byte(data))

    return hex.EncodeToString(hash[:16])

}



func (rc *RSSConnector) extractBody(item *gofeed.Item) string {

    if item.Content != "" {

        return stripHTML(item.Content)

    }

    return item.Description

}



func (rc *RSSConnector) extractAuthors(item *gofeed.Item) []string {

    authors := make([]string, 0, len(item.Authors))

    for _, author := range item.Authors {

        if author.Name != "" {

          authors = append(authors, author.Name)

          }

      }

      return authors

}



func (rc *RSSConnector) extractMediaURLs(item *gofeed.Item) []string {

      urls := make([]string, 0)

      if item.Image != nil && item.Image.URL != "" {

          urls = append(urls, item.Image.URL)

      }

      // Extract from enclosures

      for _, enc := range item.Enclosures {

          if enc.URL != "" {

              urls = append(urls, enc.URL)

          }

      }

      return urls

}

    defer span.End()



    // Build API request based on source configuration

    reqURL := nc.buildRequestURL(opts)



    req, err := http.NewRequestWithContext(ctx, "GET", reqURL, nil)

    if err != nil {

         return nil, fmt.Errorf("create request: %w", err)

    // Add authentication

// Execute with retry

var resp *http.Response

for attempt := 0; attempt < 3; attempt++ {

    resp, err = nc.httpClient.Do(req)

    if err == nil && resp.StatusCode < 500 {

        break

    }

    if attempt < 2 {

        time.Sleep(time.Duration(1<<attempt) * time.Second)

    }

}

if err != nil {

    return nil, fmt.Errorf("api request failed after retries: %w", err)

}

defer resp.Body.Close()



// Handle rate limiting

nc.updateRateLimits(resp)



// Parse response based on API format

var documents []*RawDocument

switch nc.config.CustomParams["response_format"] {

case "reuters":

    documents, err = nc.parseReutersResponse(resp.Body)

case "associated_press":

          documents, err = nc.parseAPResponse(resp.Body)

      default:

          documents, err = nc.parseGenericJSONResponse(resp.Body)

      }



      if err != nil {

          return nil, fmt.Errorf("parse response: %w", err)

      }



      return &FetchResult{

          Documents: documents,

          HasMore: nc.hasMorePages(resp),

// services/content-origination/ingestion-orchestrator/internal/application/orchestrator.go

package application



// IngestionOrchestrator coordinates all source connectors and manages the ingestion pipeline

type IngestionOrchestrator struct {

    sourceRepo         domain.SourceRepository

    connectorReg *ConnectorRegistry

    scheduler         *PollScheduler

    normalizer        NormalizerClient

    eventProducer events.Producer

    healthMonitor *SourceHealthMonitor

    rateLimiter       *RateLimiter

}



// Start begins the ingestion orchestration

func (io *IngestionOrchestrator) Start(ctx context.Context) error {

    // 1. Load all active sources

    sources, err := io.sourceRepo.ListActive(ctx)

    if err != nil {

        return fmt.Errorf("list active sources: %w", err)

    }



    // 2. Initialize connectors for each source

    for _, source := range sources {

    connector, err := io.connectorReg.CreateConnector(source)

    if err != nil {

        io.healthMonitor.RecordFailure(source.ID, err)

        continue

    }



    // Register connector

    io.connectorReg.Register(source.ID, connector)



    // Schedule polling if supported

    if connector.SupportsScheduling() {

        io.scheduler.Schedule(source.ID, source.PollInterval, func() {

             io.pollSource(ctx, source.ID)

        })

    }



    // Start streaming if supported

    if connector.SupportsStreaming() {

        go io.streamSource(ctx, source.ID, connector)

    }

}



// 3. Start scheduler

io.scheduler.Start(ctx)



// 4. Start health monitor

    go io.healthMonitor.Start(ctx)



    return nil

}



// pollSource fetches content from a single source

func (io *IngestionOrchestrator) pollSource(ctx context.Context, sourceID string) {

    ctx, span := otel.Tracer("orchestrator").Start(ctx, "pollSource",

        trace.WithAttributes(attribute.String("source_id", sourceID)))

    defer span.End()



    // Rate limiting check

    if err := io.rateLimiter.Wait(ctx, sourceID); err != nil {

        log.Warn().Err(err).Str("source_id", sourceID).Msg("Rate limit wait failed")

        return

    }



    // Get connector

    connector, err := io.connectorReg.Get(sourceID)

    if err != nil {

        log.Error().Err(err).Str("source_id", sourceID).Msg("Connector not found")

        return

    }



    // Get last fetch time

    lastFetch, _ := io.sourceRepo.GetLastFetchTime(ctx, sourceID)

    // Fetch with since parameter

    result, err := connector.Fetch(ctx, domain.FetchOptions{

         Since: lastFetch,

         Limit: 100, // Configurable per source

    })



    if err != nil {

         io.healthMonitor.RecordFailure(sourceID, err)

         io.handleFetchError(ctx, sourceID, err)

         return

    }



    // Process fetched documents

    for _, doc := range result.Documents {

         if err := io.processDocument(ctx, doc); err != nil {

             log.Error().Err(err).Str("doc_id", doc.DocumentID).Msg("Document processing failed")

             continue

         }

    }



    // Update last fetch time

    io.sourceRepo.UpdateLastFetchTime(ctx, sourceID, result.FetchedAt)

    io.healthMonitor.RecordSuccess(sourceID, result.Metadata.Duration)

}

// processDocument handles a single raw document through the pipeline

func (io *IngestionOrchestrator) processDocument(ctx context.Context, doc
*domain.RawDocument) error {

    // 1. Normalize content

    normalized, err := io.normalizer.Normalize(ctx, doc)

    if err != nil {

        return fmt.Errorf("normalize: %w", err)

    }



    // 2. Publish to raw ingest topic

    event := &events.RawContentIngestedEvent{

        BaseEvent: events.NewBaseEvent(doc.TenantID),

        DocumentID: doc.DocumentID,

        SourceID:     doc.SourceID,

        NormalizedDoc: normalized,

    }



    if err := io.eventProducer.Publish(ctx, event); err != nil {

        return fmt.Errorf("publish event: %w", err)

    }



    return nil

}



// handleFetchError processes fetch errors with appropriate retry/backoff

func (io *IngestionOrchestrator) handleFetchError(ctx context.Context, sourceID string, err
error) {

      var retryable bool

      var backoff time.Duration



      switch {

      case errors.Is(err, ErrRateLimited):

          retryable = true

          backoff = 5 * time.Minute

      case errors.Is(err, ErrNetworkTimeout):

          retryable = true

          backoff = 1 * time.Minute

      case errors.Is(err, ErrAuthenticationFailed):

          retryable = false

          io.healthMonitor.MarkUnhealthy(sourceID, "Authentication failed")

      default:

          retryable = true

          backoff = 30 * time.Second

      }



      if retryable {

          io.scheduler.ScheduleRetry(sourceID, backoff)

      }

}

package infrastructure



// PollScheduler manages polling intervals for all sources

type PollScheduler struct {

    tasks   map[string]*ScheduledTask

    redis   *redis.Client

    mu      sync.RWMutex

}



type ScheduledTask struct {

    SourceID string

    Interval time.Duration

    Handler func()

    NextRun time.Time

    RetryCount int

}



func (ps *PollScheduler) Schedule(sourceID string, interval time.Duration, handler func()) {

    ps.mu.Lock()

    defer ps.mu.Unlock()



    ps.tasks[sourceID] = &ScheduledTask{

      SourceID: sourceID,

      Interval: interval,

        Handler: handler,

        NextRun: time.Now().Add(interval), // Stagger initial run

    }

}



func (ps *PollScheduler) ScheduleRetry(sourceID string, backoff time.Duration) {

    ps.mu.Lock()

    defer ps.mu.Unlock()



    if task, exists := ps.tasks[sourceID]; exists {

        task.NextRun = time.Now().Add(backoff)

        task.RetryCount++

    }

}



func (ps *PollScheduler) Start(ctx context.Context) {

    ticker := time.NewTicker(1 * time.Second)

    defer ticker.Stop()



    for {

        select {

        case <-ticker.C:

            ps.runDueTasks(ctx)

        case <-ctx.Done():

            return

        }

    }

}



func (ps *PollScheduler) runDueTasks(ctx context.Context) {

    ps.mu.RLock()

    now := time.Now()

    var dueTasks []*ScheduledTask



    for _, task := range ps.tasks {

        if now.After(task.NextRun) {

            dueTasks = append(dueTasks, task)

        }

    }

    ps.mu.RUnlock()



    // Execute due tasks in parallel

    var wg sync.WaitGroup

    semaphore := make(chan struct{}, 10) // Max concurrent polls



    for _, task := range dueTasks {

        wg.Add(1)

        go func(t *ScheduledTask) {

            defer wg.Done()

            semaphore <- struct{}{}

            defer func() { <-semaphore }()

            t.Handler()



            // Schedule next run

            ps.mu.Lock()

            t.NextRun = time.Now().Add(t.Interval)

            ps.mu.Unlock()

          }(task)

      }



      wg.Wait()

}

class TextCleaner:

  """Cleans and normalizes raw text content."""



  async def clean(self, raw_doc: RawDocument) -> CleanedText:

    # Remove HTML tags

    body = self.strip_html(raw_doc.body or raw_doc.raw_html or "")



    # Remove extra whitespace

    body = self.normalize_whitespace(body)



    # Remove boilerplate (headers, footers, ads)

    body = await self.remove_boilerplate(body)



    # Clean title

  title = self.normalize_whitespace(raw_doc.title or "")

  title = self.decode_html_entities(title)



  return CleanedText(title=title, body=body)



def strip_html(self, html: str) -> str:

  """Strip HTML tags and extract text content."""

  from bs4 import BeautifulSoup

  soup = BeautifulSoup(html, 'html.parser')



  # Remove script and style elements

  for script in soup(["script", "style", "nav", "footer", "header"]):

    script.decompose()



  return soup.get_text(separator=' ', strip=True)



async def remove_boilerplate(self, text: str) -> str:

  """Remove common boilerplate content."""

  import re



  # Remove common patterns

  patterns = [

    r'Advertisement\s*$',

    r'Share this article',

    r'Follow us on',

    r'Copyright \d{4}',

        r'All rights reserved',

        r'Read more at',

        r'Originally published',

    ]



    for pattern in patterns:

        text = re.sub(pattern, '', text, flags=re.IGNORECASE)



    return text.strip()



  def normalize_whitespace(self, text: str) -> str:

    """Normalize whitespace to single spaces."""

    import re

    return re.sub(r'\s+', ' ', text).strip()



  def decode_html_entities(self, text: str) -> str:

    """Decode HTML entities."""

    import html

    return html.unescape(text)




class FingerprintGenerator:

  """Generates content fingerprints for deduplication."""



  def generate(self, text: str) -> tuple[str, str]:

    """Generate exact hash and SimHash for a text."""

        import hashlib

        from simhash import Simhash



        # Exact hash (SHA-256)

        exact_hash = hashlib.sha256(text.encode('utf-8')).hexdigest()



        # SimHash for near-duplicate detection

        simhash = str(Simhash(text.split()).value)



        return exact_hash, simhash

class LanguageDetector:

      """Multi-model language detection with confidence scoring."""



      SUPPORTED_LANGUAGES = {

        'en', 'es', 'fr', 'de', 'ar', 'zh', 'ja', 'pt', 'hi', 'sw',

        'ru', 'it', 'nl', 'ko', 'tr', 'pl', 'vi', 'th', 'id', 'ms',

    # ... 20+ more languages

}



def __init__(self):

    self.fasttext_model = self.load_fasttext()

    self.lingua_model = self.load_lingua()



async def detect(self, text: str) -> tuple[str, float]:

    """

    Detect language with confidence score.

    Uses ensemble of models for accuracy.

    """

    # FastText for primary detection

    lang1, conf1 = self.detect_fasttext(text)



    # If confidence is high enough, return immediately

    if conf1 > 0.95:

      return lang1, conf1



    # Fall back to Lingua for ambiguous cases

    lang2, conf2 = self.detect_lingua(text)



    # Ensemble: if both agree, use higher confidence

    if lang1 == lang2:

      return lang1, max(conf1, conf2)

        # Disagreement: use the higher confidence result

        if conf1 >= conf2:

           return lang1, conf1 * 0.9 # Penalize disagreement

        return lang2, conf2 * 0.9



      def detect_fasttext(self, text: str) -> tuple[str, float]:

        """FastText-based language detection."""

        import fasttext



        # Use pre-trained language identification model

        predictions = self.fasttext_model.predict(text.replace('\n', ' '), k=1)



        lang = predictions[0][0].replace('__label__', '')

        confidence = predictions[1][0]



        return lang, confidence



      def detect_lingua(self, text: str) -> tuple[str, float]:

        """Lingua-based language detection (better for short text)."""

        from lingua import LanguageDetectorBuilder



        result = self.lingua_model.detect_language_of(text)

        if result:

           return result.iso_code_639_1.name.lower(), result.confidence

        return 'en', 0.0 # Default to English

    DuplicateTypeNone            DuplicateType = "NONE"

    DuplicateTypeExact        DuplicateType = "EXACT"       // SHA-256 match

    DuplicateTypeNearExact DuplicateType = "NEAR_EXACT" // SimHash match

// CheckDuplicates performs multi-tier duplicate detection

func (ds *DeduplicationService) CheckDuplicates(

    ctx context.Context,

    doc *NormalizedDocument,

) (*DuplicateResult, error) {



    // Tier 1: Exact hash match (SHA-256)

  if existing, err := ds.fingerprintDB.FindByExactHash(ctx, doc.ExactHash); err == nil && existing
!= nil {

      return &DuplicateResult{

          DocumentID:      doc.DocumentID,

          IsDuplicate:   true,

          DuplicateType: DuplicateTypeExact,

          MatchedDocID: existing.DocumentID,

         SimilarityScore: 1.0,

         MatchDetail:    "Exact content match (SHA-256)",

      }, nil

  }



  // Tier 2: SimHash near-exact match

  if matches, err := ds.fingerprintDB.FindBySimHash(ctx, doc.SimHash, 3); err == nil &&
len(matches) > 0 {

      // Hamming distance <= 3 means very similar content

      return &DuplicateResult{

         DocumentID:      doc.DocumentID,

         IsDuplicate:   true,

         DuplicateType: DuplicateTypeNearExact,

         MatchedDocID: matches[0].DocumentID,

         SimilarityScore: 0.95 + float64(3-matches[0].Distance)*0.015,

     MatchDetail: fmt.Sprintf("Near-exact match (SimHash distance: %d)",
matches[0].Distance),

      }, nil

  }



  // Tier 3: Semantic similarity via vector search

  similarDocs, err := ds.vectorDB.FindSimilar(ctx,

      doc.BodyEmbedding,

      ds.similarityThreshold,

      5, // Top-K

  )

  if err != nil {

          return nil, fmt.Errorf("vector similarity search: %w", err)

      }



      for _, match := range similarDocs {

          if match.Similarity > 0.92 {

               return &DuplicateResult{

                  DocumentID:      doc.DocumentID,

                  IsDuplicate:   true,

                  DuplicateType: DuplicateTypeSemantic,

                  MatchedDocID: match.DocumentID,

                  SimilarityScore: match.Similarity,

                  MatchDetail:    fmt.Sprintf("Semantic similarity: %.2f", match.Similarity),

               }, nil

          }

      }



      // Store fingerprints for future comparisons

      ds.fingerprintDB.Store(ctx, doc)



      return &DuplicateResult{

          DocumentID: doc.DocumentID,

          IsDuplicate: false,

      }, nil

func (vt *VelocityTracker) CalculateVelocity(ctx context.Context, clusterID string) float64 {

    now := time.Now()



    // Get document count in sliding windows

    pipe := vt.redis.Pipeline()



    // Last 5 minutes

    fiveMinKey := fmt.Sprintf("cluster:%s:docs:5min", clusterID)

    pipe.ZCount(ctx, fiveMinKey,

        fmt.Sprintf("%d", now.Add(-5*time.Minute).Unix()),

        fmt.Sprintf("%d", now.Unix()),

    )



    // Last 15 minutes

    fifteenMinKey := fmt.Sprintf("cluster:%s:docs:15min", clusterID)

    pipe.ZCount(ctx, fifteenMinKey,

        fmt.Sprintf("%d", now.Add(-15*time.Minute).Unix()),

        fmt.Sprintf("%d", now.Unix()),

      )



      // Last 60 minutes

      sixtyMinKey := fmt.Sprintf("cluster:%s:docs:60min", clusterID)

      pipe.ZCount(ctx, sixtyMinKey,

          fmt.Sprintf("%d", now.Add(-60*time.Minute).Unix()),

          fmt.Sprintf("%d", now.Unix()),

      results, _ := pipe.Exec(ctx)



      count5min := results[0].(*redis.IntCmd).Val()

      count15min := results[1].(*redis.IntCmd).Val()

      count60min := results[2].(*redis.IntCmd).Val()

      // Normalize to documents per minute

       MATCH (s:Story {story_id: $storyID})

       MERGE (src:Source {source_id: $sourceID})

       ON CREATE SET src.name = $sourceName

       CREATE (s)-[:SOURCED_FROM {

          document_id: $documentID,

          relevance: $relevance,

          contributed_at: datetime($timestamp)

       }]->(src)

     `, map[string]interface{}{

       "storyID":    story.StoryID,

       "sourceID": doc.SourceID,

       "sourceName": doc.SourceName,

       "documentID": doc.DocumentID,

         "relevance": 1.0,

         "timestamp": time.Now().Format(time.RFC3339),

    })

    if err != nil {

         return nil, fmt.Errorf("create source relationship: %w", err)

    }

}



// Create INVOLVES_ENTITY relationships

for _, entity := range cluster.Entities {

    _, err := tx.Run(ctx, `

         MATCH (s:Story {story_id: $storyID})

         MERGE (e:Entity {canonical_name: $canonicalName})

         ON CREATE SET e.name = $name, e.entity_type = $entityType

         CREATE (s)-[:INVOLVES_ENTITY {role: $role}]->(e)

    `, map[string]interface{}{

         "storyID":    story.StoryID,

         "canonicalName": entity.CanonicalName,

         "name":       entity.Name,

         "entityType": entity.Type,

         "role":      "SUBJECT",

    })

    if err != nil {

         return nil, fmt.Errorf("create entity relationship: %w", err)

    }

}

           return nil, nil

      })



      if err != nil {

           return nil, fmt.Errorf("execute neo4j transaction: %w", err)

      }



      // 4. Register story in PostgreSQL

      if err := sgi.storyRegistry.Register(ctx, story); err != nil {

           return nil, fmt.Errorf("register story: %w", err)

      }



      // 5. Publish event

      sgi.eventProducer.Publish(ctx, &events.StoryCreatedEvent{

           BaseEvent: events.NewBaseEvent(story.TenantID),

           StoryID: story.StoryID,

           ClusterID: cluster.ClusterID,

           Headline: story.Headline,

           Priority: string(story.Priority),

           State:   string(story.State),

      })



      return story, nil

}
