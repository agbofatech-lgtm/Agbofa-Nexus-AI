ARCHITECTURE DECISION RECORDS (ADR)



ADR-RUN-001: LangGraph-Inspired Workflow Engine for AI Orchestration



Decision: Implement a custom workflow engine in Golang inspired by LangGraph's state
machine and node-edge execution model, with native gRPC integration for AI agent
communication. LangGraph concepts are reimplemented in Golang for production-grade
performance and type safety.

Rationale: AI newsroom workflows (story creation, verification, generation, distribution) are
complex DAGs with conditional branching, parallel execution, and human-in-the-loop steps.
LangGraph's model is proven for AI workflows but Python-only. Reimplementing in Golang
provides 10x throughput and native Kubernetes integration.

Alternatives: Temporal (rejected — too heavy for sub-second agent calls, Python SDK
dependency); Apache Airflow (rejected — batch-oriented, not real-time); direct LangGraph in
Python (rejected — performance limits at 500K+ stories/day).

Trade-offs: Build vs. buy — but Golang implementation provides 10x performance and full type
safety.



ADR-RUN-002: Event-Sourced Workflow State with Kafka



Decision: All workflow state transitions are recorded as events in Kafka. Workflow state is
materialized in Redis for fast access and PostgreSQL for durability. Checkpoints enable
pause/resume and failure recovery.

Rationale: Workflows span multiple services and can run for hours (long-form article
generation) or seconds (breaking news). Event sourcing provides complete audit trail, enables
replay for recovery, and supports real-time monitoring.

Alternatives: Database-only state (rejected — no replay, harder recovery); Redis-only (rejected
— durability concerns).

Trade-offs: Additional Kafka topics, but essential for workflow reliability.



ADR-RUN-003: Service Mesh with Istio Ambient for Zero-Trust



Decision: Istio Ambient Mesh (sidecar-less) for all service-to-service communication. mTLS
enforced by default. OPA policies at the mesh layer for authorization.

Rationale: 60+ microservices require zero-trust networking. Ambient mesh provides L4/L7
security without sidecar overhead. SPIFFE/SPIRE for workload identity.

Alternatives: Traditional Istio sidecars (rejected — 40-60% resource overhead); Linkerd (rejected
— fewer enterprise features).

Trade-offs: Requires Kubernetes 1.29+, but significant resource savings vs sidecar model.



ADR-RUN-004: Multi-Region Active-Active with Global Traffic Management



Decision: Three-region active-active deployment (us-east-1, eu-west-1, ap-southeast-1) with
Route 53 latency-based routing and Aurora Global Database for cross-region data replication.

Rationale: Autonomous newsroom requires 99.99% availability with sub-second failover. Multi-
region active-active eliminates single-region failure risk and provides data sovereignty
compliance.

Alternatives: Active-passive (rejected — failover latency unacceptable for breaking news);
single-region multi-AZ (rejected — insufficient for global audience).

Trade-offs: Higher infrastructure cost, cross-region data replication complexity.

```go
// services/runtime/workflow-engine/internal/domain/workflow.go

package domain



// WorkflowDefinition defines a complete workflow DAG

type WorkflowDefinition struct {

    ID        string           `json:"id"`

    Name          string         `json:"name"`

    Version       string         `json:"version"`

    Description     string         `json:"description"`

    Nodes         []WorkflowNode        `json:"nodes"`

    Edges         []WorkflowEdge       `json:"edges"`

    StartNode       string         `json:"start_node"`

    EndNodes        []string        `json:"end_nodes"`

    TimeoutSeconds int                `json:"timeout_seconds"`

    RetryPolicy    *RetryPolicy        `json:"retry_policy,omitempty"`

    FailureStrategy FailureStrategy      `json:"failure_strategy"`

}



// WorkflowNode represents a single step in the workflow

type WorkflowNode struct {

    ID        string           `json:"id"`

    Name          string         `json:"name"`

    Type          NodeType          `json:"type"`

    Service       string         `json:"service"`       // gRPC service to call

    Method         string          `json:"method"`        // RPC method name

    Input         map[string]interface{} `json:"input"`        // Static inputs

    Timeout        time.Duration        `json:"timeout"`

    RetryPolicy      *RetryPolicy       `json:"retry_policy,omitempty"`

    IsHumanTask        bool           `json:"is_human_task"`

    HumanTaskConfig *HumanTaskConfig `json:"human_task_config,omitempty"`

    CheckpointAfter bool               `json:"checkpoint_after"` // Save state after completion

}



type NodeType string

const (

    NodeTypeService NodeType = "SERVICE"               // gRPC service call

    NodeTypeParallel NodeType = "PARALLEL" // Fan-out parallel execution

    NodeTypeDecision NodeType = "DECISION" // Conditional branching

    NodeTypeHuman             NodeType = "HUMAN"          // Human approval required

    NodeTypeSubWorkflow NodeType = "SUB_WORKFLOW" // Nested workflow

    NodeTypeWait         NodeType = "WAIT"         // Wait for external event

    NodeTypeTerminate NodeType = "TERMINATE" // End workflow

)



// WorkflowEdge defines a transition between nodes

type WorkflowEdge struct {

    From      string           `json:"from"`

    To      string            `json:"to"`

    Condition *EdgeCondition           `json:"condition,omitempty"` // Optional condition

    OnFailure bool               `json:"on_failure"`       // Edge for failure path

    OnTimeout bool                  `json:"on_timeout"`       // Edge for timeout path

}



type EdgeCondition struct {

    Field     string             `json:"field"`      // JSONPath to field in node output

    Operator string                `json:"operator"` // ==, !=, >, <, IN, CONTAINS

    Value      interface{}          `json:"value"`

}



// WorkflowInstance represents a running workflow

type WorkflowInstance struct {

    InstanceID     string             `json:"instance_id"`

    WorkflowID         string           `json:"workflow_id"`

    Version       string             `json:"version"`

    Status       WorkflowStatus           `json:"status"`

    CurrentNode         string          `json:"current_node"`

    Input        json.RawMessage            `json:"input"`

    Output        json.RawMessage            `json:"output,omitempty"`

    NodeOutputs         map[string]json.RawMessage `json:"node_outputs"`

    CheckpointData json.RawMessage                `json:"checkpoint_data,omitempty"`

    RetryCount         int            `json:"retry_count"`

    StartedAt      time.Time             `json:"started_at"`

    CompletedAt         *time.Time          `json:"completed_at,omitempty"`

    LastCheckpointAt *time.Time               `json:"last_checkpoint_at,omitempty"`

    ErrorMessage string                  `json:"error_message,omitempty"`

    TenantID       string             `json:"tenant_id"`

    StoryID       string             `json:"story_id,omitempty"`

}



type WorkflowStatus string

const (

      WorkflowStatusPending WorkflowStatus = "PENDING"

      WorkflowStatusRunning WorkflowStatus = "RUNNING"

      WorkflowStatusPaused       WorkflowStatus = "PAUSED"

      WorkflowStatusWaitingHuman WorkflowStatus = "WAITING_HUMAN"

      WorkflowStatusCompleted WorkflowStatus = "COMPLETED"

      WorkflowStatusFailed     WorkflowStatus = "FAILED"

      WorkflowStatusCancelled WorkflowStatus = "CANCELLED"

      WorkflowStatusRetrying WorkflowStatus = "RETRYING"

)



type HumanTaskConfig struct {

      Deadline      time.Duration      `json:"deadline"`
```

1.3 Workflow Execution Engine



```go

// services/runtime/workflow-engine/internal/domain/executor.go

package domain



// WorkflowExecutor executes workflow instances

type WorkflowExecutor struct {

    workflowRepo WorkflowRepository

    instanceStore InstanceStore

    serviceRegistry ServiceRegistry

    eventBus      EventBus

    checkpointMgr *CheckpointManager

    humanTaskMgr *HumanTaskManager

}



// StartWorkflow initiates a new workflow instance

func (we *WorkflowExecutor) StartWorkflow(

    ctx context.Context,

    workflowID string,

    input interface{},

    metadata WorkflowMetadata,

) (*WorkflowInstance, error) {



    // 1. Load workflow definition

    wf, err := we.workflowRepo.GetLatestVersion(ctx, workflowID)

if err != nil {

    return nil, fmt.Errorf("load workflow: %w", err)

}



// 2. Create instance

inputJSON, _ := json.Marshal(input)

instance := &WorkflowInstance{

    InstanceID: uuid.NewV7().String(),

    WorkflowID: workflowID,

    Version:      wf.Version,

    Status:       WorkflowStatusPending,

    CurrentNode: wf.StartNode,

    Input:        inputJSON,

    NodeOutputs: make(map[string]json.RawMessage),

    StartedAt: time.Now(),

    TenantID: metadata.TenantID,

    StoryID:      metadata.StoryID,

}



// 3. Persist instance

if err := we.instanceStore.Save(ctx, instance); err != nil {

    return nil, fmt.Errorf("save instance: %w", err)

}



// 4. Publish workflow started event

we.eventBus.Publish(ctx, &WorkflowStartedEvent{

          InstanceID: instance.InstanceID,

          WorkflowID: workflowID,

          StoryID:   metadata.StoryID,

          StartedAt: instance.StartedAt,

     })



     // 5. Begin execution

     go we.executeWorkflow(context.Background(), instance, wf)



     return instance, nil

}



// executeWorkflow runs the workflow DAG

func (we *WorkflowExecutor) executeWorkflow(

     ctx context.Context,

     instance *WorkflowInstance,

     wf *WorkflowDefinition,

){



     instance.Status = WorkflowStatusRunning

     we.instanceStore.Update(ctx, instance)



     currentNodeID := instance.CurrentNode



     for currentNodeID != "" {

          // Find current node

node := wf.GetNode(currentNodeID)

if node == nil {

    we.failWorkflow(ctx, instance, fmt.Errorf("node not found: %s", currentNodeID))

    return

}



// Check timeout

if wf.TimeoutSeconds > 0 {

    elapsed := time.Since(instance.StartedAt)

    if elapsed.Seconds() > float64(wf.TimeoutSeconds) {

        we.handleTimeout(ctx, instance, wf, node)

        return

    }

}



// Execute node

output, err := we.executeNode(ctx, instance, node)

if err != nil {

    we.handleNodeFailure(ctx, instance, wf, node, err)

    return

}



// Store output

instance.NodeOutputs[node.ID] = output



// Checkpoint if configured

if node.CheckpointAfter {

    we.checkpointMgr.SaveCheckpoint(ctx, instance, node.ID)

    instance.LastCheckpointAt = timePtr(time.Now())

}



// Determine next node

nextNode := we.determineNextNode(wf, node, output)

currentNodeID = nextNode



if nextNode == "" {

    // Workflow complete

    instance.Status = WorkflowStatusCompleted

    instance.Output = output

    now := time.Now()

    instance.CompletedAt = &now

    we.instanceStore.Update(ctx, instance)



    we.eventBus.Publish(ctx, &WorkflowCompletedEvent{

         InstanceID: instance.InstanceID,

         WorkflowID: instance.WorkflowID,

         CompletedAt: now,

         Duration: now.Sub(instance.StartedAt),

    })

    return

}

        instance.CurrentNode = currentNodeID

        we.instanceStore.Update(ctx, instance)

    }

}



// executeNode executes a single workflow node

func (we *WorkflowExecutor) executeNode(

    ctx context.Context,

    instance *WorkflowInstance,

    node *WorkflowNode,

) (json.RawMessage, error) {



    switch node.Type {

    case NodeTypeService:

        return we.executeServiceNode(ctx, instance, node)



    case NodeTypeParallel:

        return we.executeParallelNode(ctx, instance, node)



    case NodeTypeDecision:

        return we.executeDecisionNode(ctx, instance, node)



    case NodeTypeHuman:

        return we.executeHumanNode(ctx, instance, node)



    case NodeTypeSubWorkflow:

        return we.executeSubWorkflow(ctx, instance, node)



    case NodeTypeWait:

        return we.executeWaitNode(ctx, instance, node)



    default:

        return nil, fmt.Errorf("unknown node type: %s", node.Type)

    }

}



// executeServiceNode calls a gRPC service

func (we *WorkflowExecutor) executeServiceNode(

    ctx context.Context,

    instance *WorkflowInstance,

    node *WorkflowNode,

) (json.RawMessage, error) {



    // Resolve service endpoint

    endpoint, err := we.serviceRegistry.Resolve(node.Service)

    if err != nil {

        return nil, fmt.Errorf("resolve service %s: %w", node.Service, err)

    }



    // Build request from node input and previous outputs

    request := we.buildRequest(node.Input, instance.NodeOutputs)

    // Execute with timeout and retry

    var response json.RawMessage

    err = we.executeWithRetry(ctx, node, func(ctx context.Context) error {

         resp, callErr := endpoint.Call(ctx, node.Method, request)

         if callErr != nil {

             return callErr

         }

         response = resp

         return nil

    })



    if err != nil {

         return nil, fmt.Errorf("service call %s/%s: %w", node.Service, node.Method, err)

    }



    return response, nil

}



// executeParallelNode executes multiple nodes in parallel

func (we *WorkflowExecutor) executeParallelNode(

    ctx context.Context,

    instance *WorkflowInstance,

    node *WorkflowNode,

) (json.RawMessage, error) {



    // Parse parallel configuration

parallelConfig := node.Input["parallel"].([]interface{})



type parallelResult struct {

    BranchID string       `json:"branch_id"`

    Output json.RawMessage `json:"output"`

    Error string        `json:"error,omitempty"`

}



results := make(chan parallelResult, len(parallelConfig))



for _, branch := range parallelConfig {

    branchConfig := branch.(map[string]interface{})

    branchID := branchConfig["id"].(string)

    branchNodes := branchConfig["nodes"].([]interface{})



    go func(bID string, nodes []interface{}) {

      var lastOutput json.RawMessage

      for _, n := range nodes {

        nodeConfig := n.(map[string]interface{})

        subNode := &WorkflowNode{

            ID:   nodeConfig["id"].(string),

            Type: NodeTypeService,

            Service: nodeConfig["service"].(string),

            Method: nodeConfig["method"].(string),

            Input: nodeConfig["input"].(map[string]interface{}),

        }

                output, err := we.executeServiceNode(ctx, instance, subNode)

                if err != nil {

                    results <- parallelResult{BranchID: bID, Error: err.Error()}

                    return

                }

                lastOutput = output

            }

            results <- parallelResult{BranchID: bID, Output: lastOutput}

        }(branchID, branchNodes)

    }



    // Collect results

    parallelOutputs := make(map[string]json.RawMessage)

    for i := 0; i < len(parallelConfig); i++ {

        result := <-results

        if result.Error != "" {

            return nil, fmt.Errorf("parallel branch %s failed: %s", result.BranchID, result.Error)

        }

        parallelOutputs[result.BranchID] = result.Output

    }



    output, _ := json.Marshal(parallelOutputs)

    return output, nil

}

// executeHumanNode pauses workflow for human input

func (we *WorkflowExecutor) executeHumanNode(

  ctx context.Context,

  instance *WorkflowInstance,

  node *WorkflowNode,

) (json.RawMessage, error) {



  instance.Status = WorkflowStatusWaitingHuman

  we.instanceStore.Update(ctx, instance)



  // Create human task

  task := &HumanTask{

      TaskID:     uuid.NewV7().String(),

      InstanceID: instance.InstanceID,

      NodeID:      node.ID,

      TaskType:    node.HumanTaskConfig.TaskType,

      Title:     node.HumanTaskConfig.Title,

      Description: node.HumanTaskConfig.Description,

      Options:     node.HumanTaskConfig.Options,

      AssignedRole: node.HumanTaskConfig.AssignedRole,

      Deadline:    time.Now().Add(node.HumanTaskConfig.Deadline),

      Status:     HumanTaskStatusPending,

  }



  // Submit to human task manager

  if err := we.humanTaskMgr.CreateTask(ctx, task); err != nil {

           return nil, fmt.Errorf("create human task: %w", err)

      }



      // Publish event

      we.eventBus.Publish(ctx, &HumanTaskCreatedEvent{

           TaskID:      task.TaskID,

           InstanceID: instance.InstanceID,

           TaskType: task.TaskType,

           Deadline: task.Deadline,

      })



      // Wait for human response (via callback)

  response, err := we.humanTaskMgr.WaitForResponse(ctx, task.TaskID,
node.HumanTaskConfig.Deadline)

      if err != nil {

           return nil, fmt.Errorf("human task failed: %w", err)

      }



      instance.Status = WorkflowStatusRunning

      we.instanceStore.Update(ctx, instance)



      return json.Marshal(response)

}
```

1.5 Checkpoint & Recovery



```go

// services/runtime/workflow-engine/internal/domain/checkpoint.go

package domain

// CheckpointManager handles workflow state persistence and recovery

type CheckpointManager struct {

    redis *redis.Client

    pg   *pgxpool.Pool

}



type WorkflowCheckpoint struct {

    InstanceID    string        `json:"instance_id"`

    NodeID       string        `json:"node_id"`

    WorkflowState *WorkflowInstance `json:"workflow_state"`

    NodeOutputs       map[string]json.RawMessage `json:"node_outputs"`

    CreatedAt     time.Time        `json:"created_at"`

    Version      int64         `json:"version"`

}



// SaveCheckpoint persists workflow state for recovery

func (cm *CheckpointManager) SaveCheckpoint(

    ctx context.Context,

    instance *WorkflowInstance,

    nodeID string,

) error {



    checkpoint := &WorkflowCheckpoint{

      InstanceID: instance.InstanceID,

      NodeID:        nodeID,

        WorkflowState: instance,

        NodeOutputs: instance.NodeOutputs,

        CreatedAt:    time.Now(),

        Version:     instance.CurrentCheckpointVersion() + 1,

    }



    data, _ := json.Marshal(checkpoint)



    // Write to Redis for fast access

    cm.redis.Set(ctx,

        fmt.Sprintf("checkpoint:%s", instance.InstanceID),

        data,

        24*time.Hour,

    )



    // Write to PostgreSQL for durability

    _, err := cm.pg.Exec(ctx, `

        INSERT INTO workflow_checkpoints (instance_id, node_id, state, created_at, version)

        VALUES ($1, $2, $3, $4, $5)

        ON CONFLICT (instance_id, version) DO NOTHING

    `, instance.InstanceID, nodeID, data, checkpoint.CreatedAt, checkpoint.Version)



    return err

}



// RecoverFromCheckpoint restores a workflow from its last checkpoint

func (cm *CheckpointManager) RecoverFromCheckpoint(

  ctx context.Context,

  instanceID string,

) (*WorkflowInstance, error) {



  // Try Redis first

  data, err := cm.redis.Get(ctx, fmt.Sprintf("checkpoint:%s", instanceID)).Result()

  if err == nil {

      var checkpoint WorkflowCheckpoint

      json.Unmarshal([]byte(data), &checkpoint)

      return checkpoint.WorkflowState, nil

  }



  // Fall back to PostgreSQL

  row := cm.pg.QueryRow(ctx, `

      SELECT state FROM workflow_checkpoints

      WHERE instance_id = $1

      ORDER BY version DESC

      LIMIT 1

  `, instanceID)



  var stateJSON []byte

  if err := row.Scan(&stateJSON); err != nil {

      return nil, fmt.Errorf("no checkpoint found: %w", err)

  }

    var checkpoint WorkflowCheckpoint

    json.Unmarshal(stateJSON, &checkpoint)



    return checkpoint.WorkflowState, nil

}



// RecoverStalledWorkflows finds and recovers stalled workflows

func (cm *CheckpointManager) RecoverStalledWorkflows(ctx context.Context) ([]string, error) {

    // Find workflows that haven't been updated in the expected timeframe

    rows, err := cm.pg.Query(ctx, `

         SELECT instance_id

         FROM workflow_instances

         WHERE status IN ('RUNNING', 'WAITING_HUMAN')

         AND updated_at < now() - INTERVAL '10 minutes'

    `)

    if err != nil {

         return nil, err

    }

    defer rows.Close()



    var recoveredIDs []string

    for rows.Next() {

         var instanceID string

         rows.Scan(&instanceID)



         // Attempt recovery

          instance, err := cm.RecoverFromCheckpoint(ctx, instanceID)

          if err != nil {

              continue

          }



          // Resume from checkpoint

          instance.Status = WorkflowStatusRetrying

          // ... trigger re-execution



          recoveredIDs = append(recoveredIDs, instanceID)

      }



      return recoveredIDs, nil

}
```

```go
type AgentAutoScaler struct {

type AutoScalerConfig struct {

      MinInstances int               `json:"min_instances"`

      MaxInstances int               `json:"max_instances"`

      TargetLoad     float64         `json:"target_load"`     // Target load per agent (0-1)

    ScaleUpThreshold float64              `json:"scale_up_threshold"` // Queue depth or latency

    ScaleDownThreshold float64              `json:"scale_down_threshold"`

    CooldownPeriod time.Duration             `json:"cooldown_period"`

    AgentTypes          map[string]AgentScalingConfig `json:"agent_types"`

type AgentScalingConfig struct {

    MinInstances int          `json:"min_instances"`

    MaxInstances int           `json:"max_instances"`

    TargetLoad          float64 `json:"target_load"`

    Priority      int     `json:"priority"` // Higher priority agents scale first

}



// EvaluateScaling evaluates and executes scaling decisions

func (as *AgentAutoScaler) EvaluateScaling(ctx context.Context) error {



    for agentType, config := range as.config.AgentTypes {

      // Get current agent count and load

      agents, err := as.registry.ListByType(ctx, agentType)

      if err != nil {

          continue

      }



      activeAgents := 0

      totalLoad := 0.0

for _, agent := range agents {

    if agent.Status == AgentStatusReady || agent.Status == AgentStatusBusy {

        activeAgents++

        totalLoad += float64(agent.CurrentLoad) / float64(agent.MaxLoad)

    }

}



avgLoad := 0.0

if activeAgents > 0 {

    avgLoad = totalLoad / float64(activeAgents)

}



// Scale up if load is high

if avgLoad > config.TargetLoad*1.3 && activeAgents < config.MaxInstances {

    scaleCount := int(math.Ceil(float64(activeAgents) * 0.5)) // Add 50% more

    newCount := min(activeAgents+scaleCount, config.MaxInstances)



    as.scaleUp(ctx, agentType, newCount-activeAgents)

}



// Scale down if load is low

if avgLoad < config.TargetLoad*0.5 && activeAgents > config.MinInstances {

    scaleCount := max(1, int(float64(activeAgents)*0.3)) // Remove 30%

    newCount := max(activeAgents-scaleCount, config.MinInstances)



    as.scaleDown(ctx, agentType, activeAgents-newCount)

           }

      }



      return nil

}
```

```go

// services/runtime/event-platform/internal/domain/dlq.go

package domain



// DeadLetterQueueManager handles failed events

type DeadLetterQueueManager struct {

    kafka   kafka.Client

    redis   *redis.Client

    alertMgr AlertManager

}



type DLQMessage struct {

    MessageID      string        `json:"message_id"`

    OriginalTopic string        `json:"original_topic"`

    OriginalKey   string        `json:"original_key"`

    OriginalValue json.RawMessage       `json:"original_value"`

    ErrorMessage string           `json:"error_message"`

    ErrorType     string        `json:"error_type"`

    AttemptCount int             `json:"attempt_count"`

    SentToDLQAt     time.Time       `json:"sent_to_dlq_at"`

    NextRetryAt    *time.Time       `json:"next_retry_at,omitempty"`

    RequiresHuman bool             `json:"requires_human"`

    ResolutionNotes string         `json:"resolution_notes,omitempty"`

}



// SendToDLQ sends a failed event to the dead letter queue

func (dlq *DeadLetterQueueManager) SendToDLQ(

  ctx context.Context,

  originalTopic string,

  originalKey []byte,

  originalValue []byte,

  err error,

  attemptCount int,

) error {



  msg := &DLQMessage{

      MessageID:     uuid.NewV7().String(),

      OriginalTopic: originalTopic,

      OriginalKey: string(originalKey),

      OriginalValue: originalValue,

      ErrorMessage: err.Error(),

      ErrorType:   classifyError(err),

      AttemptCount: attemptCount,

      SentToDLQAt: time.Now(),

      RequiresHuman: attemptCount >= 5,

  }



  // Calculate exponential backoff retry

  if attemptCount < 5 {

      retryDelay := time.Duration(math.Pow(2, float64(attemptCount))) * time.Minute

      nextRetry := time.Now().Add(retryDelay)

      msg.NextRetryAt = &nextRetry

  }

    data, _ := json.Marshal(msg)



    // Send to DLQ topic

    if err := dlq.kafka.Produce(ctx, "dead_letter_queue", msg.MessageID, data); err != nil {

        return fmt.Errorf("send to DLQ: %w", err)

    }



    // Alert if high severity

    if msg.RequiresHuman || msg.ErrorType == "CRITICAL" {

        dlq.alertMgr.Trigger(ctx, Alert{

             Severity: "WARNING",

             Title: "Event sent to Dead Letter Queue",

             Detail: fmt.Sprintf("Topic: %s, Error: %s, Attempts: %d",

               originalTopic, err.Error(), attemptCount),

        })

    }



    return nil

}



// ReprocessDLQ attempts to reprocess dead letter queue messages

func (dlq *DeadLetterQueueManager) ReprocessDLQ(ctx context.Context) (int, error) {

    // Read messages due for retry

    messages, err := dlq.kafka.Consume(ctx, "dead_letter_queue")

    if err != nil {

    return 0, err

}



reprocessed := 0

for _, msg := range messages {

    var dlqMsg DLQMessage

    json.Unmarshal(msg.Value, &dlqMsg)



    // Skip if not due for retry

    if dlqMsg.NextRetryAt != nil && time.Now().Before(*dlqMsg.NextRetryAt) {

        continue

    }



    // Attempt to republish to original topic

    err := dlq.kafka.Produce(ctx,

        dlqMsg.OriginalTopic,

        []byte(dlqMsg.OriginalKey),

        dlqMsg.OriginalValue,

    )



    if err != nil {

        // Still failing — update retry count

        dlqMsg.AttemptCount++

        dlq.SendToDLQ(ctx,

          dlqMsg.OriginalTopic,

          []byte(dlqMsg.OriginalKey),

                  dlqMsg.OriginalValue,

                  err,

                  dlqMsg.AttemptCount,

              )

          } else {

              reprocessed++

          }

      }



      return reprocessed, nil

}
```

```go
type HealingAction struct {

    Action       HealingActionType `json:"action"`

    Status       HealingStatus      `json:"status"`

    CompletedAt *time.Time             `json:"completed_at,omitempty"`

const (

    HealingRestartPod      HealingActionType = "RESTART_POD"

    HealingScaleUp        HealingActionType = "SCALE_UP"

    HealingRollback       HealingActionType = "ROLLBACK"

    HealingFailover       HealingActionType = "FAILOVER"

    HealingClearCache       HealingActionType = "CLEAR_CACHE"

    HealingResetConnection HealingActionType = "RESET_CONNECTION"

    HealingRecycleWorker HealingActionType = "RECYCLE_WORKER"

)



// MonitorAndHeal continuously monitors and heals the platform

func (ah *AutoHealer) MonitorAndHeal(ctx context.Context) {

    ticker := time.NewTicker(15 * time.Second)

    defer ticker.Stop()



    for {

      select {

      case <-ticker.C:

            ah.evaluateAndHeal(ctx)

        case <-ctx.Done():

            return

        }

    }

}



func (ah *AutoHealer) evaluateAndHeal(ctx context.Context) {

    // 1. Check for crashed pods

    crashedPods := ah.healthMon.GetCrashedPods(ctx)

    for _, pod := range crashedPods {

        if pod.RestartCount > 5 {

            ah.executeHealing(ctx, &HealingAction{

                 TargetService: pod.Service,

                 Issue:       fmt.Sprintf("Pod %s crashed %d times", pod.Name, pod.RestartCount),

                 Action:      HealingRestartPod,

                 StartedAt:     time.Now(),

            })

        }

    }



    // 2. Check for memory leaks

    memoryLeakedPods := ah.healthMon.GetMemoryLeakedPods(ctx)

    for _, pod := range memoryLeakedPods {

        ah.executeHealing(ctx, &HealingAction{

            TargetService: pod.Service,

         Issue:       fmt.Sprintf("Memory usage %.1f%% exceeding limit", pod.MemoryUsage),

         Action:      HealingRecycleWorker,

         StartedAt:     time.Now(),

    })

}



// 3. Check for stale connections

staleConnections := ah.healthMon.GetStaleConnections(ctx)

for _, conn := range staleConnections {

    ah.executeHealing(ctx, &HealingAction{

         TargetService: conn.Service,

         Issue:       fmt.Sprintf("Stale connection pool: %d idle connections", conn.IdleCount),

         Action:      HealingResetConnection,

         StartedAt:     time.Now(),

    })

}



// 4. Check for degraded circuit breakers

openBreakers := ah.healthMon.GetOpenCircuitBreakers(ctx)

for _, breaker := range openBreakers {

    // If breaker has been open too long, attempt reset

    if breaker.OpenDuration > 10*time.Minute {

         ah.executeHealing(ctx, &HealingAction{

           TargetService: breaker.Service,

           Issue:       fmt.Sprintf("Circuit breaker open for %v", breaker.OpenDuration),

           Action:       HealingFailover,

                    StartedAt:   time.Now(),

               })

           }

      }

}
```
