ARCHITECTURE DECISION RECORDS (ADR)



ADR-ANA-001: Unified Analytics Data Lake



Decision: All analytics events flow into a unified data lake (S3 + Apache Iceberg) before being
processed into specialized stores (ClickHouse for real-time, PostgreSQL for operational, Neo4j
for relationship analytics).

Rationale: Single source of truth prevents data discrepancies. Iceberg provides schema
evolution and time travel. Specialized stores optimize for different query patterns.

Alternatives Considered: Direct write to multiple stores (rejected — data consistency issues);
single store for everything (rejected — performance trade-offs).



ADR-ANA-002: Real-Time + Batch Analytics Dual Path



Decision: Implement dual processing paths: sub-second real-time stream for dashboards/alerts,
hourly batch for deep analytics and ML training.

Rationale: Breaking news requires instant metrics; deep analysis requires historical context.
Separate paths optimize each use case.

Consequences: Additional infrastructure complexity but necessary for news platform
requirements.



ADR-ANA-003: Closed-Loop AI Optimization



Decision: All content performance data flows back to AI agents via a structured feedback
protocol, enabling continuous model improvement without manual intervention.

Rationale: AI-generated content quality improves only with feedback. Manual feedback doesn't
scale to 500,000+ stories/day.

Consequences: Requires careful feedback validation to prevent reward hacking.



ADR-ANA-004: Platform-Agnostic Analytics Abstraction



Decision: Normalize all platform metrics into a unified schema regardless of source platform
differences.

Rationale: Cross-platform comparison requires consistent metrics. Platform APIs change;
abstraction protects downstream systems.

Consequences: Some platform-specific nuance lost, but essential for unified intelligence.

```go
func (sm *PublishingStateMachine) Transition(event PublishingEvent, metadata interface{})
error {

    sm.mu.Lock()

    defer sm.mu.Unlock()



    validStates, exists := sm.transitions[sm.currentState]

    if !exists {

        return fmt.Errorf("no transitions defined from state: %s", sm.currentState)

    }

nextState, valid := validStates[event]

if !valid {

    return fmt.Errorf("invalid transition: %s -> %s", sm.currentState, event)

}



// Record transition

transition := StateTransition{

    FromState: sm.currentState,

    ToState: nextState,

    Event:    event,

    Timestamp: time.Now(),

}



if metadata != nil {

    transition.Metadata, _ = json.Marshal(metadata)

}



sm.stateHistory = append(sm.stateHistory, transition)

sm.currentState = nextState



// Persist state change as event

sm.eventBus.Publish("distribution.state.changed", StateChangedEvent{

    JobID:     sm.job.JobID,

    Transition: transition,

    CurrentState: nextState,

      })



      // Handle side effects based on new state

      sm.handleStateEntry(nextState, event)



      return nil

}



func (sm *PublishingStateMachine) handleStateEntry(state PublishingState, event
PublishingEvent) {

      switch state {

      case StateAdapting:

           go sm.startAdaptation()

      case StatePublishing:

           go sm.startPlatformPublishing()

      case StateRetrying:

           go sm.scheduleRetry()

      case StateRetracting:

           go sm.startRetraction()

      case StateFailed:

           sm.sendFailureAlert()

      case StateDeadLettered:

           sm.sendDLQAlert()

      }

}
```

```go
type RetryManager struct {

    policy     RetryPolicy

    dlq       *DeadLetterQueue

    metrics *MetricsStore

}



func (rm *RetryManager) ExecuteWithRetry(

    ctx context.Context,

    operation func(context.Context) error,

    jobID string,

) error {



    var lastErr error

delay := rm.policy.BaseDelay



for attempt := 0; attempt <= rm.policy.MaxRetries; attempt++ {

  if attempt > 0 {

      // Wait before retry

      select {

      case <-time.After(delay):

          delay = time.Duration(float64(delay) * rm.policy.BackoffMultiplier)

          if delay > rm.policy.MaxDelay {

              delay = rm.policy.MaxDelay

          }

      case <-ctx.Done():

          return ctx.Err()

      }



      rm.metrics.IncrementRetry(jobID)

  }



  err := operation(ctx)

  if err == nil {

      return nil

  }



  lastErr = err



  // Classify error

        errorType := rm.classifyError(err)



        if rm.isNonRetryable(errorType) {

            rm.sendToDLQ(jobID, err, attempt)

            return fmt.Errorf("non-retryable error: %w", err)

        }



        if !rm.isRetryable(errorType) {

            rm.sendToDLQ(jobID, err, attempt)

            return fmt.Errorf("unclassified error: %w", err)

        }

    }



    // Max retries exceeded

    rm.sendToDLQ(jobID, lastErr, rm.policy.MaxRetries)

    return fmt.Errorf("max retries (%d) exceeded: %w", rm.policy.MaxRetries, lastErr)

}



type DeadLetterQueue struct {

    kafka       kafka.Producer

    topic       string

    alertManager *AlertManager

}



func (dlq *DeadLetterQueue) Send(

    ctx context.Context,

  jobID string,

  err error,

  attemptCount int,

  jobData interface{},

) error {



  dlqMessage := &DLQMessage{

      MessageID: uuid.NewV7().String(),

      JobID:         jobID,

      Error:      err.Error(),

      AttemptCount: attemptCount,

      JobData:        jobData,

      SentToDLQAt: time.Now(),

      RequiresHuman: true,

  }



  data, _ := json.Marshal(dlqMessage)



  if err := dlq.kafka.Produce(ctx, &kafka.Message{

      Topic: dlq.topic,

      Key: []byte(jobID),

      Value: data,

  }); err != nil {

      return fmt.Errorf("DLQ write failed: %w", err)

  }

      // Alert on DLQ entries

      dlq.alertManager.TriggerAlert(ctx, AlertDLQMessage, map[string]interface{}{

           "job_id": jobID,

           "error":   err.Error(),

           "attempts": attemptCount,

      })



      return nil

}
```

```go
type MediaUploadJob struct {

type MediaVariant struct {

    Platform       string `json:"platform"`

    Variant       string `json:"variant"`

    URL           string `json:"url"`

    CDNURL          string `json:"cdn_url"`

    FileSize      int64         `json:"file_size"`

    Format         string `json:"format"`

    Width          int         `json:"width"`

    Height        int          `json:"height"`

    UploadedAt time.Time `json:"uploaded_at"`

    PlatformID string `json:"platform_id,omitempty"`

}



func (mup *MediaUploadPipeline) ProcessMedia(

    ctx context.Context,

    media *MediaAsset,

    platforms []string,

) (*MediaUploadJob, error) {



    job := &MediaUploadJob{

        JobID:           uuid.NewV7().String(),

        ContentID:         media.ContentID,

        OriginalMedia: media,

        Status:          MediaUploadStatusProcessing,

        Progress:         0,

    }

// 1. Determine required variants for each platform

for _, platform := range platforms {

    rules := GetPlatformRules(platform)

    variants := mup.determineVariants(media, rules)

    job.Variants = append(job.Variants, variants...)

}



totalVariants := len(job.Variants)



// 2. Process variants in parallel

var mu sync.Mutex

var wg sync.WaitGroup

results := make(chan MediaUploadResult, totalVariants)

semaphore := make(chan struct{}, 5) // Limit concurrent uploads



for _, variant := range job.Variants {

    wg.Add(1)

    go func(v MediaVariant) {

      defer wg.Done()

      semaphore <- struct{}{}

      defer func() { <-semaphore }()



      // Process variant

      processed, err := mup.processVariant(ctx, media, v)

      if err != nil {

    log.Error().Err(err).Str("platform", v.Platform).Msg("Variant processing failed")

    return

}



// Upload to CDN

cdnURL, err := mup.cdnUploader.Upload(ctx, processed)

if err != nil {

    log.Error().Err(err).Msg("CDN upload failed")

    return

}



result := MediaUploadResult{

    Platform: v.Platform,

    Variant: fmt.Sprintf("%dx%d", v.Dimensions.Width, v.Dimensions.Height),

    URL:      cdnURL,

    CDNURL:        cdnURL,

    FileSize: processed.Size,

    Format:       processed.Format,

    Width:        processed.Width,

    Height:       processed.Height,

    UploadedAt: time.Now(),

}



results <- result



mu.Lock()

            job.Progress = float64(len(job.Results)+1) / float64(totalVariants)

            mu.Unlock()

          }(variant)

      }



      wg.Wait()

      close(results)



      for result := range results {

          job.Results = append(job.Results, result)

      }



      job.Status = MediaUploadStatusComplete

      job.Progress = 1.0



      return job, nil

}
```

```go
func (rtae *RealTimeAnalyticsEngine) ProcessEvent(

    ctx context.Context,

    event *AnalyticsEvent,

) error {



  // 1. Write to ClickHouse

  if err := rtae.clickhouse.Insert(ctx, "analytics_events", event); err != nil {

      return fmt.Errorf("clickhouse insert: %w", err)

  }



  // 2. Update real-time counters in Redis

  pipeline := rtae.redis.Pipeline()



  // Increment total counters

  pipeline.HIncrBy(ctx, fmt.Sprintf("story:%s:metrics", event.StoryID),

      fmt.Sprintf("%s_%s", event.Platform, event.EventType), 1)



  // Update velocity counter (sliding window)

  now := time.Now().Unix()

  windowKey := fmt.Sprintf("story:%s:velocity:%d", event.StoryID, now/60) // Per-minute
window

  pipeline.Incr(ctx, windowKey)

  pipeline.Expire(ctx, windowKey, 5*time.Minute)



  // Update trending score

  pipeline.ZIncrBy(ctx, "trending:stories", 1, event.StoryID)



  if _, err := pipeline.Exec(ctx); err != nil {

      return fmt.Errorf("redis update: %w", err)

    }



    // 3. Check alerts

    rtae.checkAlerts(ctx, event)



    return nil

}
```

```python
  def analyze_story_performance(self, events) -> StoryPerformanceReport:

    """

    Analyze performance of all stories.

    """



    return events.groupBy("story_id", "content_type", "content_category") \

      .agg(

          F.countDistinct("user_id").alias("unique_readers"),

        F.sum(F.when(F.col("event_type") == "IMPRESSION",
1).otherwise(0)).alias("impressions"),

          F.sum(F.when(F.col("event_type") == "CLICK", 1).otherwise(0)).alias("clicks"),

        F.sum(F.when(F.col("event_type") == "ENGAGEMENT",
1).otherwise(0)).alias("engagements"),

          F.sum(F.when(F.col("event_type") == "SHARE", 1).otherwise(0)).alias("shares"),

          F.avg("read_depth").alias("avg_read_depth"),

          F.avg("time_on_page").alias("avg_time_on_page"),

          F.sum("revenue_impact").alias("total_revenue"),

      )\

      .withColumn("ctr", F.col("clicks") / F.col("impressions") * 100) \

      .withColumn("engagement_rate", F.col("engagements") / F.col("impressions") * 100) \

      .orderBy(F.col("engagements").desc())



  def analyze_viral_content(self, events) -> ViralContentReport:

    """

    Identify viral content patterns.

    """



    # Define viral threshold (top 1% by engagement velocity)

    velocity_window = Window.partitionBy("story_id").orderBy("event_timestamp")



    viral_candidates = events \

      .withColumn("engagement_count",

        F.when(F.col("event_type").isin(["LIKE", "SHARE", "COMMENT", "RETWEET"]),
1).otherwise(0)) \

      .withColumn("hour", F.window("event_timestamp", "1 hour")) \

      .groupBy("story_id", "hour") \

      .agg(F.sum("engagement_count").alias("hourly_engagements")) \

    # Identify viral patterns

          .filter(F.col("peak_velocity") > 100) # 100+ engagements/hour



      # Analyze common characteristics

      viral_characteristics = events \

          .filter(F.col("story_id").isin([r.story_id for r in viral_stories.collect()])) \

          .groupBy("content_type", "content_category", "content_archetype") \

          .agg(

              F.countDistinct("story_id").alias("viral_stories"),

              F.avg("read_depth").alias("avg_read_depth"),

          )



      return ViralContentReport(

          viral_stories=viral_stories,

          characteristics=viral_characteristics,

      )
```

```go
func (re *RecommendationEngine) GetRecommendations(

    ctx context.Context,

    userID string,

    context RecommendationContext,

    limit int,

) ([]*ContentRecommendation, error) {

// 1. Get user profile

userProfile, err := re.getUserProfile(ctx, userID)

if err != nil {

    // Fall back to non-personalized recommendations

    return re.getTrendingRecommendations(ctx, limit)

}



// 2. Multi-strategy recommendation blending

strategies := []RecommendationStrategy{

    &CollaborativeFiltering{weight: 0.35},

    &ContentBasedFiltering{weight: 0.30},

    &TrendingBoost{weight: 0.15},

    &FreshnessBoost{weight: 0.10},

    &ExplorationDiversification{weight: 0.10},

}



// 3. Generate candidates from each strategy

var allCandidates []*ContentRecommendation

var mu sync.Mutex

var wg sync.WaitGroup



for _, strategy := range strategies {

    wg.Add(1)

    go func(s RecommendationStrategy) {

      defer wg.Done()

      candidates, err := s.GetCandidates(ctx, userProfile, context, limit*2)

      if err != nil {

          return

      }



      // Apply strategy weight

      for _, c := range candidates {

          c.Score *= s.GetWeight()

      }



      mu.Lock()

      allCandidates = append(allCandidates, candidates...)

      mu.Unlock()

    }(strategy)

}



wg.Wait()



// 4. Merge and deduplicate

merged := re.mergeAndDeduplicate(allCandidates)



// 5. Re-rank with ML model

ranked, err := re.rankWithML(ctx, userProfile, merged, context)

if err != nil {

    // Fall back to score-based ranking

        sort.Slice(merged, func(i, j int) bool {

             return merged[i].Score > merged[j].Score

        })

        ranked = merged

    }



    // 6. Apply diversity and freshness constraints

    diversified := re.ensureDiversity(ranked)



    // 7. Apply business rules

    filtered := re.applyBusinessRules(ctx, diversified, userID)



    if len(filtered) > limit {

        filtered = filtered[:limit]

    }



    return filtered, nil

}



func (re *RecommendationEngine) rankWithML(

    ctx context.Context,

    userProfile *UserProfile,

    candidates []*ContentRecommendation,

    context RecommendationContext,

) ([]*ContentRecommendation, error) {

      // Load ranking model

      model, err := re.modelRegistry.LoadModel("content_ranking_v3")

      if err != nil {

           return nil, err

      }



      // Build feature vectors for each candidate

      features := make([]float64, len(candidates))



      for i, candidate := range candidates {

           featureVector := re.buildFeatureVector(userProfile, candidate, context)

           score, _ := model.Predict(featureVector)

           candidate.Score = score

      }



      // Sort by predicted score

      sort.Slice(candidates, func(i, j int) bool {

           return candidates[i].Score > candidates[j].Score

      })



      return candidates, nil

}
```

```go
type BanditSolver struct {

    algorithm string // THOMPSON_SAMPLING, UCB, EPSILON_GREEDY

}



func (bs *BanditSolver) SelectVariant(

    ctx context.Context,

    experiment *ContentExperiment,

) (*ExperimentVariant, error) {



    switch bs.algorithm {

    case "THOMPSON_SAMPLING":

        return bs.thompsonSampling(experiment)

    case "UCB":

        return bs.upperConfidenceBound(experiment)

    case "EPSILON_GREEDY":

        return bs.epsilonGreedy(experiment, 0.1)

    default:

        return bs.thompsonSampling(experiment)

    }

}



func (bs *BanditSolver) thompsonSampling(

    experiment *ContentExperiment,

) (*ExperimentVariant, error) {



    var bestVariant *ExperimentVariant

    var bestSample float64



    for i := range experiment.Variants {

        variant := &experiment.Variants[i]



        // Beta distribution sampling

        alpha := float64(variant.Conversions + 1)

        beta := float64(variant.Impressions - variant.Conversions + 1)



        sample := sampleBetaDistribution(alpha, beta)



        if sample > bestSample {

            bestSample = sample

            bestVariant = variant

        }

    }



    return bestVariant, nil

}



func (ee *ExperimentEngine) AnalyzeExperiment(

    ctx context.Context,

  experimentID string,

) (*ExperimentResults, error) {



  experiment, err := ee.experimentStore.Get(ctx, experimentID)

  if err != nil {

      return nil, err

  }



  // Calculate statistics for each variant

  for i := range experiment.Variants {

      variant := &experiment.Variants[i]



      // Calculate conversion rate

      if variant.Impressions > 0 {

          variant.ConversionRate = float64(variant.Conversions) / float64(variant.Impressions)

      }



      // Calculate confidence interval (Wilson score)

      variant.ConfidenceInterval = calculateWilsonCI(

          variant.Conversions, variant.Impressions, 0.95,

      )

  }



  // Find winner using statistical test

  baseline := &experiment.Variants[0] // First variant is control

  for i := 1; i < len(experiment.Variants); i++ {

          variant := &experiment.Variants[i]



          // Perform chi-squared test

          pValue := chiSquaredTest(

              variant.Conversions, variant.Impressions,

              baseline.Conversions, baseline.Impressions,

          )



          if pValue < 0.05 && variant.ConversionRate > baseline.ConversionRate {

              variant.IsWinner = true

              variant.ImprovementOverBaseline =

                (variant.ConversionRate - baseline.ConversionRate) / baseline.ConversionRate * 100

          }

      }



      results := &ExperimentResults{

          ExperimentID: experimentID,

          Variants:    experiment.Variants,

          DeclaredWinner: findWinner(experiment.Variants),

          StatisticalPower: calculateStatisticalPower(experiment),

          Recommendation: generateRecommendation(experiment),

      }



      return results, nil

}
```

```go
func (sm *StoryMonitor) calculateViralScore(

    ctx context.Context,

    metrics *StoryMetrics,

) float64 {



    // Viral score based on:

    // 1. Share velocity relative to baseline

    // 2. Engagement velocity acceleration

    // 3. External referrer diversity

    // 4. Comment sentiment velocity



    shareScore := normalizeScore(metrics.ShareVelocity, 0, 100)

    accelerationScore := normalizeScore(metrics.EngagementAcceleration, -10, 50)

    referrerScore := normalizeScore(float64(metrics.UniqueReferrers), 0, 50)

    sentimentScore := normalizeScore(metrics.PositiveSentimentRatio, 0, 1) * 100



    weights := map[string]float64{

      "share": 0.35,

      "acceleration": 0.30,

        "referrer": 0.20,

        "sentiment": 0.15,

    }



    viralScore := shareScore*weights["share"] +

        accelerationScore*weights["acceleration"] +

        referrerScore*weights["referrer"] +

        sentimentScore*weights["sentiment"]



    return math.Min(100, viralScore)

}

      }



      for _, cat := range evergreenCategories {

          if strings.Contains(result.Category, cat) {

          }

      }



      return false
```

```markdown

## ADR-ANA-001: Unified Analytics Data Lake

**Decision:** All events → S3/Iceberg → Specialized stores

**Rationale:** Single source of truth, schema evolution, time travel

**Trade-offs:** Additional storage layer, but prevents data discrepancies



## ADR-ANA-002: Real-Time + Batch Dual Path

**Decision:** Separate real-time (ClickHouse) and batch (Spark) processing

**Rationale:** Different latency requirements for dashboards vs ML training

**Trade-offs:** Infrastructure complexity, but optimal for each use case



## ADR-ANA-003: Closed-Loop AI Optimization

**Decision:** Automated performance feedback to AI agents

**Rationale:** Manual feedback doesn't scale; continuous improvement required

**Trade-offs:** Risk of reward hacking; mitigated by human oversight



## ADR-ANA-004: Platform-Agnostic Analytics

**Decision:** Normalize all platform metrics to unified schema

**Rationale:** Cross-platform comparison, API abstraction

**Trade-offs:** Some platform nuance lost, but essential for unified view



## ADR-ANA-005: Redis for Real-Time, ClickHouse for Analytics

**Decision:** Redis for sub-second counters, ClickHouse for analytical queries

**Rationale:** Different data access patterns require different stores

**Trade-offs:** Two systems to maintain, but optimal performance



## ADR-ANA-006: Thompson Sampling for Multi-Armed Bandit

**Decision:** Use Thompson Sampling for A/B test optimization

**Rationale:** Better exploration-exploitation balance than epsilon-greedy

**Trade-offs:** Slightly more complex implementation, but better convergence



## ADR-ANA-007: Event Sourcing for Experiment Tracking

**Decision:** All experiment events stored as immutable event stream

**Rationale:** Complete audit trail, replay capability, prevents data manipulation

**Trade-offs:** Storage overhead, but essential for experiment validity
```
