```go
func (sie *StoryIntelligenceEngine) AnalyzeStory(

    ctx context.Context,

    story *CanonicalStory,

) (*EnrichedStory, error) {



    enriched := &EnrichedStory{

        CanonicalStory: story,

    }



    // 1. Determine story archetype

    enriched.StoryArchetype = sie.classifyArchetype(story)



    // 2. Entity analysis with knowledge graph enrichment

    entityAnalysis, err := sie.entityAnalyzer.AnalyzeEntities(ctx, story.Entities)

if err != nil {

    return nil, fmt.Errorf("entity analysis: %w", err)

}

enriched.EntityAnalysis = *entityAnalysis



// 3. Find entity relationships

relationships, err := sie.knowledgeGraph.FindEntityRelationships(ctx, story.Entities)

if err != nil {

    return nil, fmt.Errorf("entity relationships: %w", err)

}

enriched.EntityRelationships = relationships



// 4. Topic detection

topics := sie.topicDetector.DetectTopics(story)



// 5. Context enrichment

historicalCtx, err := sie.contextEnricher.EnrichHistoricalContext(ctx, story, topics)

if err != nil {

    return nil, fmt.Errorf("historical context: %w", err)

}

enriched.HistoricalContext = historicalCtx



// 6. Find related stories

related, err := sie.knowledgeGraph.FindRelatedStories(ctx, story.StoryID, 10)

if err != nil {

    return nil, fmt.Errorf("related stories: %w", err)

}

enriched.RelatedStories = related



// 7. Background information

background, err := sie.contextEnricher.GatherBackground(ctx, story, topics)

if err != nil {

    return nil, fmt.Errorf("background info: %w", err)

}

enriched.BackgroundInfo = *background



// 8. Audience profiling

audiences := sie.audienceProfiler.ProfileAudiences(ctx, story, topics)

enriched.TargetAudiences = audiences

enriched.AudienceRelevance = sie.audienceProfiler.CalculateRelevance(audiences)



// 9. Newsworthiness scoring

enriched.NewsworthinessScore, enriched.NewsworthinessFactors =

    sie.newsworthinessCalc.Calculate(story, enriched)



// 10. Determine story angles

enriched.DominantAngle = sie.determineDominantAngle(story, enriched)

enriched.AlternativeAngles = sie.generateAlternativeAngles(story, enriched)



// 11. Content recommendations

enriched.ContentRecommendations = sie.generateContentRecommendations(enriched)

    return enriched, nil

}



func (sie *StoryIntelligenceEngine) classifyArchetype(story *CanonicalStory) StoryArchetype {

    // Classification logic based on story characteristics

    switch {

    case story.State == "BREAKING" && time.Since(story.CreatedAt) < 1*time.Hour:

        return ArchetypeBreakingNews

    case story.State == "DEVELOPING":

        return ArchetypeDevelopingStory

    case len(story.Entities) > 5 && story.Category == "POLITICS":

        return ArchetypeFeatureAnalysis

    case story.Category == "HUMAN_INTEREST":

        return ArchetypeHumanInterest

    case len(story.Claims) > 10:

        return ArchetypeInvestigative

    case story.Category == "SCIENCE" || story.Category == "TECHNOLOGY":

        return ArchetypeExplainer

    case len(story.Entities) == 1 && story.Entities[0].Type == "PERSON":

        return ArchetypeProfile

    case len(story.Claims) > 5 && story.Category == "DATA":

        return ArchetypeDataDriven

    default:

        return ArchetypeFeatureAnalysis

    }

}
```

2.3 Newsworthiness Calculator



```go

type NewsworthinessCalculator struct {

      weights NewsworthinessWeights

}



type NewsworthinessWeights struct {

      Timeliness float64 // 0.30

      Impact       float64 // 0.25

      Prominence float64 // 0.15

      Proximity     float64 // 0.10

      Conflict     float64 // 0.10

      HumanInterest float64 // 0.05

      Novelty       float64 // 0.05

}



type NewsworthinessFactor struct {

      Factor      string `json:"factor"`

      Score      float64 `json:"score"`     // 0-100

      Weight      float64 `json:"weight"`

      Explanation string `json:"explanation"`

}

func (nc *NewsworthinessCalculator) Calculate(

    story *CanonicalStory,

    enriched *EnrichedStory,

) (float64, []NewsworthinessFactor) {



    factors := []NewsworthinessFactor{

        nc.calculateTimeliness(story),

        nc.calculateImpact(story, enriched),

        nc.calculateProminence(story, enriched),

        nc.calculateProximity(story),

        nc.calculateConflict(story, enriched),

        nc.calculateHumanInterest(story, enriched),

        nc.calculateNovelty(story, enriched),

    }



    var totalScore float64

    for _, factor := range factors {

        totalScore += factor.Score * factor.Weight

    }



    return math.Round(totalScore*100) / 100, factors

}



func (nc *NewsworthinessCalculator) calculateImpact(

    story *CanonicalStory,

    enriched *EnrichedStory,

) NewsworthinessFactor {

  factor := NewsworthinessFactor{

      Factor: "Impact",

      Weight: nc.weights.Impact,

  }



  // Calculate impact based on affected population, geographic scope, severity

  impactScore := 0.0



  // Geographic scope impact

  switch story.GeographicScope {

  case "GLOBAL":

      impactScore += 40

  case "CONTINENTAL":

      impactScore += 30

  case "NATIONAL":

      impactScore += 20

  case "REGIONAL":

      impactScore += 10

  case "LOCAL":

      impactScore += 5

  }



  // Entity prominence impact

  for _, entity := range story.Entities {

      if entity.Prominence > 80 {

              impactScore += 20

          } else if entity.Prominence > 50 {

              impactScore += 10

          }

      }



      // Category impact

      highImpactCategories := map[string]float64{

          "EMERGENCY":         35,

          "HEALTH":      30,

          "FINANCIAL":     25,

          "POLITICS":    20,

          "ENVIRONMENT": 15,

      }

      if impact, exists := highImpactCategories[story.Category]; exists {

          impactScore += impact

      }



      factor.Score = math.Min(100, impactScore)

      factor.Explanation = fmt.Sprintf("Affects %s scope with %d prominent entities",

          story.GeographicScope, countProminentEntities(story.Entities))



      return factor

}
```

```go
func (sie *StoryIntelligenceEngine) recommendPlatforms(

      enriched *EnrichedStory,

) []PlatformTarget {

      platforms := []PlatformTarget{}



      for _, audience := range enriched.TargetAudiences {

          for _, platform := range audience.PreferredPlatforms {

              platforms = append(platforms, PlatformTarget{

                   Platform:    platform,

                   Priority:   audience.RelevanceScore > 70 ? "HIGH" : "MEDIUM",

                   Format:     sie.determinePlatformFormat(platform, enriched.StoryArchetype),

                   BestTime:    sie.determineOptimalTime(platform, audience),

              })

          }

      }



      return platforms

}
```

```python
# Python LLM orchestration for article generation

class ArticleGenerator:

      """

      Generates articles using LLM with structured prompting and validation.

      """



      def __init__(self, llm_gateway, prompt_templates, brand_voice, fact_checker):

        self.llm = llm_gateway

        self.prompts = prompt_templates

        self.brand_voice = brand_voice

        self.fact_checker = fact_checker



      async def generate_article(

  self,

  enriched_story: EnrichedStory,

  article_config: ArticleConfig,

) -> GeneratedArticle:

  """

  Generate a complete article from enriched story.

  """



  # 1. Generate article outline

  outline = await self.generate_outline(enriched_story, article_config)



  # 2. Generate sections in parallel

  sections = await self.generate_sections_parallel(

      enriched_story,

      outline,

      article_config

  )



  # 3. Generate headline variants

  headlines = await self.generate_headlines(enriched_story, article_config)



  # 4. Generate summary

  summary = await self.generate_summary(enriched_story, sections, article_config)



  # 5. Assemble full article

  article = self.assemble_article(

      enriched_story,

      headlines,

      summary,

      sections,

      article_config

  )



  # 6. Quality check

  article = await self.quality_check(article, enriched_story)



  return article



async def generate_outline(

  self,

  story: EnrichedStory,

  config: ArticleConfig,

) -> ArticleOutline:

  """

  Generate structured article outline.

  """



  prompt = self.prompts.get_outline_prompt(

      story=story,

      archetype=story.story_archetype,

      length=config.target_length,

      tone=config.tone,

      angle=config.angle

  )



  response = await self.llm.generate(

      prompt=prompt,

      model=config.model or "claude-3-opus",

      temperature=0.3, # Lower temperature for structure

      max_tokens=2000,

      response_format={"type": "json_object"} # Structured output

  )



  outline = ArticleOutline.parse(response)



  # Validate outline structure

  self.validate_outline(outline, config)



  return outline



async def generate_sections_parallel(

  self,

  story: EnrichedStory,

  outline: ArticleOutline,

  config: ArticleConfig,

) -> List[ArticleSection]:

  """

  Generate article sections in parallel for speed.

  """



  tasks = []

  for section_plan in outline.sections:

    task = self.generate_section(

        story=story,

        section_plan=section_plan,

        outline=outline,

        config=config

    )

    tasks.append(task)



  # Execute all section generations in parallel

  sections = await asyncio.gather(*tasks, return_exceptions=True)



  # Handle any failures

  for i, section in enumerate(sections):

    if isinstance(section, Exception):

        # Retry failed section

        sections[i] = await self.generate_section_with_retry(

            story, outline.sections[i], outline, config

        )



  return sections



async def generate_section(

  self,

  story: EnrichedStory,

  section_plan: SectionPlan,

  outline: ArticleOutline,

  config: ArticleConfig,

) -> ArticleSection:

  """

  Generate a single article section with fact verification.

  """



  # Build section prompt with context

  prompt = self.prompts.get_section_prompt(

      story=story,

      section_plan=section_plan,

      previous_sections=outline.get_previous_sections(section_plan.id),

      config=config

  )



  response = await self.llm.generate(

      prompt=prompt,

      model=config.model or "claude-3-opus",

      temperature=config.creativity or 0.7,

      max_tokens=section_plan.max_tokens or 1000,

  )



  section = ArticleSection(

      id=section_plan.id,

      heading=section_plan.heading,

      content=response.text,

      word_count=len(response.text.split()),

  )



  # Verify facts in section against canonical story

  verified_section = await self.fact_checker.verify_section_facts(

      section=section,

      source_story=story.canonical_story,

  )



  # Apply brand voice

  branded_section = self.brand_voice.apply(verified_section, config.tone)



  return branded_section



def assemble_article(

  self,

  story: EnrichedStory,

  headlines: List[Headline],

  summary: Summary,

  sections: List[ArticleSection],

  config: ArticleConfig,

) -> GeneratedArticle:

  """

Assemble all components into final article.

"""



# Sort sections by order

sections.sort(key=lambda s: s.order)



# Build full article HTML

article_html = self.build_article_html(

    headlines[0], # Primary headline

    summary,

    sections,

    config

)



# Build article metadata

metadata = ArticleMetadata(

    story_id=story.canonical_story.story_id,

    archetype=story.story_archetype,

    category=story.canonical_story.category,

    word_count=sum(s.word_count for s in sections),

    reading_time=self.calculate_reading_time(sections),

    generated_at=datetime.utcnow(),

    model=config.model,

    generation_version="2.1.0",

)

        return GeneratedArticle(

            article_id=str(uuid.uuid4()),

            story_id=story.canonical_story.story_id,

            headlines=headlines,

            summary=summary,

            sections=sections,

            full_content_html=article_html,

            metadata=metadata,

            related_stories=story.related_stories,

            seo_metadata=None, # Generated separately

        )
```

```go
    CharacterCount int `json:"character_count"`

func (hgs *HeadlineGenerationService) GenerateHeadlines(

  ctx context.Context,

  story *EnrichedStory,

  config *HeadlineConfig,

) ([]*HeadlineVariant, error) {



  // 1. Prepare prompt with story context

  prompt := hgs.buildHeadlinePrompt(story, config)



  // 2. Generate headlines via LLM

  response, err := hgs.llmGateway.Generate(ctx, LLMRequest{

       Prompt:      prompt,

       Model:       "claude-3-opus",

       Temperature: 0.4,

       MaxTokens: 500,

       ResponseFormat: "json",

  })

  if err != nil {

       return nil, fmt.Errorf("LLM generation: %w", err)

  }



  // 3. Parse and validate headlines

  headlines, err := hgs.parseHeadlines(response)

  if err != nil {

       return nil, fmt.Errorf("parse headlines: %w", err)

    }



    // 4. Score and rank headlines

    for _, headline := range headlines {

         headline.ReadabilityScore = hgs.calculateReadability(headline.Text)

         headline.ClickProbability = hgs.predictClickProbability(headline, story)

         headline.SEOKeywords = hgs.seoOptimizer.ExtractKeywords(headline.Text)

    }



    // 5. Filter by quality thresholds

    qualityHeadlines := hgs.filterByQuality(headlines, config)



    // 6. Sort by predicted performance

    sort.Slice(qualityHeadlines, func(i, j int) bool {

         return qualityHeadlines[i].ClickProbability > qualityHeadlines[j].ClickProbability

    })



    // 7. Register for A/B testing if applicable

    if config.ABTestGroup != "" {

         hgs.abTestManager.RegisterHeadlines(ctx, config.ABTestGroup, qualityHeadlines)

    }



    return qualityHeadlines, nil

}



func (hgs *HeadlineGenerationService) calculateReadability(text string) float64 {

    // Implement Flesch-Kincaid or similar readability metric

    words := strings.Fields(text)

    sentences := strings.Count(text, ".") + strings.Count(text, "!") + strings.Count(text, "?")



    if sentences == 0 || len(words) == 0 {

        return 0

    }



    syllables := hgs.countSyllables(text)



    // Flesch Reading Ease score

  score := 206.835 - 1.015*(float64(len(words))/float64(sentences)) -
84.6*(float64(syllables)/float64(len(words)))



    return math.Max(0, math.Min(100, score))

}



func (hgs *HeadlineGenerationService) predictClickProbability(

    headline *HeadlineVariant,

    story *EnrichedStory,

) float64 {

    // ML-based click prediction using headline features

    features := []float64{

        float64(len(headline.Text)),

        float64(strings.Count(headline.Text, "?")),

        float64(strings.Count(headline.Text, "!")),

          headline.ReadabilityScore,

          hgs.hasNumber(headline.Text),

          hgs.hasPowerWord(headline.Text),

          float64(len(headline.SEOKeywords)),

      }



      // Call prediction model

      probability := hgs.mlModel.Predict(features)



      return probability

}
```

```python
class BrandVoiceService:

      """

      Enforces editorial style guide and brand voice consistency.

      """

def __init__(self, style_guides, tone_analyzer, grammar_checker):

  self.style_guides = style_guides # Per-publication style guides

  self.tone_analyzer = tone_analyzer

  self.grammar_checker = grammar_checker

def apply_brand_voice(

  self,

  content: str,

  publication: str,

  content_type: str,

  target_tone: str,

) -> Tuple[str, VoiceComplianceReport]:

  """

  Apply brand voice to content and return compliance report.

  """



  report = VoiceComplianceReport()

  style_guide = self.style_guides[publication]



  # 1. Check and correct grammar

  grammar_issues = self.grammar_checker.check(content)

  report.grammar_issues = grammar_issues

  if grammar_issues:

  content = self.grammar_checker.correct(content)



# 2. Check style guide compliance

style_issues = self.check_style_compliance(content, style_guide)

report.style_issues = style_issues



# 3. Check tone alignment

tone_analysis = self.tone_analyzer.analyze(content)

report.tone_analysis = tone_analysis



if tone_analysis.dominant_tone != target_tone:

  # Adjust tone to match target

  content = self.adjust_tone(content, target_tone, tone_analysis)



# 4. Check for banned phrases

banned_found = self.check_banned_phrases(content)

report.banned_phrases = banned_found

for phrase in banned_found:

  content = self.replace_phrase(content, phrase, self.get_alternative(phrase))



# 5. Apply style-specific transformations

content = self.apply_style_transformations(content, content_type, style_guide)



# 6. Calculate overall compliance score

report.compliance_score = self.calculate_compliance(report)

  return content, report



def check_style_compliance(

  self,

  content: str,

  style_guide: StyleGuide,

) -> List[StyleIssue]:

  """

  Check content against style guide rules.

  """

  issues = []



  # Oxford comma check

  if style_guide.oxford_comma and not self.uses_oxford_comma(content):

     content = self.add_oxford_commas(content)

     issues.append(StyleIssue(

          rule="oxford_comma",

          severity="MEDIUM",

          fix_applied=True

     ))



  # Number formatting

  if style_guide.spell_out_numbers_under_10:

     content = self.format_numbers(content, style_guide)



  # Date formatting

  content = self.format_dates(content, style_guide.date_format)



  # Quote attribution style

  content = self.format_quotes(content, style_guide.quote_style)



  # Headline capitalization

  if style_guide.headline_case:

     content = self.apply_headline_case(content)



  # Abbreviation rules

  content = self.expand_abbreviations(content, style_guide)



  # Profanity/appropriateness check

  profanity_issues = self.check_profanity(content, style_guide.profanity_policy)

  issues.extend(profanity_issues)



  return issues



def adjust_tone(

  self,

  content: str,

  target_tone: str,

  current_analysis: ToneAnalysis,

) -> str:

  """

  Adjust content tone to match target.

  """

  tone_profile = self.tone_profiles[target_tone]



  adjustment_prompt = f"""

  Adjust the following text to match the target tone profile.



  CURRENT TONE: {current_analysis.dominant_tone} ({current_analysis.confidence})

  TARGET TONE: {target_tone}

  TARGET CHARACTERISTICS: {tone_profile.characteristics}



  RULES:

  - Maintain all factual information exactly

  - Preserve quotes verbatim

  - Adjust vocabulary, sentence structure, and pacing

  - Do not change the meaning or add new information



  TEXT: {content}



  ADJUSTED TEXT:

  """



  adjusted = self.llm.generate(adjustment_prompt)

  return adjusted



def calculate_compliance(self, report: VoiceComplianceReport) -> float:

  """

      Calculate overall brand voice compliance score.

      """

      score = 100.0



      # Grammar deductions

      score -= len(report.grammar_issues) * 5



      # Style deductions

      for issue in report.style_issues:

        if issue.severity == "HIGH":

            score -= 10

        elif issue.severity == "MEDIUM":

            score -= 5

        else:

            score -= 2



      # Banned phrase penalty

      score -= len(report.banned_phrases) * 10



      # Tone alignment bonus/penalty

      if report.tone_analysis.dominant_tone == report.target_tone:

        score += 5



      return max(0, min(100, score))
```

```python
class TranslationService:

  """

  Manages content translation with quality assurance.

  """



  SUPPORTED_LANGUAGES = {

def __init__(self, llm_gateway, terminology_mgr, quality_checker):

    self.llm = llm_gateway

    self.terminology = terminology_mgr

    self.quality = quality_checker

    self.glossaries = self.load_glossaries()



async def translate_article(

    self,

    article: GeneratedArticle,

    target_languages: List[str],

    translation_config: TranslationConfig,

) -> Dict[str, TranslatedArticle]:

    """

    Translate article into multiple target languages.

    """



    translations = {}



    # Translate in parallel

    tasks = []

    for lang in target_languages:

      task = self.translate_single_language(

            article=article,

          target_lang=lang,

          config=translation_config,

    )

    tasks.append((lang, task))



  for lang, task in tasks:

    try:

          translated = await task

          translations[lang] = translated

    except Exception as e:

          # Log error and continue with other languages

          log.error(f"Translation failed for {lang}: {e}")

          translations[lang] = self.create_failed_translation_record(article, lang, str(e))



  return translations



async def translate_single_language(

  self,

  article: GeneratedArticle,

  target_lang: str,

  config: TranslationConfig,

) -> TranslatedArticle:

  """

  Translate article to a single target language.

  """

# 1. Get language-specific terminology glossary

glossary = self.glossaries.get(target_lang, {})



# 2. Translate headline

translated_headlines = await self.translate_headlines(

    article.headlines,

    target_lang,

    glossary,

)



# 3. Translate summary

translated_summary = await self.translate_summary(

    article.summary,

    target_lang,

    glossary,

)



# 4. Translate article sections

translated_sections = await self.translate_sections(

    article.sections,

    target_lang,

    glossary,

    config,

)



# 5. Apply cultural adaptation

adapted_sections = await self.cultural_adaptation(

    translated_sections,

    target_lang,

    article.metadata.category,

)



# 6. Localize SEO metadata

localized_seo = await self.localize_seo(

    article.seo_metadata,

    target_lang,

)



# 7. Quality check

quality_report = await self.quality.check_translation(

    original=article,

    translated_headlines=translated_headlines,

    translated_summary=translated_summary,

    translated_sections=adapted_sections,

    target_lang=target_lang,

)



# 8. If quality below threshold, retry or flag for human review

if quality_report.overall_score < config.minimum_quality_score:

    if config.retry_on_low_quality:

      # Retry with more explicit instructions

      return await self.translate_single_language_retry(

              article, target_lang, config

          )

      else:

          quality_report.requires_human_review = True



  return TranslatedArticle(

      translation_id=str(uuid.uuid4()),

      original_article_id=article.article_id,

      target_language=target_lang,

      headlines=translated_headlines,

      summary=translated_summary,

      sections=adapted_sections,

      seo_metadata=localized_seo,

      quality_report=quality_report,

      translated_at=datetime.utcnow(),

  )



async def translate_sections(

  self,

  sections: List[ArticleSection],

  target_lang: str,

  glossary: Dict[str, str],

  config: TranslationConfig,

) -> List[ArticleSection]:

  """

  Translate article sections with terminology consistency.

    """



    # Build translation prompt with glossary

    glossary_text = "\n".join([

         f'"{src}" → "{tgt}"' for src, tgt in glossary.items()

    ])



    # Translate sections in batches for efficiency

    batch_size = 3

    translated_sections = []



    for i in range(0, len(sections), batch_size):

         batch = sections[i:i+batch_size]



         prompt = f"""

        Translate the following news article sections from English to
{self.SUPPORTED_LANGUAGES[target_lang]}.



         TERMINOLOGY GLOSSARY (must use exactly):

         {glossary_text}



         REQUIREMENTS:

         - Maintain factual accuracy exactly

         - Preserve all quotes (translate if appropriate, note if original language)

         - Maintain HTML structure

         - Adapt idioms culturally

    - Keep proper names unchanged unless translated version is standard

    - Target reading level: {config.target_reading_level}

    - Tone: {config.target_tone}



    SECTIONS TO TRANSLATE:

    {self.format_sections_for_translation(batch)}



    TRANSLATIONS:

    """



    response = await self.llm.generate(prompt, temperature=0.2)



    translated_batch = self.parse_translated_sections(response, batch)

    translated_sections.extend(translated_batch)



  return translated_sections



async def cultural_adaptation(

for section in sections:

      cultural_rules,

  )



  # Adjust measurements/units

      section.content,

      target_lang,

  )



  # Adjust dates/times

  section.content = self.localize_datetime(

      section.content,

      target_lang,

  )



  # Add local context if needed

  if cultural_rules.add_local_context:

      section.content = await self.add_local_context(
```

```python
class QualityAssurancePipeline:

async def run_full_qa(

  self,

  article: GeneratedArticle,

  source_story: CanonicalStory,

  qa_config: QAConfig,

) -> QAReport:

  """

  Run complete quality assurance on generated content.

  """



  report = QAReport(article_id=article.article_id)



  # Run all checks in parallel

  tasks = [

    self.check_grammar(article),

    self.check_fact_consistency(article, source_story),

    self.check_duplicates(article),

    self.check_brand_compliance(article),

    self.check_toxicity(article),

    self.check_bias(article),

    self.check_copyright(article),

    self.check_readability(article),

    self.check_quote_accuracy(article, source_story),

]



results = await asyncio.gather(*tasks, return_exceptions=True)



# Process results

for i, result in enumerate(results):

    if isinstance(result, Exception):

      report.add_error(f"Check {i} failed: {result}")

      continue

    report.add_check_result(result)



# Calculate overall quality score

report.calculate_overall_score()



# Determine if article passes QA

report.passes_qa = report.overall_score >= qa_config.minimum_pass_score



# Generate actionable feedback if failed

if not report.passes_qa:

    report.fix_suggestions = self.generate_fix_suggestions(report)

  return report



async def check_fact_consistency(

  self,

  article: GeneratedArticle,

  source_story: CanonicalStory,

) -> FactConsistencyReport:

  """

  Verify all facts in generated content match source story exactly.

  """



  # Extract all factual claims from article

  article_claims = await self.extract_claims(article.full_content)



  # Extract all factual claims from source story

  source_claims = source_story.claims



  inconsistencies = []



  for article_claim in article_claims:

    # Find matching source claim

    matching_source = self.find_matching_claim(article_claim, source_claims)



    if not matching_source:

          inconsistencies.append(FactInconsistency(

            claim=article_claim,

         issue_type="UNVERIFIED_CLAIM",

         severity="HIGH",

         description=f"Claim not found in verified source: {article_claim.text}",

    ))

    continue



  # Check for hallucinated details

  if self.has_hallucinated_details(article_claim, matching_source):

    inconsistencies.append(FactInconsistency(

         claim=article_claim,

         issue_type="HALLUCINATED_DETAILS",

         severity="CRITICAL",

         description="Added details not present in source",

         source_claim=matching_source,

    ))



# Check statistics accuracy

stat_inconsistencies = self.verify_statistics(article, source_story)

inconsistencies.extend(stat_inconsistencies)



# Check entity accuracy

entity_issues = self.verify_entities(article, source_story)

inconsistencies.extend(entity_issues)



score = self.calculate_fact_score(

  total_claims=len(article_claims),

      inconsistencies=len(inconsistencies),

  )



  return FactConsistencyReport(

      score=score,

      total_claims=len(article_claims),

      inconsistencies=inconsistencies,

      passes=score >= 95.0,

  )



async def check_bias(self, article: GeneratedArticle) -> BiasReport:

  """

  Detect potential bias in content.

  """



  biases = []



  # Check for political bias

  political_bias = await self.bias_detector.detect_political_bias(

      article.full_content

  )

  if political_bias.score > 0.3:

      biases.append(political_bias)



  # Check for gender bias

  gender_bias = await self.bias_detector.detect_gender_bias(

    article.full_content

)

if gender_bias.score > 0.2:

    biases.append(gender_bias)



# Check for racial/ethnic bias

racial_bias = await self.bias_detector.detect_racial_bias(

    article.full_content

)

if racial_bias.score > 0.2:

    biases.append(racial_bias)



# Check for confirmation bias

confirmation_bias = await self.bias_detector.detect_confirmation_bias(

    article.full_content,

    article.metadata.category,

)

if confirmation_bias.score > 0.3:

    biases.append(confirmation_bias)



# Check for source bias (over-reliance on certain sources)

source_bias = await self.bias_detector.detect_source_bias(

    article.sources_used,

)

if source_bias.score > 0.4:

    biases.append(source_bias)

  overall_bias_score = self.calculate_overall_bias(biases)



  return BiasReport(

      score=100 - overall_bias_score, # Invert so 100 = unbiased

      detected_biases=biases,

      passes=overall_bias_score < 20.0,

  )



async def check_copyright(self, article: GeneratedArticle) -> CopyrightReport:

  """

  Check for potential copyright issues.

  """



  issues = []



  # Check for plagiarized content

  plagiarism_check = await self.copyright_checker.check_plagiarism(

      article.full_content

  )

  if plagiarism_check.similarity_score > 0.3:

      issues.append(CopyrightIssue(

        type="PLAGIARISM",

        similarity_score=plagiarism_check.similarity_score,

        matching_sources=plagiarism_check.matching_sources,

        severity="HIGH" if plagiarism_check.similarity_score > 0.5 else "MEDIUM",

          ))



      # Check for excessive quoting

      quote_ratio = self.copyright_checker.calculate_quote_ratio(article)

      if quote_ratio > 0.3: # More than 30% is quotes

          issues.append(CopyrightIssue(

               type="EXCESSIVE_QUOTING",

               quote_ratio=quote_ratio,

               severity="MEDIUM",

          ))



      # Check image rights

      for image in article.media_attachments:

          if not image.has_proper_license:

               issues.append(CopyrightIssue(

                    type="IMAGE_RIGHTS",

                    image_url=image.url,

                    severity="HIGH",

               ))



      return CopyrightReport(

          issues=issues,

          passes=len(issues) == 0,

      )
```

11.1 Security Controls



```go

package security



// Content Factory Security

type ContentFactorySecurity struct {

      // Content integrity verification

      contentSigner ContentSigner

      integrityChecker IntegrityChecker



      // Access control

      accessController RBACController

    // Audit

    auditLogger AuditLogger



    // Copyright protection

    watermarkService WatermarkService

}



type ContentSigner struct {

    signingKey ed25519.PrivateKey

}



func (cs *ContentSigner) SignContent(content *GeneratedArticle) (*SignedContent, error) {

    // Create content hash

    contentHash := cs.hashContent(content)



    // Sign hash with private key

    signature := ed25519.Sign(cs.signingKey, contentHash)



    return &SignedContent{

       Article: content,

       ContentHash: hex.EncodeToString(contentHash),

       Signature: hex.EncodeToString(signature),

       SignedAt: time.Now(),

       SignerID: "content-factory-service",

    }, nil

}

func (cs *ContentSigner) VerifyContent(signed *SignedContent) (bool, error) {

      // Recalculate content hash

      recalculatedHash := cs.hashContent(signed.Article)



      // Verify hash matches

      if hex.EncodeToString(recalculatedHash) != signed.ContentHash {

          return false, ErrContentTampered

      }



      // Verify signature

      publicKey := cs.getPublicKey(signed.SignerID)

      signature, _ := hex.DecodeString(signed.Signature)



      return ed25519.Verify(publicKey, recalculatedHash, signature), nil

}
```

PART XII — ARCHITECTURE DECISION RECORDS



```markdown

## ADR-CF-001: Multi-Stage Content Generation Pipeline

**Decision:** Implement content generation as a staged pipeline: Story Intelligence → Content
Planning → Parallel Generation → Assembly → QA.

**Rationale:** Staged approach allows optimization at each stage, parallel generation improves
speed, and clear separation enables independent scaling.

**Consequences:** Increased architectural complexity but significant quality and performance
benefits.



## ADR-CF-002: LLM-Agnostic Gateway

**Decision:** Route all AI generation through a model-agnostic gateway supporting Claude,
GPT, Gemini, and open-source models.

**Rationale:** Avoid vendor lock-in, enable cost optimization, and allow model selection based
on task requirements.

**Consequences:** Additional abstraction layer but provides flexibility and cost control.



## ADR-CF-003: Platform-Specific Content Templates

**Decision:** Use declarative platform rules and templates rather than hardcoded generation
logic.

**Rationale:** Platforms frequently change their requirements; declarative rules enable rapid
adaptation without code changes.

**Consequences:** Template management overhead but significantly faster platform
adaptation.



## ADR-CF-004: Mandatory QA Before Distribution

**Decision:** All generated content must pass QA pipeline before distribution; failures trigger
automatic retry or human review.

**Rationale:** Content quality is critical for brand reputation; automated QA catches issues
before publication.

**Consequences:** Added latency but essential quality safeguard.

## ADR-CF-005: Translation with Cultural Adaptation

**Decision:** Implement translation as a two-stage process: linguistic translation + cultural
adaptation.

**Rationale:** Direct translation without cultural context produces inappropriate content for
target audiences.

**Consequences:** Additional processing time but significantly better audience reception.



## ADR-CF-006: Content Versioning with Rollback

**Decision:** Maintain full version history of all generated content with ability to rollback.

**Rationale:** Enables audit trails, correction workflows, and continuous improvement.

**Consequences:** Storage overhead but essential for editorial integrity.



## ADR-CF-007: Cost-Optimized Model Selection

**Decision:** Use smaller, cheaper models for structured tasks (headlines, summaries) and
larger models for complex tasks (full articles, analysis).

**Rationale:** Optimize cost-performance ratio across the generation pipeline.

**Consequences:** Multiple model management but 40-60% cost reduction.



## ADR-CF-008: Real-Time Content Updates

**Decision:** Support real-time content updates for developing stories via incremental
regeneration.

**Rationale:** Developing stories require frequent updates; full regeneration is wasteful.

**Consequences:** Complex state management but essential for breaking news coverage.

```
