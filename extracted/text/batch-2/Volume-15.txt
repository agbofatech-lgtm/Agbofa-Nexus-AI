ARCHITECTURE DECISION RECORDS (ADR)



ADR-CF-001: Fact-Locking Before Generation



Decision: Implement a "fact-locking" phase where all verifiable facts from the Story Graph are
extracted and frozen before any content generation begins.

Rationale: Prevents hallucination and ensures all generated content variants remain factually
consistent with the source of truth.

Alternatives Considered: Real-time fact verification during generation (rejected due to latency
and inconsistency risks).



ADR-CF-002: Multi-Pass Generation with Editorial Scoring



Decision: Generate content in three passes: structural outline, full draft, editorial refinement.
Each pass is scored by an editorial quality model before proceeding.

Rationale: Multi-pass approach allows early detection of quality issues and reduces wasted
computation on poor generations.

Alternatives Considered: Single-pass generation with post-hoc QA (rejected due to higher
revision costs).

ADR-CF-003: Platform-Native Generation (Not Translation)



Decision: Generate platform-specific content natively using platform-aware prompts rather than
generating once and adapting.

Rationale: True platform-native content requires different narrative structures, not just
formatting changes. Native generation produces better engagement.

Alternatives Considered: Generate canonical version, then adapt (partially adopted for
efficiency in MVP).

package factlock



// FactLocker extracts and freezes all verifiable facts before generation

type FactLocker struct {

      storyGraph StoryGraphClient

      truthEngine TruthEngineClient

}

type LockedFactSet struct {

    StoryID         string         `json:"story_id"`

    LockedAt         time.Time           `json:"locked_at"`

    LockVersion        string           `json:"lock_version"`



    // Immutable Facts

    CoreFacts        []LockedFact          `json:"core_facts"`

    VerifiedClaims []LockedClaim             `json:"verified_claims"`

    AttributedQuotes []LockedQuote              `json:"attributed_quotes"`

    Statistics      []LockedStatistic `json:"statistics"`

    Entities        []LockedEntity       `json:"entities"`

    Timeline         []LockedTimelineEvent `json:"timeline"`

    SourceAttributions []SourceAttribution `json:"source_attributions"`



    // Generation Constraints

    RequiredAttributions []string           `json:"required_attributions"`

    ForbiddenPhrases         []string       `json:"forbidden_phrases"`

    RequiredDisclaimers []string            `json:"required_disclaimers"`

}



type LockedFact struct {

    FactID       string `json:"fact_id"`

    FactText     string `json:"fact_text"`

    FactType      string `json:"fact_type"`      // WHO, WHAT, WHERE, WHEN, WHY, HOW

    VeracityScore float64 `json:"veracity_score"`

    MustInclude bool `json:"must_include"` // Critical fact that must appear

    IsSensitive bool `json:"is_sensitive"` // Requires careful handling

}



type LockedQuote struct {

    QuoteID           string `json:"quote_id"`

    QuoteText string `json:"quote_text"`          // VERBATIM - never alter

    AttributedTo string `json:"attributed_to"`

    Attribution string `json:"attribution"`       // Exact attribution text

    Context       string `json:"context"`

    VerifiedBy string `json:"verified_by"`

}



func (fl *FactLocker) LockFacts(

    ctx context.Context,

    storyID string,

) (*LockedFactSet, error) {



    // 1. Retrieve canonical story from Story Graph

    story, err := fl.storyGraph.GetCanonicalStory(ctx, storyID)

    if err != nil {

        return nil, fmt.Errorf("retrieve story: %w", err)

    }



    // 2. Verify story is in verified state

    if story.VerificationStatus != "VERIFIED" {

        return nil, ErrStoryNotVerified

}



locked := &LockedFactSet{

    StoryID:   storyID,

    LockedAt: time.Now(),

    LockVersion: uuid.NewV7().String(),

}



// 3. Extract and lock core facts

locked.CoreFacts = fl.extractCoreFacts(story)



// 4. Lock verified claims exactly as verified

locked.VerifiedClaims = fl.lockClaims(story.Claims)



// 5. Lock quotes verbatim - they must NEVER be altered

locked.AttributedQuotes = fl.lockQuotes(story.Quotes)



// 6. Lock statistics with exact numbers

locked.Statistics = fl.lockStatistics(story.Statistics)



// 7. Lock entity references

locked.Entities = fl.lockEntities(story.Entities)



// 8. Lock timeline events in order

locked.Timeline = fl.lockTimeline(story.Timeline)

    // 9. Lock source attributions

    locked.SourceAttributions = fl.lockSourceAttributions(story.Sources)



    // 10. Generate generation constraints

    locked.RequiredAttributions = fl.generateRequiredAttributions(story)

    locked.ForbiddenPhrases = fl.generateForbiddenPhrases(story)

    locked.RequiredDisclaimers = fl.generateRequiredDisclaimers(story)



    // 11. Cryptographically sign the locked fact set

    locked.FactSetHash = fl.signFactSet(locked)



    return locked, nil

}



// Quote locking is especially critical - quotes must be preserved EXACTLY

func (fl *FactLocker) lockQuotes(quotes []Quote) []LockedQuote {

    locked := make([]LockedQuote, len(quotes))



    for i, quote := range quotes {

      locked[i] = LockedQuote{

        QuoteID:     quote.ID,

        QuoteText: quote.Text,        // VERBATIM

        AttributedTo: quote.AttributedTo,

        Attribution: quote.Attribution, // VERBATIM

        Context:     quote.Context,

        VerifiedBy: quote.VerifiedBy,

              // Add verification metadata

              OriginalSource: quote.OriginalSource,

              VerificationDate: quote.VerifiedAt,

              CannotBeParaphrased: true,      // CRITICAL FLAG

          }

      }



      return locked

}

class ArticleGenerationService:

      """

      Generates journalistic articles from locked fact sets.

      Uses multi-pass generation with editorial scoring at each stage.

      """

def __init__(self, llm_gateway, prompt_library, editorial_scorer, brand_voice):

  self.llm = llm_gateway

  self.prompts = prompt_library

  self.scorer = editorial_scorer

  self.brand_voice = brand_voice



async def generate_article(

  self,

  locked_facts: LockedFactSet,

  generation_plan: GenerationPlan,

) -> GeneratedArticle:

  """

  Three-pass article generation:

  Pass 1: Structure & outline

  Pass 2: Full draft

  Pass 3: Editorial refinement

  """



  # PASS 1: Generate article structure

  outline = await self.generate_outline(locked_facts, generation_plan)



  # Score outline quality

  outline_score = await self.scorer.score_outline(outline, generation_plan)

  if outline_score < generation_plan.min_outline_score:

    # Retry with adjusted parameters

  outline = await self.retry_outline(locked_facts, generation_plan, outline_score)



# PASS 2: Generate full draft

draft = await self.generate_draft(locked_facts, outline, generation_plan)



# Verify fact adherence

fact_adherence = await self.verify_fact_adherence(draft, locked_facts)

if not fact_adherence.passes:

  # Hallucination detected - must regenerate

  draft = await self.regenerate_with_fact_corrections(

      locked_facts, outline, generation_plan, fact_adherence

  )



# PASS 3: Editorial refinement

refined = await self.refine_article(draft, locked_facts, generation_plan)



# Apply brand voice

branded = await self.brand_voice.apply(refined, generation_plan.publication)



# Final quality score

final_score = await self.scorer.score_final_article(branded, generation_plan)



return GeneratedArticle(

  article_id=str(uuid.uuid4()),

  locked_fact_version=locked_facts.lock_version,

  outline=outline,

      draft=draft,

      refined=branded,

      quality_scores={

           "outline": outline_score,

           "fact_adherence": fact_adherence.score,

           "final": final_score,

      },

      generation_metadata={

           "model": generation_plan.model,

           "passes": 3,

           "tokens_used": self.get_token_count(),

           "generation_time_ms": self.get_generation_time(),

      },

  )



async def generate_draft(

  self,

  locked_facts: LockedFactSet,

  outline: ArticleOutline,

  plan: GenerationPlan,

) -> ArticleDraft:

  """

  Generate full article draft with strict fact constraints.

  """



  # Build generation prompt with locked facts

prompt = self.prompts.build_draft_prompt(

    locked_facts=locked_facts,

    outline=outline,

    plan=plan,

)



# The prompt explicitly includes:

# - All locked facts that MUST appear

# - All quotes that MUST be included VERBATIM

# - All required attributions

# - Forbidden phrases to avoid

# - Required disclaimers



response = await self.llm.generate(

    prompt=prompt,

    model=plan.model or "claude-3-opus",

    temperature=plan.temperature or 0.4,

    max_tokens=plan.max_tokens or 4000,

)



return ArticleDraft(

    sections=response.sections,

    word_count=response.word_count,

    quotes_included=response.quotes_used,

)

async def verify_fact_adherence(

  self,

  draft: ArticleDraft,

  locked_facts: LockedFactSet,

) -> FactAdherenceReport:

  """

  Verify that generated content doesn't deviate from locked facts.

  """



  violations = []



  # Check every locked fact appears (if marked MustInclude)

  for fact in locked_facts.CoreFacts:

    if fact.MustInclude:

          found, match_quality = self.find_fact_in_text(fact, draft.full_text)

          if not found:

            violations.append(FactViolation(

                 fact_id=fact.FactID,

                 violation_type="MISSING_REQUIRED_FACT",

                 severity="CRITICAL",

                 detail=f"Required fact not found: {fact.FactText[:100]}...",

            ))



  # Check all quotes are verbatim

  for quote in locked_facts.AttributedQuotes:

    if not self.verify_quote_verbatim(quote, draft.full_text):

    violations.append(FactViolation(

         quote_id=quote.QuoteID,

         violation_type="QUOTE_ALTERED",

         severity="CRITICAL",

         detail="Quote text was modified from verified original",

    ))



# Check for hallucinated facts (claims not in locked set)

extracted_claims = await self.extract_claims_from_text(draft.full_text)

for claim in extracted_claims:

  if not self.claim_in_locked_set(claim, locked_facts):

    violations.append(FactViolation(

         violation_type="HALLUCINATED_CLAIM",

         severity="CRITICAL",

         detail=f"Claim not in verified fact set: {claim[:100]}...",

    ))



# Check attributions present

for required_attr in locked_facts.RequiredAttributions:

  if required_attr not in draft.full_text:

    violations.append(FactViolation(

         violation_type="MISSING_ATTRIBUTION",

         severity="HIGH",

         detail=f"Required attribution missing: {required_attr}",

    ))

        # Check forbidden phrases absent

        for forbidden in locked_facts.ForbiddenPhrases:

            if forbidden.lower() in draft.full_text.lower():

              violations.append(FactViolation(

                   violation_type="FORBIDDEN_PHRASE_USED",

                   severity="HIGH",

                   detail=f"Forbidden phrase found: {forbidden}",

              ))



        passes = len([v for v in violations if v.severity == "CRITICAL"]) == 0



        return FactAdherenceReport(

            passes=passes,

            violations=violations,

            score=self.calculate_adherence_score(violations),

        )

class HeadlineGenerationService:

      """

Generates platform-optimized headlines with strict factual constraints.

"""



HEADLINE_RULES = {

  "primary": {

       "max_chars": 100,

       "must_include": ["key_entity", "key_action"],

       "must_not": ["clickbait", "question_headline_unless_investigative"],

       "tone": "informative_authoritative",

  },

  "seo": {

       "max_chars": 70,

       "must_include": ["primary_keyword_frontloaded", "key_entity"],

       "must_not": ["keyword_stuffing"],

       "tone": "search_optimized_informative",

  },

  "mobile": {

       "max_chars": 60,

       "must_include": ["key_entity"],

       "must_not": ["truncation_likely"],

       "tone": "concise_direct",

  },

  "social_twitter": {

       "max_chars": 100,

       "must_include": ["hook_element"],

       "must_not": ["misleading_truncation"],

         "tone": "engaging_conversational",

    },

    "push_notification": {

         "max_chars": 65,

         "must_include": ["urgency_signal_if_breaking", "key_takeaway"],

         "must_not": ["vague", "clickbait"],

         "tone": "urgent_direct",

    },

}



async def generate_headlines(

    self,

    locked_facts: LockedFactSet,

    requested_types: List[str],

) -> List[Headline]:

    """

    Generate headlines for each requested type.

    """



    headlines = []



    for headline_type in requested_types:

         rules = self.HEADLINE_RULES.get(headline_type)

         if not rules:

            continue

# Generate multiple variants

variants = await self.generate_variants(

     locked_facts=locked_facts,

     headline_type=headline_type,

     rules=rules,

     count=5, # Generate 5 candidates

)



# Score each variant

scored_variants = []

for variant in variants:

     scores = await self.score_headline(variant, locked_facts, rules)

     scored_variants.append((variant, scores))



# Select best variant

scored_variants.sort(key=lambda x: x[1].overall_score, reverse=True)

best_variant, best_scores = scored_variants[0]



headlines.append(Headline(

     headline_id=str(uuid.uuid4()),

     text=best_variant,

     type=headline_type,

     character_count=len(best_variant),

     scores=best_scores,

     alternatives=[v for v, _ in scored_variants[1:3]], # Keep top 2 alternatives

))

  return headlines



async def score_headline(

  self,

  headline: str,

  locked_facts: LockedFactSet,

  rules: dict,

) -> HeadlineScores:

  """

  Score headline across multiple dimensions.

  """



  scores = HeadlineScores()



  # 1. Factual accuracy - does it match locked facts?

  scores.factual_accuracy = await self.check_factual_accuracy(headline, locked_facts)



  # 2. Clickbait detection

  scores.clickbait_risk = self.detect_clickbait(headline)



  # 3. Entity inclusion

  scores.entity_coverage = self.check_entity_inclusion(headline, locked_facts)



  # 4. Readability

  scores.readability = self.calculate_readability(headline)

      # 5. Length compliance

      scores.length_score = 1.0 if len(headline) <= rules["max_chars"] else 0.5



      # 6. Click prediction (ML model)

      scores.predicted_ctr = await self.predict_ctr(headline, locked_facts)



      # Weighted overall score

      scores.overall_score = (

          scores.factual_accuracy * 0.35 +

          (1 - scores.clickbait_risk) * 0.25 +

          scores.entity_coverage * 0.15 +

          scores.readability * 0.10 +

          scores.length_score * 0.10 +

          scores.predicted_ctr * 0.05

      )



      return scores

go func() {

wg.Wait()



// PHASE 3: Send push notification immediately

// PHASE 4: Generate rapid content (<15 seconds total)

wg.Add(2)



go func() {

    content.ShortArticle = bng.generateShortArticle(ctx, lockedFacts)

}()

      go func() {

        defer wg.Done()

        content.SocialPosts = bng.generateSocialPosts(ctx, lockedFacts)

      }()



      wg.Wait()



      // PHASE 5: Generate full content (<30 seconds total)

      content.FullArticle = bng.generateFullArticle(ctx, lockedFacts)

      content.SEOMetadata = bng.generateSEO(ctx, lockedFacts, content.FullArticle)

class QualityAssurancePipeline:

      """

      Multi-stage quality assurance for all generated content.

      """



      def __init__(self):

        self.checks = [

            FactConsistencyCheck(),

            GrammarCheck(),

            HallucinationDetector(),

            ToxicityCheck(),

            BiasCheck(),

      CopyrightCheck(),

      BrandComplianceCheck(),

      ReadabilityCheck(),

      QuoteAccuracyCheck(),

      AttributionCheck(),

  ]



async def run_full_qa(

  self,

  content: GeneratedContent,

  locked_facts: LockedFactSet,

) -> QAReport:

  """

  Run all QA checks in parallel.

  """



  # Run all checks concurrently

  tasks = [check.run(content, locked_facts) for check in self.checks]

  results = await asyncio.gather(*tasks, return_exceptions=True)



  report = QAReport(content_id=content.id)



  critical_issues = []

  high_issues = []

  for i, result in enumerate(results):

    if isinstance(result, Exception):

          report.add_error(f"Check {self.checks[i].name} failed: {result}")

          continue



    report.add_check_result(result)



    if not result.passes:

          if result.severity == "CRITICAL":

            critical_issues.append(result)

          elif result.severity == "HIGH":

            high_issues.append(result)



  # Determine pass/fail

  report.passes_qa = len(critical_issues) == 0

  report.requires_human_review = len(high_issues) > 2



  # Generate overall score

  report.overall_score = self.calculate_overall_score(report)



  return report



async def fact_consistency_check(

  self,

  content: GeneratedContent,

  locked_facts: LockedFactSet,

) -> QACheckResult:

  """

  CRITICAL: Verify all facts match locked fact set exactly.

  """



  result = QACheckResult(check_name="fact_consistency")



  # Extract all claims from generated content

  generated_claims = await self.extract_claims(content.full_text)



  violations = []



  for claim in generated_claims:

    # Check if claim exists in locked facts

    match = self.find_matching_fact(claim, locked_facts)



    if not match:

        violations.append({

             "claim": claim.text,

             "issue": "HALLUCINATED_CLAIM",

             "severity": "CRITICAL",

        })

    elif not self.is_accurate_representation(claim, match):

        violations.append({

             "claim": claim.text,

             "expected": match.FactText,

               "issue": "FACT_DISTORTION",

               "severity": "CRITICAL",

          })



  result.passes = len(violations) == 0

  result.severity = "CRITICAL" if not result.passes else "PASS"

  result.details = {"violations": violations}



  return result



async def hallucination_detector(

  self,

  content: GeneratedContent,

  locked_facts: LockedFactSet,

) -> QACheckResult:

  """

  Detect AI-generated hallucinations beyond fact checking.

  """



  result = QACheckResult(check_name="hallucination")



  indicators = []



  # Check for fabricated quotes

  quotes = self.extract_quotes(content.full_text)

  for quote in quotes:

  if not self.quote_in_locked_facts(quote, locked_facts):

     indicators.append({

          "type": "FABRICATED_QUOTE",

          "text": quote[:100],

          "severity": "CRITICAL",

     })



# Check for non-existent entities

entities = self.extract_entities(content.full_text)

for entity in entities:

  if not self.entity_in_locked_facts(entity, locked_facts):

     indicators.append({

          "type": "HALLUCINATED_ENTITY",

          "entity": entity,

          "severity": "HIGH",

     })



# Check for invented statistics

statistics = self.extract_statistics(content.full_text)

for stat in statistics:

  if not self.statistic_matches_locked(stat, locked_facts):

     indicators.append({

          "type": "FABRICATED_STATISTIC",

          "statistic": stat,

          "severity": "CRITICAL",

     })

        result.passes = len(indicators) == 0

        result.severity = "CRITICAL" if not result.passes else "PASS"

        result.details = {"indicators": indicators}



        return result

package security



type ContentSecurityService struct {

    contentSigner ContentSigner

    watermarkService WatermarkService

// ContentSigner cryptographically signs all generated content

type ContentSigner struct {

    signingKey ed25519.PrivateKey

}



func (cs *ContentSigner) SignContent(content *GeneratedContent) (*SignedContent, error) {

    // Create canonical representation for signing

    canonical := cs.canonicalize(content)



    // Hash

    hash := sha256.Sum256(canonical)

    // Sign

    signature := ed25519.Sign(cs.signingKey, hash[:])



    return &SignedContent{

       Content:     content,

       ContentHash: hex.EncodeToString(hash[:]),

       Signature: hex.EncodeToString(signature),

       SignedAt: time.Now(),

       SignerID: "content-factory-v1",

    }, nil

}



// PromptProtector prevents prompt injection and data leakage

type PromptProtector struct{}



func (pp *PromptProtector) SanitizePromptInput(input string) (string, error) {

    // Remove prompt injection patterns

    injectionPatterns := []string{

       "ignore previous instructions",

       "ignore all previous",

       "system prompt:",

       "<<SYS>>",

       "you are now",

       "new instructions:",

       "override:",

    }



    sanitized := input

    for _, pattern := range injectionPatterns {

        if strings.Contains(strings.ToLower(sanitized), pattern) {

            return "", ErrPromptInjectionDetected

        }

    }



    return sanitized, nil

}



// WatermarkService embeds invisible watermarks for content tracking

type WatermarkService struct{}



func (ws *WatermarkService) EmbedWatermark(

    content string,

    metadata WatermarkMetadata,

) (string, error) {

    // Embed zero-width characters for text watermarking

    watermark := ws.encodeWatermark(metadata)



    // Insert watermark at strategic positions

    watermarked := ws.insertWatermark(content, watermark)



    return watermarked, nil

}
