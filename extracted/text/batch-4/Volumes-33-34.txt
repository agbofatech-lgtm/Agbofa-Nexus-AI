FRONTEND DECISION RECORDS (FDR)



FDR-FE-001: Next.js 15 App Router as Primary Framework



Decision: Next.js 15 with App Router for all frontend applications. Server Components by
default, Client Components only when necessary for interactivity.

Rationale: App Router provides nested layouts, streaming SSR, React Server Components, and
built-in performance optimizations. Server-first architecture reduces client-side JavaScript and
improves initial load performance.

Alternatives: Remix (rejected — smaller ecosystem); Vite + React Router (rejected — lacks SSR
streaming); Next.js Pages Router (rejected — legacy, no RSC support).

Trade-offs: Learning curve for RSC mental model, but significant performance and DX benefits.



FDR-FE-002: Tailwind CSS + Design Tokens for Styling



Decision: Tailwind CSS with CSS custom properties (design tokens) for the component library.
Glassmorphism aesthetic with dark mode as default, light mode optional.

Rationale: Tailwind provides utility-first consistency across 100+ components. Design tokens
enable future white-label theming. Glassmorphism conveys the AI-native, data-rich aesthetic
appropriate for an AI operating system.

Alternatives: CSS Modules (rejected — less consistent at scale); styled-components (rejected —
runtime overhead); MUI/Ant Design (rejected — not distinctive enough).

Trade-offs: Learning curve for Tailwind, but unmatched consistency at scale.



FDR-FE-003: Zustand + TanStack Query for State Management



Decision: Zustand for client-side state (UI state, user preferences), TanStack Query for server
state (API data, caching, real-time updates).

Rationale: Zustand is minimal, TypeScript-first, and performs well. TanStack Query handles
caching, background refetch, optimistic updates, and real-time synchronization out of the box.

Alternatives: Redux Toolkit (rejected — too verbose); Jotai (rejected — less mature ecosystem);
Apollo Client (rejected — GraphQL-only).

Trade-offs: Two libraries instead of one, but each is best-in-class for its domain.



FDR-FE-004: TipTap as Rich Text Editor

Decision: TipTap (ProseMirror-based) for all rich text editing, with custom AI collaboration
extensions.

Rationale: TipTap is extensible, headless, and TypeScript-first. Custom extensions enable AI-
assisted editing (suggestions, fact overlays, source attributions) that are impossible with off-the-
shelf editors.

Alternatives: Slate.js (rejected — less active development); Quill (rejected — limited
extensibility); Lexical (rejected — less mature).

Trade-offs: Custom extension development, but essential for AI editorial experience.

2.3 Middleware Strategy



```typescript

// apps/newsroom/middleware.ts

import { NextResponse } from 'next/server';

import type { NextRequest } from 'next/server';



export function middleware(request: NextRequest) {

    const { pathname } = request.nextUrl;



    // Auth routes — redirect to dashboard if authenticated

    if (pathname.startsWith('/login') || pathname.startsWith('/mfa')) {

      const token = request.cookies.get('access_token');

        if (token) {

            return NextResponse.redirect(new URL('/dashboard', request.url));

        }

        return NextResponse.next();

    }



    // Protected routes — redirect to login if not authenticated

    if (!pathname.startsWith('/_next') && !pathname.startsWith('/api')) {

        const token = request.cookies.get('access_token');

        if (!token) {

            return NextResponse.redirect(new URL('/login', request.url));

        }



        // Validate token expiry

        // Add tenant context header for backend routing

        const requestHeaders = new Headers(request.headers);

        requestHeaders.set('x-tenant-id', extractTenantFromToken(token.value));



        return NextResponse.next({

            request: { headers: requestHeaders },

        });

    }



    return NextResponse.next();

}

export const config = {

 matcher: ['/((?!_next/static|_next/image|favicon.ico|public).*)'],

};

```

export function useStory(storyId: string) {

    const queryClient = useQueryClient();

     queryKey: storyKeys.detail(storyId),

     queryFn: () => storyApi.getStory(storyId),



     // Subscribe to real-time updates via SSE

     onSuccess: () => {

          const eventSource = new EventSource(`/api/stories/${storyId}/stream`);

          eventSource.onmessage = (event) => {

// Optimistic update for story approval

export function useApproveStory() {

    const queryClient = useQueryClient();



    return useMutation({

     mutationFn: (storyId: string) => storyApi.approveStory(storyId),



     onMutate: async (storyId) => {

          await queryClient.cancelQueries({ queryKey: storyKeys.detail(storyId) });

          const previous = queryClient.getQueryData(storyKeys.detail(storyId));



          // Optimistic update

          queryClient.setQueryData(storyKeys.detail(storyId), (old: any) => ({

           ...old,

           editorialStatus: 'APPROVED',

          }));



          return { previous };

      },



      onError: (err, storyId, context) => {

          // Rollback on error

          queryClient.setQueryData(storyKeys.detail(storyId), context?.previous);

          toast.error('Failed to approve story');

      },



      onSettled: (storyId) => {

          queryClient.invalidateQueries({ queryKey: storyKeys.detail(storyId) });

      },

    });

}

6.1 Authentication Flow

```typescript

// packages/utils/src/auth/client.ts

import { jwtDecode } from 'jwt-decode';



interface TokenManager {

    getAccessToken: () => string | null;

    getRefreshToken: () => string | null;

    setTokens: (access: string, refresh: string) => void;

    clearTokens: () => void;

    isAuthenticated: () => boolean;

    refreshAccessToken: () => Promise<string>;

}



export const tokenManager: TokenManager = {

    getAccessToken: () => {

     if (typeof window === 'undefined') return null;

     return localStorage.getItem('access_token');

    },



    getRefreshToken: () => {

     return localStorage.getItem('refresh_token');

    },



    setTokens: (access: string, refresh: string) => {

     localStorage.setItem('access_token', access);

     localStorage.setItem('refresh_token', refresh);

},



clearTokens: () => {

 localStorage.removeItem('access_token');

 localStorage.removeItem('refresh_token');

 window.location.href = '/login';

},



isAuthenticated: () => {

 const token = tokenManager.getAccessToken();

 if (!token) return false;



 try {

     const decoded = jwtDecode<{ exp: number }>(token);

     return decoded.exp * 1000 > Date.now();

 } catch {

     return false;

 }

},



refreshAccessToken: async () => {

 const refresh = tokenManager.getRefreshToken();

 if (!refresh) throw new Error('No refresh token');



 const response = await fetch('/api/auth/refresh', {

     method: 'POST',

          headers: { 'Content-Type': 'application/json' },

          body: JSON.stringify({ refresh_token: refresh }),

      });



      if (!response.ok) {

          tokenManager.clearTokens();

          throw new Error('Token refresh failed');

      }



      const { access_token, refresh_token } = await response.json();

      tokenManager.setTokens(access_token, refresh_token);

      return access_token;

 },

};

```

// packages/api-client/src/client.ts

import { tokenManager } from '@agbofa/utils/auth';



const apiClient: typeof fetch = async (input, init) => {

 // Get current token

 let token = tokenManager.getAccessToken();

 // Add auth header

 const headers = new Headers(init?.headers);

 if (token) {

      headers.set('Authorization', `Bearer ${token}`);

 }



 let response = await fetch(input, { ...init, headers });



 // Auto-refresh on 401

 if (response.status === 401 && token) {

      try {

          token = await tokenManager.refreshAccessToken();

          headers.set('Authorization', `Bearer ${token}`);

          response = await fetch(input, { ...init, headers });

      } catch {

          tokenManager.clearTokens();

      }

 }



 return response;

};

import { clientsClaim } from 'workbox-core';

clientsClaim();



// Precache app shell

// API calls — Network first, fallback to cache

 new NetworkFirst({

  cacheName: 'api-cache',

  networkTimeoutSeconds: 5,

 })

// Static assets — Cache first

 ({ request }) => request.destination === 'image' || request.destination === 'font',

 new CacheFirst({ cacheName: 'static-assets' })

);



// Pages — Stale while revalidate

registerRoute(

 ({ request }) => request.mode === 'navigate',

 new StaleWhileRevalidate({ cacheName: 'pages' })

);



// Background sync for offline actions

self.addEventListener('sync', (event) => {

 if (event.tag === 'sync-pending-actions') {

      event.waitUntil(syncPendingActions());

 }

});



async function syncPendingActions() {

 const db = await openDB();

 const pendingActions = await db.getAll('pendingActions');



 for (const action of pendingActions) {

        try {

            await fetch(action.url, {

             method: action.method,

             headers: action.headers,

             body: action.body,

            });

            await db.delete('pendingActions', action.id);

        } catch (error) {

            console.error('Sync failed for action:', action.id, error);

        }

    }

}

// apps/newsroom/components/error-boundary.tsx

'use client';



import * as Sentry from '@sentry/nextjs';



interface ErrorBoundaryProps {

 children: React.ReactNode;

    fallback?: React.ReactNode;

    section?: string;

}



export function ErrorBoundary({ children, fallback, section }: ErrorBoundaryProps) {

    return (

      <Sentry.ErrorBoundary

         fallback={({ error, resetError }) => (

          fallback || (

              <div className="flex flex-col items-center justify-center p-8 glass rounded-lg">

               <AlertTriangleIcon className="h-12 w-12 text-accent-error mb-4" />

               <h2 className="text-xl font-semibold mb-2">Something went wrong</h2>

               <p className="text-text-secondary mb-4">

                {section ? `Error in ${section}` : 'An unexpected error occurred'}

               </p>

               <Button onClick={resetError}>Try Again</Button>

              </div>

          )

         )}

      >

         {children}

      </Sentry.ErrorBoundary>

    );

}
